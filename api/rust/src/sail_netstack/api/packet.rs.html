26    pub queue_count: usize,
27    pub headroom: usize,

34impl PacketCapabilities {
35    /// Rejects nonsensical platform declarations before the runner starts.
36    ///
37    /// # Errors
38    ///
39    /// Returns [`io::ErrorKind::InvalidInput`] when a required count is zero or
40    /// a GSO declaration cannot describe a segmented packet.
41    pub fn validate(self) -> io::Result<Self> {
42        if self.max_batch == 0 {
43            return Err(io::Error::new(
44                io::ErrorKind::InvalidInput,
45                "max_batch must be at least one",
46            ));
47        }
48        if self.queue_count == 0 {
49            return Err(io::Error::new(
50                io::ErrorKind::InvalidInput,
51                "queue_count must be at least one",
52            ));
53        }
54        if let Some(gso) = self.gso {
55            if gso.max_segments < 2 || gso.max_segment_size == 0 {
56                return Err(io::Error::new(
57                    io::ErrorKind::InvalidInput,
58                    "invalid GSO capabilities",
59                ));
60            }
61        }
62        Ok(self)
63    }
64}

82/// An owned packet and its explicit return identity.
83#[derive(Debug)]
84pub struct Packet {
85    token: PacketToken,
86    storage: Box<[u8]>,
87    start: usize,
88    len: usize,
89    lease: Option<BudgetLease>,
90}
91
92impl Packet {
93    #[must_use]
94    pub fn from_payload(token: PacketToken, headroom: usize, payload: &[u8]) -> Self {
95        let mut storage = vec![0; headroom];
96        storage.extend_from_slice(payload);
97        Self {
98            token,
99            storage: storage.into_boxed_slice(),
100            start: headroom,
101            len: payload.len(),
102            lease: None,
103        }
104    }
105
106    #[must_use]
107    pub fn from_arena(token: PacketToken, packet: ArenaPacket) -> Self {
108        let (storage, start, len, lease) = packet.into_parts();
109        Self {
110            token,
111            storage,
112            start,
113            len,
114            lease: Some(lease),
115        }
116    }
117
118    #[must_use]
119    pub const fn token(&self) -> PacketToken {
120        self.token
121    }
122
123    #[must_use]
124    pub const fn headroom(&self) -> usize {
125        self.start
126    }
127
128    #[must_use]
129    pub fn is_budgeted(&self) -> bool {
130        self.lease.is_some()
131    }
132
133    #[must_use]
134    pub fn payload(&self) -> &[u8] {
135        &self.storage[self.start..self.start + self.len]
136    }
137
138    #[must_use]
139    pub fn payload_mut(&mut self) -> &mut [u8] {
140        &mut self.storage[self.start..self.start + self.len]
141    }
142}

144/// Ownership-preserving batch. Successfully sent packets are removed from the
145/// front; a partial send leaves the unsent suffix in this batch.
146#[derive(Debug)]
147pub struct PacketBatch {
148    limit: usize,
149    packets: VecDeque<Packet>,
150}
151
152impl PacketBatch {
153    #[must_use]
154    /// # Panics
155    ///
156    /// Panics when `limit` is zero. A platform must advertise at least one
157    /// packet per batch.
158    pub fn with_limit(limit: usize) -> Self {
159        assert!(limit > 0, "packet batch limit must be non-zero");
160        Self {
161            limit,
162            packets: VecDeque::with_capacity(limit),
163        }
164    }
165
166    #[must_use]
167    pub fn len(&self) -> usize {
168        self.packets.len()
169    }
170
171    #[must_use]
172    pub fn is_empty(&self) -> bool {
173        self.packets.is_empty()
174    }
175
176    #[must_use]
177    pub const fn limit(&self) -> usize {
178        self.limit
179    }
180
181    /// # Errors
182    ///
183    /// Returns the unchanged packet when the batch is full.
184    pub fn push(&mut self, packet: Packet) -> Result<(), Packet> {
185        if self.packets.len() == self.limit {
186            return Err(packet);
187        }
188        self.packets.push_back(packet);
189        Ok(())
190    }
191
192    pub fn pop_front(&mut self) -> Option<Packet> {
193        self.packets.pop_front()
194    }
195
196    /// # Errors
197    ///
198    /// Returns [`io::ErrorKind::InvalidData`] if the platform claims to have
199    /// sent more packets than the batch contains.
200    pub fn acknowledge_sent(&mut self, count: usize) -> io::Result<()> {
201        if count > self.packets.len() {
202            return Err(io::Error::new(
203                io::ErrorKind::InvalidData,
204                "platform reported more sent packets than supplied",
205            ));
206        }
207        self.packets.drain(..count);
208        Ok(())
209    }
210
211    #[must_use]
212    pub fn iter(&self) -> impl ExactSizeIterator<Item = &Packet> {
213        self.packets.iter()
214    }
215}

217/// Capability-negotiated packet transport.
218///
219/// `send` returns the number of packets accepted by the platform. The caller
220/// acknowledges that prefix with [`PacketBatch::acknowledge_sent`], preserving
221/// ownership of a partial-send suffix. `WouldBlock` is temporary; all other I/O
222/// errors are classified by the runner, and permanent failures stop the stack.
223///
224/// # Cancellation
225///
226/// Adapters race [`crate::SingleShardRunner::step`] against timers and
227/// application commands, so either future may be dropped at any await point.
228/// Both must therefore be cancellation safe. `send` may await only before it
229/// accepts the first packet of the batch; once any packet is accepted it must
230/// return that count without awaiting again, reporting a partial send when the
231/// next packet is not immediately writable. A dropped `send` has then accepted
232/// nothing, and the runner's retry of the unacknowledged batch cannot put an
233/// accepted packet on the wire twice. `recv` likewise must not lose packets it
234/// has already taken from the platform when dropped.
235pub trait PacketIo: Send + 'static {
236    fn recv(&mut self, out: &mut PacketBatch) -> impl Future<Output = io::Result<usize>> + Send;
237    fn send(&mut self, packets: &PacketBatch) -> impl Future<Output = io::Result<usize>> + Send;
238    fn capabilities(&self) -> PacketCapabilities;
239
240    /// Whether packets of `endpoint`, as they arrive from the network, are
241    /// delivered to this adapter. A stack that opens a flow picks a local
242    /// port for which they are, so that the replies find the flow. An adapter
243    /// that is the only one sees every flow.
244    fn owns_flow(&self, endpoint: crate::IpEndpoint) -> bool {
245        let _ = endpoint;
246        true
247    }
