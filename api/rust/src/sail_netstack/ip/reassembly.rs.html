82impl fmt::Display for FragmentError {
83    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
84        match self {
85            Self::Wire(error) => write!(formatter, "fragment wire error: {error}"),
86            Self::Budget(error) => write!(formatter, "fragment resource error: {error}"),
87            Self::ClockWentBackwards => formatter.write_str("fragment clock moved backwards"),
88            Self::Timer(error) => write!(formatter, "fragment timer error: {error}"),
89            Self::Malformed(message) => write!(formatter, "malformed fragment: {message}"),
90            Self::Overlap => formatter.write_str("overlapping IP fragments"),
91        }
92    }
93}

156    /// Buffers one fragment and returns a complete reconstructed IP packet.
157    /// Non-fragmented and IPv6 atomic-fragment packets pass through unchanged.
158    ///
159    /// # Errors
160    ///
161    /// Returns an error for malformed, overlapping, over-budget, or
162    /// time-regressing input.
163    pub fn ingest(&mut self, packet: &[u8], now_ms: u64) -> Result<Option<Vec<u8>>, FragmentError> {
164        self.advance_current(now_ms)?;
165        let ip = parse_ip_packet(packet, true)?;
166        let Some(fragment) = ip.fragment else {
167            return Ok(Some(packet.to_vec()));
168        };
169        if fragment.is_atomic() {
170            return Ok(Some(packet.to_vec()));
171        }
172        let fragment_payload = ip.fragment_payload.unwrap_or(ip.payload);
173        let end = fragment_end(fragment, fragment_payload.len())?;
174        let (header, previous_next_header, protocol) = fragment_header(packet, ip.version)?;
175        validate_reassembled_size(ip.version, header.len(), end)?;
176        let key = FragmentKey {
177            version: ip.version,
178            source: ip.source,
179            destination: ip.destination,
180            protocol: if ip.version == IpVersion::V4 {
181                protocol
182            } else {
183                0
184            },
185            identification: fragment.identification,
186        };
187        if self.overlaps(key, fragment.offset_bytes, end) {
188            self.remove_assembly(&key);
189            increment_counter(&mut self.stats.overlap_drops);
190            return Err(FragmentError::Overlap);
191        }
192        if self.conflicts(key, protocol, fragment, end, header.len()) {
193            self.remove_assembly(&key);
194            return Err(FragmentError::Malformed(
195                "fragment conflicts with established datagram bounds",
196            ));
197        }
198        let credit = self.acquire_fragment_credit(key, fragment_payload.len())?;
199        if !self.assemblies.contains_key(&key) {
200            let metadata = credit.assembly_metadata.ok_or(FragmentError::Malformed(
201                "new assembly lacked metadata credit",
202            ))?;
203            let deadline = now_ms.saturating_add(self.effective_timeout(self.timeout_divisor));
204            let expiry_timer = self.expiry.schedule(deadline, key)?;
205            let eviction_key = self.next_eviction_key(now_ms);
206            self.assemblies.insert(
207                key,
208                Assembly {
209                    header: None,
210                    ipv6_previous_next_header: None,
211                    protocol,
212                    pieces: BTreeMap::new(),
213                    total_payload_len: None,
214                    created_at_ms: now_ms,
215                    expiry_timer,
216                    eviction_key,
217                    _metadata: metadata,
218                },
219            );
220            self.eviction_index.insert(eviction_key, key);
221            self.stats.active_datagrams = self.assemblies.len();
222        }
223        let assembly = self
224            .assemblies
225            .get_mut(&key)
226            .ok_or(FragmentError::Malformed("fragment assembly disappeared"))?;
227        if fragment.offset_bytes == 0 {
228            assembly.header = Some(header);
229            assembly.ipv6_previous_next_header = previous_next_header;
230        }
231        if !fragment.more_fragments {
232            if assembly.total_payload_len.is_some_and(|total| total != end) {
233                self.remove_assembly(&key);
234                return Err(FragmentError::Malformed(
235                    "conflicting final fragment lengths",
236                ));
237            }
238            assembly.total_payload_len = Some(end);
239        }
240        assembly.pieces.insert(
241            fragment.offset_bytes,
242            Piece {
243                bytes: fragment_payload.into(),
244                _slot: credit.slot,
245                _payload: credit.payload,
246                _metadata: credit.piece_metadata,
247            },
248        );
249        self.stats.buffered_fragments = self.stats.buffered_fragments.saturating_add(1);
250        if !is_complete(assembly) {
251            return Ok(None);
252        }
253        let assembly = self
254            .remove_assembly(&key)
255            .ok_or(FragmentError::Malformed("complete assembly disappeared"))?;
256        let packet = rebuild(assembly, key.version)?;
257        increment_counter(&mut self.stats.completed_datagrams);
258        Ok(Some(packet))
259    }

270    /// Expires incomplete datagrams using a pressure-dependent timeout
271    /// divisor. Existing assemblies retain their original creation time.
272    ///
273    /// # Errors
274    ///
275    /// Returns [`FragmentError::ClockWentBackwards`] if time regresses.
276    pub fn advance_time_under_pressure(
277        &mut self,
278        now_ms: u64,
279        timeout_divisor: u64,
280    ) -> Result<usize, FragmentError> {
281        self.advance_time_under_pressure_with_quotes(now_ms, timeout_divisor, 0)
282            .map(|expired| expired.expired_datagrams)
283    }

285    /// Expires incomplete datagrams and retains a bounded number of initial
286    /// fragments suitable for ICMP reassembly-timeout quotes. Assemblies that
287    /// never received offset zero are counted and released without a quote.
288    ///
289    /// # Errors
290    ///
291    /// Returns [`FragmentError::ClockWentBackwards`] if time regresses.
292    pub fn advance_time_under_pressure_with_quotes(
293        &mut self,
294        now_ms: u64,
295        timeout_divisor: u64,
296        max_quotes: usize,
297    ) -> Result<FragmentExpirations, FragmentError> {
298        self.update_time(now_ms)?;
299        let timeout_divisor = timeout_divisor.max(1);
300        let due = if timeout_divisor == self.timeout_divisor {
301            self.expiry.advance_to(now_ms)?
302        } else {
303            self.rebuild_expiry(now_ms, timeout_divisor)?
304        };
305        let expired = self.expire_due(due, max_quotes);
306        self.stats.expired_datagrams = self
307            .stats
308            .expired_datagrams
309            .saturating_add(u64::try_from(expired.expired_datagrams).unwrap_or(u64::MAX));
310        Ok(expired)
311    }

326    fn overlaps(&self, key: FragmentKey, start: u32, end: u32) -> bool {
327        self.assemblies.get(&key).is_some_and(|assembly| {
328            assembly.pieces.iter().any(|(&offset, piece)| {
329                let piece_end =
330                    offset.saturating_add(u32::try_from(piece.bytes.len()).unwrap_or(u32::MAX));
331                start < piece_end && offset < end
332            })
333        })
334    }

336    fn conflicts(
337        &self,
338        key: FragmentKey,
339        protocol: u8,
340        fragment: FragmentInfo,
341        end: u32,
342        header_len: usize,
343    ) -> bool {
344        self.assemblies.get(&key).is_some_and(|assembly| {
345            if assembly.protocol != protocol
346                || assembly.total_payload_len.is_some_and(|total| end > total)
347            {
348                return true;
349            }
350            if !fragment.more_fragments
351                && assembly.pieces.iter().any(|(&offset, piece)| {
352                    offset.saturating_add(u32::try_from(piece.bytes.len()).unwrap_or(u32::MAX))
353                        > end
354                })
355            {
356                return true;
357            }
358            let total = if fragment.more_fragments {
359                assembly.total_payload_len
360            } else {
361                Some(end)
362            };
363            let final_header_len = if fragment.offset_bytes == 0 {
364                Some(header_len)
365            } else {
366                assembly.header.as_ref().map(Vec::len)
367            };
368            total
369                .zip(final_header_len)
370                .is_some_and(|(total, header_len)| {
371                    reassembled_size_exceeds(key.version, header_len, total)
372                })
373        })
374    }

376    fn update_time(&mut self, now_ms: u64) -> Result<(), FragmentError> {
377        if now_ms < self.now_ms {
378            return Err(FragmentError::ClockWentBackwards);
379        }
380        self.now_ms = now_ms;
381        Ok(())
382    }

384    fn acquire_fragment_credit(
385        &mut self,
386        key: FragmentKey,
387        payload_len: usize,
388    ) -> Result<PendingFragmentCredit, FragmentError> {
389        loop {
390            let needs_assembly = !self.assemblies.contains_key(&key);
391            match self.try_acquire_fragment_credit(payload_len, needs_assembly) {
392                Ok(credit) => return Ok(credit),
393                Err(_) if self.evict_oldest_assembly() => {
394                    self.stats.evicted_datagrams = self.stats.evicted_datagrams.saturating_add(1);
395                }
396                Err(error) => return Err(error.into()),
397            }
398        }
399    }
400
401    fn try_acquire_fragment_credit(
402        &self,
403        payload_len: usize,
404        needs_assembly: bool,
405    ) -> Result<PendingFragmentCredit, BudgetError> {
406        let slot = self.ledger.try_acquire(ResourceKind::Fragments, 1)?;
407        let payload = self
408            .ledger
409            .try_acquire(ResourceKind::FragmentBytes, payload_len)?;
410        let piece_metadata = self
411            .ledger
412            .try_acquire(ResourceKind::MetadataBytes, PIECE_METADATA_BYTES)?;
413        let assembly_metadata = needs_assembly
414            .then(|| {
415                self.ledger
416                    .try_acquire(ResourceKind::MetadataBytes, ASSEMBLY_METADATA_BYTES)
417            })
418            .transpose()?;
419        Ok(PendingFragmentCredit {
420            slot,
421            payload,
422            piece_metadata,
423            assembly_metadata,
424        })
425    }

427    fn evict_oldest_assembly(&mut self) -> bool {
428        let Some((_, key)) = self.eviction_index.first_key_value() else {
429            return false;
430        };
431        let key = *key;
432        self.remove_assembly(&key).is_some()

435    fn next_eviction_key(&mut self, created_at_ms: u64) -> (u64, u64) {
436        loop {
437            let key = (created_at_ms, self.next_eviction_serial);
438            self.next_eviction_serial = self.next_eviction_serial.wrapping_add(1);
439            if !self.eviction_index.contains_key(&key) {
440                return key;
441            }
442        }
443    }

452    fn advance_current(&mut self, now_ms: u64) -> Result<usize, FragmentError> {
453        self.update_time(now_ms)?;
454        let due = self.expiry.advance_to(now_ms)?;
455        let expired = self.expire_due(due, 0).expired_datagrams;
456        self.stats.expired_datagrams = self
457            .stats
458            .expired_datagrams
459            .saturating_add(u64::try_from(expired).unwrap_or(u64::MAX));
460        Ok(expired)
461    }

463    fn expire_due(&mut self, due: Vec<FragmentKey>, max_quotes: usize) -> FragmentExpirations {
464        let mut expired = FragmentExpirations::default();
465        for key in due {
466            let Some(assembly) = self.take_assembly(&key) else {
467                continue;
468            };
469            expired.expired_datagrams = expired.expired_datagrams.saturating_add(1);
470            if expired.invoking_packets.len() < max_quotes {
471                if let Some(packet) = initial_fragment_packet(&assembly, key) {
472                    expired.invoking_packets.push(packet);
473                }
474            }
475        }
476        expired
477    }

479    fn rebuild_expiry(
480        &mut self,
481        now_ms: u64,
482        timeout_divisor: u64,
483    ) -> Result<Vec<FragmentKey>, FragmentError> {
484        let effective_timeout = self.effective_timeout(timeout_divisor);
485        self.expiry = TimerWheel::new(self.expiry_tick_ms, now_ms);
486        let mut expired = Vec::new();
487        for (key, assembly) in &mut self.assemblies {
488            if now_ms.saturating_sub(assembly.created_at_ms) >= effective_timeout {
489                expired.push(*key);
490                continue;
491            }
492            let deadline = assembly.created_at_ms.saturating_add(effective_timeout);
493            assembly.expiry_timer = self.expiry.schedule(deadline, *key)?;
494        }
495        self.timeout_divisor = timeout_divisor;
496        Ok(expired)
497    }

499    fn remove_assembly(&mut self, key: &FragmentKey) -> Option<Assembly> {
500        let assembly = self.take_assembly(key)?;
501        self.expiry.cancel(assembly.expiry_timer);
502        Some(assembly)

505    fn take_assembly(&mut self, key: &FragmentKey) -> Option<Assembly> {
506        let assembly = self.assemblies.remove(key)?;
507        self.eviction_index.remove(&assembly.eviction_key);
508        self.stats.active_datagrams = self.assemblies.len();
509        self.stats.buffered_fragments = self
510            .stats
511            .buffered_fragments
512            .saturating_sub(assembly.pieces.len());
513        Some(assembly)
514    }

517fn initial_fragment_packet(assembly: &Assembly, key: FragmentKey) -> Option<Vec<u8>> {
518    let first = assembly.pieces.get(&0)?;
519    let mut packet = assembly.header.clone()?;
520    match key.version {
521        IpVersion::V4 => packet.extend_from_slice(&first.bytes),
522        IpVersion::V6 => {
523            let payload_len = packet
524                .len()
525                .checked_sub(40)?
526                .checked_add(8)?
527                .checked_add(first.bytes.len())?;
528            let payload_len = u16::try_from(payload_len).ok()?;
529            packet[4..6].copy_from_slice(&payload_len.to_be_bytes());
530            packet.push(assembly.protocol);
531            packet.push(0);
532            packet.extend_from_slice(&1_u16.to_be_bytes());
533            packet.extend_from_slice(&key.identification.to_be_bytes());
534            packet.extend_from_slice(&first.bytes);
535        }
536    }
537    Some(packet)
538}

540fn fragment_header(
541    packet: &[u8],
542    version: IpVersion,
543) -> Result<(Vec<u8>, Option<usize>, u8), FragmentError> {
544    match version {
545        IpVersion::V4 => {
546            let header_len = usize::from(packet[0] & 0x0f) * 4;
547            Ok((packet[..header_len].to_vec(), None, packet[9]))
548        }
549        IpVersion::V6 => ipv6_fragment_header(packet),
550    }
551}
552
553fn fragment_end(fragment: FragmentInfo, payload_len: usize) -> Result<u32, FragmentError> {
554    if payload_len == 0 {
555        return Err(FragmentError::Malformed("empty non-atomic fragment"));
556    }
557    if fragment.more_fragments && !payload_len.is_multiple_of(8) {
558        return Err(FragmentError::Malformed(
559            "non-final fragment payload is not a multiple of eight",
560        ));
561    }
562    let length = u32::try_from(payload_len)
563        .map_err(|_| FragmentError::Malformed("fragment payload length exceeds u32"))?;
564    let end = fragment
565        .offset_bytes
566        .checked_add(length)
567        .ok_or(FragmentError::Malformed("fragment range overflow"))?;
568    if usize::try_from(end).unwrap_or(usize::MAX) > MAX_REASSEMBLED_BYTES {
569        return Err(FragmentError::Malformed(
570            "reassembled datagram is too large",
571        ));
572    }
573    Ok(end)
574}

576fn validate_reassembled_size(
577    version: IpVersion,
578    header_len: usize,
579    payload_len: u32,
580) -> Result<(), FragmentError> {
581    if reassembled_size_exceeds(version, header_len, payload_len) {
582        return Err(FragmentError::Malformed(
583            "reassembled IP datagram is too large",
584        ));
585    }
586    Ok(())
587}

589fn reassembled_size_exceeds(version: IpVersion, header_len: usize, payload_len: u32) -> bool {
590    let payload_len = usize::try_from(payload_len).unwrap_or(usize::MAX);
591    let field_len = match version {
592        IpVersion::V4 => header_len.checked_add(payload_len),
593        IpVersion::V6 => header_len
594            .checked_sub(40)
595            .and_then(|unfragmentable_len| unfragmentable_len.checked_add(payload_len)),
596    };
597    field_len.is_none_or(|length| length > usize::from(u16::MAX))
598}

600fn ipv6_fragment_header(packet: &[u8]) -> Result<(Vec<u8>, Option<usize>, u8), FragmentError> {
601    let mut next_header = packet[6];
602    let mut offset = 40_usize;
603    let mut previous_next_header = 6_usize;
604    for _ in 0..8 {
605        if next_header == 44 {
606            if offset + 8 > packet.len() {
607                return Err(FragmentError::Malformed("truncated IPv6 fragment header"));
608            }
609            return Ok((
610                packet[..offset].to_vec(),
611                Some(previous_next_header),
612                packet[offset],
613            ));
614        }
615        let length = match next_header {
616            0 | 43 | 60 => (usize::from(packet[offset + 1]) + 1) * 8,
617            51 => (usize::from(packet[offset + 1]) + 2) * 4,
618            _ => return Err(FragmentError::Malformed("IPv6 fragment header not found")),
619        };
620        if offset + length > packet.len() {
621            return Err(FragmentError::Malformed("truncated IPv6 extension header"));
622        }
623        next_header = packet[offset];
624        previous_next_header = offset;
625        offset += length;
626    }
627    Err(FragmentError::Malformed(
628        "IPv6 extension chain exceeds limit",
629    ))
630}

632fn is_complete(assembly: &Assembly) -> bool {
633    let (Some(total), Some(_)) = (assembly.total_payload_len, &assembly.header) else {
634        return false;
635    };
636    let mut expected = 0_u32;
637    for (&offset, piece) in &assembly.pieces {
638        if offset != expected {
639            return false;
640        }
641        expected = expected.saturating_add(u32::try_from(piece.bytes.len()).unwrap_or(u32::MAX));
642    }
643    expected == total
644}

646fn rebuild(mut assembly: Assembly, version: IpVersion) -> Result<Vec<u8>, FragmentError> {
647    let mut packet = assembly
648        .header
649        .take()
650        .ok_or(FragmentError::Malformed("missing first fragment"))?;
651    for piece in assembly.pieces.into_values() {
652        packet.extend_from_slice(&piece.bytes);
653    }
654    match version {
655        IpVersion::V4 => {
656            let total = u16::try_from(packet.len())
657                .map_err(|_| FragmentError::Malformed("IPv4 datagram too large"))?;
658            packet[2..4].copy_from_slice(&total.to_be_bytes());
659            packet[6..8].copy_from_slice(&0_u16.to_be_bytes());
660            packet[10..12].copy_from_slice(&0_u16.to_be_bytes());
661            let checksum = ipv4_checksum(&packet[..usize::from(packet[0] & 0x0f) * 4]);
662            packet[10..12].copy_from_slice(&checksum.to_be_bytes());
663        }
664        IpVersion::V6 => {
665            let previous = assembly
666                .ipv6_previous_next_header
667                .ok_or(FragmentError::Malformed(
668                    "missing IPv6 fragment predecessor",
669                ))?;
670            packet[previous] = assembly.protocol;
671            let payload_len = u16::try_from(packet.len().saturating_sub(40))
672                .map_err(|_| FragmentError::Malformed("IPv6 datagram too large"))?;
673            packet[4..6].copy_from_slice(&payload_len.to_be_bytes());
674        }
675    }
676    Ok(packet)
677}
678
679fn ipv4_checksum(header: &[u8]) -> u16 {
680    let mut sum = 0_u32;
681    for chunk in header.as_chunks::<2>().0 {
682        sum += u32::from(u16::from_be_bytes(*chunk));
683    }
684    while sum >> 16 != 0 {
685        sum = (sum & 0xffff) + (sum >> 16);
686    }
687    !u16::try_from(sum).unwrap_or(u16::MAX)
688}
