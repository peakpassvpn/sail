198fn remove_comments(text: &str) -> Cow<'_, str> {
199    static RE: std::sync::LazyLock<Regex> =
200        std::sync::LazyLock::new(|| Regex::new(r"(#[^*]*)").expect("a valid regex"));
201    RE.replace(text, "")
202}

204fn get_section(text: &str) -> Option<&str> {
205    static RE: std::sync::LazyLock<Regex> = std::sync::LazyLock::new(|| {
206        Regex::new(r"^\s*\[\s*([^\]]*)\s*\]\s*$").expect("a valid regex")
207    });
208    Some(RE.captures(text)?.get(1)?.as_str())
209}

215fn get_certificate_sections<'a, I>(lines: I) -> HashMap<String, String>
216where
217    I: Iterator<Item = &'a io::Result<String>>,
218{
219    let mut certificates = HashMap::new();
220    let mut current_name: Option<String> = None;
221    let mut current_lines: Vec<String> = Vec::new();
222
223    for line in lines {
224        let line = match line {
225            Ok(line) => line,
226            Err(_) => continue,
227        };
228        let trimmed = line.trim();
229        if let Some(section) = get_section(trimmed) {
230            if let Some(name) = current_name.take() {
231                if !current_lines.is_empty() {
232                    let mut content = current_lines.join("\n");
233                    content.push('\n');
234                    certificates.insert(name, content);
235                }
236                current_lines.clear();
237            }
238            let lower = section.to_lowercase();
239            if let Some(name) = section.strip_prefix("Certificate.") {
240                let name = name.trim();
241                if !name.is_empty() {
242                    current_name = Some(name.to_string());
243                }
244            } else if lower.starts_with("certificate.") {
245                let name = &section["certificate.".len()..];
246                let name = name.trim();
247                if !name.is_empty() {
248                    current_name = Some(name.to_string());
249                }
250            } else if lower.starts_with("certificate_") {
251                let name = &section["certificate_".len()..];
252                let name = name.trim();
253                if !name.is_empty() {
254                    current_name = Some(name.to_string());
255                }
256            } else if lower.starts_with("certificate ") {
257                let name = &section["certificate ".len()..];
258                let name = name.trim();
259                if !name.is_empty() {
260                    current_name = Some(name.to_string());
261                }
262            } else if lower.starts_with("certificate") {
263                let name = &section["certificate".len()..];
264                let name = name.trim();
265                if !name.is_empty() {
266                    current_name = Some(name.to_string());
267                }
268            }
269            continue;
270        }
271        if current_name.is_some() {
272            if trimmed.is_empty() || trimmed.starts_with('#') {
273                continue;
274            }
275            current_lines.push(trimmed.to_string());
276        }
277    }
278
279    if let Some(name) = current_name.take() {
280        if !current_lines.is_empty() {
281            let mut content = current_lines.join("\n");
282            content.push('\n');
283            certificates.insert(name, content);
284        }
285    }
286
287    certificates
288}

365fn get_lines_by_section<'a, I>(section: &str, lines: I) -> Vec<String>
366where
367    I: Iterator<Item = &'a io::Result<String>>,
368{
369    let mut new_lines = Vec::new();
370    let mut curr_sect: String = "".to_string();
371    let normalized_target = normalize_section(section);
372    for line in lines.flatten().map(|x| x.trim()) {
373        let line = remove_comments(line);
374        if let Some(s) = get_section(line.as_ref()) {
375            curr_sect = s.to_string();
376            continue;
377        }
378        if normalize_section(&curr_sect) == normalized_target && !line.is_empty() {
379            new_lines.push(line.to_string());
380        }
381    }
382    new_lines
383}

385fn get_char_sep_slice(text: &str, pat: char) -> Option<Vec<String>>
386where
387{
388    let mut items = Vec::new();
389    for item in text.split(pat).map(str::trim) {
390        if !item.is_empty() {
391            items.push(item.to_string());
392        }
393    }
394    if !items.is_empty() {
395        Some(items)
396    } else {
397        None
398    }

421fn parse_bool(key: &str, value: &str) -> Result<bool> {
422    match value {
423        "true" => Ok(true),
424        "false" => Ok(false),
425        _ => Err(anyhow!(
426            "{}: expected true or false, got \"{}\"",
427            key,
428            value
429        )),
430    }

433pub fn from_lines(lines: Vec<io::Result<String>>) -> Result<Config> {
434    let certificates = get_certificate_sections(lines.iter());
435    let ech_configs = get_ech_sections(lines.iter());
436    if !get_lines_by_section("Env", lines.iter()).is_empty() {
437        return Err(anyhow!(
438            "[Env] is not supported: tuning is given at start (--profile, --set), \
439             and everything else has a configuration field"
440        ));
441    }
442
443    let mut general = General::default();
444    let general_lines = get_lines_by_section("General", lines.iter());
445    for line in general_lines {
446        let parts: Vec<&str> = line.split('=').map(str::trim).collect();
447        if parts.len() != 2 {
448            continue;
449        }
450        match parts[0] {
451            "tun-fd" => {
452                general.tun_fd = get_value::<i32>(parts[1]);
453            }
454            "tun" => {
455                if let Some(items) = get_char_sep_slice(parts[1], ',') {
456                    if !items.is_empty() && items[0] == "auto" {
457                        general.tun_auto = Some(true);
458                        continue;
459                    }
460                    if items.len() < 5 {
461                        continue;
462                    }
463                    let tun = Tun {
464                        name: Some(items[0].clone()),
465                        address: Some(items[1].clone()),
466                        netmask: Some(items[2].clone()),
467                        gateway: Some(items[3].clone()),
468                        mtu: get_value::<i32>(&items[4]),
469                    };
470                    general.tun = Some(tun);
471                }
472            }
473            "nf" => {
474                // nf = driver_name, path/to/nfapi.dll
475                if let Some(items) = get_char_sep_slice(parts[1], ',') {
476                    let nfapi = if items.len() >= 2 {
477                        Some(items[1].trim().to_owned())
478                    } else {
479                        None
480                    };
481                    let nf = Nf {
482                        driver_name: items[0].trim().to_owned(),
483                        nfapi,
484                    };
485                    general.nf = Some(nf);
486                }
487            }
488            "loglevel" => {
489                general.loglevel = Some(parts[1].to_string());
490            }
491            "logoutput" => {
492                general.logoutput = Some(parts[1].to_string());
493            }
494            "logformat" => {
495                general.logformat = Some(parts[1].to_string());
496            }
497            "dns-server" => {
498                general.dns_server = get_char_sep_slice(parts[1], ',');
499            }
500            "dns-interface" => {
501                general.dns_interface = get_string(parts[1]);
502            }
503            "always-real-ip" => {
504                general.always_real_ip = get_char_sep_slice(parts[1], ',');
505            }
506            "always-fake-ip" => {
507                general.always_fake_ip = get_char_sep_slice(parts[1], ',');
508            }
509            "routing-domain-resolve" => {
510                general.routing_domain_resolve = if parts[1] == "true" {
511                    Some(true)
512                } else {
513                    Some(false)
514                };
515            }
516            "http-interface" | "interface" => {
517                general.http_interface = get_string(parts[1]);
518            }
519            "http-port" | "port" => {
520                general.http_port = get_value::<u16>(parts[1]);
521            }
522            "socks-interface" => {
523                general.socks_interface = get_string(parts[1]);
524            }
525            "socks-port" => {
526                general.socks_port = get_value::<u16>(parts[1]);
527            }
528            "api-interface" => {
529                general.api_interface = get_string(parts[1]);
530            }
531            "api-port" => {
532                general.api_port = get_value::<u16>(parts[1]);
533            }
534            "wintun" => {
535                general.wintun = get_string(parts[1]);
536            }
537            "tun-dns-server" => {
538                general.tun_dns_server = get_char_sep_slice(parts[1], ',');
539            }
540            "ipv6" => {
541                general.ipv6 = Some(parse_bool("ipv6", parts[1])?);
542            }
543            "gateway-mode" => {
544                general.gateway_mode = Some(parse_bool("gateway-mode", parts[1])?);
545            }
546            "sniffing" => {
547                general.sniffing = Some(parse_bool("sniffing", parts[1])?);
548            }
549            "dns-reverse-mapping" => {
550                general.dns_reverse_mapping = Some(parse_bool("dns-reverse-mapping", parts[1])?);
551            }
552            _ => {}
553        }
554    }
555
556    let mut proxies = Vec::new();
557    let proxy_lines = get_lines_by_section("Proxy", lines.iter());
558    for line in proxy_lines {
559        let parts: Vec<&str> = line.splitn(2, '=').map(str::trim).collect();
560        if parts.len() != 2 {
561            continue;
562        }
563        let mut proxy = Proxy::default();
564        let tag = parts[0];
565        if tag.is_empty() {
566            // empty tag is not allowed
567            continue;
568        }
569        proxy.tag = tag.to_string();
570        let params = if let Some(p) = get_char_sep_slice(parts[1], ',') {
571            p
572        } else {
573            continue;
574        };
575        if params.is_empty() {
576            // there must be at least one param, i.e. the protocol field
577            continue;
578        }
579        proxy.protocol = params[0].clone();
580
581        // extract key-value params
582        // let params = &params[2..];
583        for param in &params {
584            let parts: Vec<&str> = param.split('=').map(str::trim).collect();
585            if parts.len() != 2 {
586                continue;
587            }
588            let k = parts[0];
589            let v = parts[1];
590            if k.is_empty() || v.is_empty() {
591                continue;
592            }
593            match k {
594                "encrypt-method" => {
595                    proxy.encrypt_method = Some(v.to_string());
596                }
597                "prefix" => {
598                    proxy.prefix = Some(v.to_string());
599                }
600                "password" => {
601                    proxy.password = Some(v.to_string());
602                }
603                "obfs" => {
604                    proxy.obfs_type = Some(v.to_string());
605                }
606                "obfs-host" => {
607                    proxy.obfs_host = Some(v.to_string());
608                }
609                "obfs-path" => {
610                    proxy.obfs_path = Some(v.to_string());
611                }
612                "ws" => proxy.ws = if v == "true" { Some(true) } else { Some(false) },
613                "tls" => proxy.tls = if v == "true" { Some(true) } else { Some(false) },
614                "tls-cert" => {
615                    proxy.tls_cert = Some(v.to_string());
616                }
617                "tls-insecure" => {
618                    proxy.tls_insecure = if v == "true" { Some(true) } else { Some(false) }
619                }
620                "tls-ech" => proxy.tls_ech = if v == "true" { Some(true) } else { Some(false) },
621                "tls-ech-disable-dns-lookup" => {
622                    proxy.tls_ech_disable_dns_lookup =
623                        if v == "true" { Some(true) } else { Some(false) }
624                }
625                "tls-ech-config-list" | "ech-config-list" => {
626                    proxy.tls_ech_config_list = Some(v.to_string());
627                }
628                "ws-path" => {
629                    proxy.ws_path = Some(v.to_string());
630                }
631                "ws-host" => {
632                    proxy.ws_host = Some(v.to_string());
633                }
634                "sni" => {
635                    proxy.sni = Some(v.to_string());
636                }
637                "username" => {
638                    proxy.username = Some(v.to_string());
639                }
640                "uuid" => {
641                    proxy.uuid = Some(v.to_string());
642                }
643                "amux" => proxy.amux = if v == "true" { Some(true) } else { Some(false) },
644                "amux-max" => {
645                    let i = v.parse::<i32>().ok();
646                    proxy.amux_max = i;
647                }
648                "amux-con" => {
649                    let i = v.parse::<i32>().ok();
650                    proxy.amux_con = i;
651                }
652                "amux-max-recv" => {
653                    let i = v.parse::<u64>().ok();
654                    proxy.amux_max_recv = i;
655                }
656                "amux-max-lifetime" => {
657                    let i = v.parse::<u64>().ok();
658                    proxy.amux_max_lifetime = i;
659                }
660                "quic" => proxy.quic = if v == "true" { Some(true) } else { Some(false) },
661                "reality" => proxy.reality = if v == "true" { Some(true) } else { Some(false) },
662                "reality-public-key" => {
663                    proxy.reality_public_key = Some(v.to_string());
664                }
665                "reality-short-id" => {
666                    proxy.reality_short_id = Some(v.to_string());
667                }
668                "client-fingerprint" => {
669                    proxy.client_fingerprint = Some(v.to_string());
670                }
671                "flow" => {
672                    proxy.flow = Some(v.to_string());
673                }
674                "interface" => {
675                    proxy.interface = Some(v.to_string());
676                }
677                _ => {}
678            }
679        }
680
681        // built-in protocols have no address port, password
682        match proxy.protocol.as_str() {
683            "direct" => {
684                proxies.push(proxy);
685                continue;
686            }
687            "drop" => {
688                proxies.push(proxy);
689                continue;
690            }
691            // compat
692            "reject" => {
693                proxy.protocol = "drop".to_string();
694                proxies.push(proxy);
695                continue;
696            }
697            _ => {}
698        }
699
700        // parse address and port
701        let params = &params[1..];
702        if params.len() < 2 {
703            // address and port are required
704            continue;
705        }
706        proxy.address = Some(params[0].clone());
707        let port = if let Ok(p) = params[1].parse::<u16>() {
708            p
709        } else {
710            continue; // not valid port
711        };
712        proxy.port = Some(port);
713
714        // parse positional params
715        let pos_params = &params[2..];
716        for (i, param) in pos_params.iter().enumerate() {
717            if param.contains('=') {
718                continue;
719            }
720            match (proxy.protocol.as_str(), i) {
721                ("ss" | "shadowsocks", 0) => proxy.encrypt_method = Some(param.clone()),
722                ("ss" | "shadowsocks", 1) => proxy.password = Some(param.clone()),
723                ("trojan", 0) => proxy.password = Some(param.clone()),
724                ("vmess", 0) => proxy.username = Some(param.clone()),
725                ("vless", 0) => proxy.password = Some(param.clone()),
726                _ => (),
727            }
728        }
729
730        // compat
731        if let "ss" = proxy.protocol.as_str() {
732            proxy.protocol = "shadowsocks".to_string();
733        }
734
735        proxies.push(proxy);
736    }
737
738    let mut proxy_groups = Vec::new();
739    let proxy_group_lines = get_lines_by_section("Proxy Group", lines.iter());
740    for line in proxy_group_lines {
741        let parts: Vec<&str> = line.splitn(2, '=').map(str::trim).collect();
742        if parts.len() != 2 {
743            continue;
744        }
745        let mut group = ProxyGroup::default();
746        let tag = parts[0];
747        if tag.is_empty() {
748            // empty tag is not allowed
749            continue;
750        }
751        group.tag = tag.to_string();
752        let params = if let Some(p) = get_char_sep_slice(parts[1], ',') {
753            p
754        } else {
755            continue;
756        };
757        if params.is_empty() {
758            // there must be at least one param, i.e. the protocol field
759            continue;
760        }
761        group.protocol = params[0].clone();
762
763        let params = &params[1..];
764        if params.is_empty() {
765            // require at least one proxy
766            continue;
767        }
768
769        let mut actors = Vec::new();
770        for param in params {
771            if !param.contains('=') && !param.is_empty() {
772                actors.push(param.to_string());
773            }
774        }
775        if actors.is_empty() {
776            // require at least one actor
777            continue;
778        }
779        group.actors = Some(actors);
780
781        for param in params {
782            if let Some((k, v)) = param.split_once('=') {
783                let (k, v) = (k.trim(), v.trim());
784                if !k.is_empty() && !v.is_empty() {
785                    group.params.push((k.to_string(), v.to_string()));
786                }
787            }
788        }
789
790        proxy_groups.push(group);
791    }
792
793    let mut rules = Vec::new();
794    let rule_lines = get_lines_by_section("Rule", lines.iter());
795    for line in rule_lines {
796        let params = if let Some(p) = get_char_sep_slice(&line, ',') {
797            p
798        } else {
799            continue;
800        };
801        if params.len() < 2 {
802            continue; // at lease 2 params
803        }
804        let mut rule = Rule {
805            type_field: params[0].to_string(),
806            ..Default::default()
807        };
808
809        // handle the FINAL rule first
810        if rule.type_field == "FINAL" {
811            rule.target = params[1].to_string();
812            rules.push(rule);
813            break; // FINAL is final.
814        }
815
816        if params.len() < 3 {
817            continue; // at lease 3 params except the FINAL rule
818        }
819
820        // the 3th must be the target
821        rule.target = params[2].to_string();
822
823        match rule.type_field.as_str() {
824            "IP-CIDR" | "DOMAIN" | "DOMAIN-SUFFIX" | "DOMAIN-KEYWORD" | "GEOIP" | "EXTERNAL"
825            | "PORT-RANGE" | "NETWORK" | "INBOUND-TAG" | "PROCESS-NAME" => {
826                rule.filter = Some(params[1].to_string());
827            }
828            _ => {}
829        }
830
831        rules.push(rule);
832    }
833
834    let mut hosts = HashMap::new();
835    let host_lines = get_lines_by_section("Host", lines.iter());
836    for line in host_lines {
837        let parts: Vec<&str> = line.split('=').map(str::trim).collect();
838        if parts.len() != 2 {
839            continue;
840        }
841        let name = parts[0];
842        let ips: Vec<String> = parts[1]
843            .split(',')
844            .map(str::trim)
845            .map(|x| x.to_owned())
846            .collect();
847        hosts.insert(name.to_owned(), ips);
848    }
849
850    Ok(Config {
851        general: Some(general),
852        proxy: Some(proxies),
853        proxy_group: Some(proxy_groups),
854        rule: Some(rules),
855        host: Some(hosts),
856        certificates: if certificates.is_empty() {
857            None
858        } else {
859            Some(certificates)
860        },
861        ech_configs: if ech_configs.is_empty() {
862            None
863        } else {
864            Some(ech_configs)
865        },
866    })
867}

869/// Turns a parsed `.conf` into the configuration model.
870pub fn to_config(conf: &Config) -> Result<model::Config> {
871    let mut config = model::Config::default();
872
873    if let Some(ext_general) = &conf.general {
874        config.log = to_log(ext_general)?;
875
876        if let (Some(interface), Some(port)) = (
877            ext_general.http_interface.as_ref(),
878            ext_general.http_port.as_ref(),
879        ) {
880            config
881                .inbounds
882                .push(inbound("http", "http", Some((interface, *port)), json!({})));
883        }
884
885        if let (Some(interface), Some(port)) = (
886            ext_general.socks_interface.as_ref(),
887            ext_general.socks_port.as_ref(),
888        ) {
889            config.inbounds.push(inbound(
890                "socks",
891                "socks",
892                Some((interface, *port)),
893                json!({}),
894            ));
895        }
896
897        if let Some(nf) = &ext_general.nf {
898            config.inbounds.push(inbound(
899                "nf",
900                "nf",
901                None,
902                json!({
903                    "driver_name": nf.driver_name,
904                    "nfapi": nf.nfapi,
905                    "fake_dns_exclude": ext_general.always_real_ip,
906                    "fake_dns_include": ext_general.always_fake_ip,
907                }),
908            ));
909        }
910
911        if ext_general.gateway_mode.is_some() && ext_general.tun_auto != Some(true) {
912            return Err(anyhow!("gateway-mode needs tun = auto"));
913        }
914        if ext_general.tun_fd.is_some()
915            || ext_general.tun_auto.is_some()
916            || ext_general.tun.is_some()
917        {
918            let mut tun = json!({
919                "fake_dns_exclude": ext_general.always_real_ip,
920                "fake_dns_include": ext_general.always_fake_ip,
921                "wintun": ext_general.wintun,
922                "dns_servers": ext_general.tun_dns_server,
923            });
924            if let Some(fd) = ext_general.tun_fd {
925                tun["fd"] = json!(fd);
926            } else if ext_general.tun_auto == Some(true) {
927                tun["auto"] = json!(true);
928                config.route.auto_detect_interface = true;
929                if let Some(gateway_mode) = ext_general.gateway_mode {
930                    tun["gateway_mode"] = json!(gateway_mode);
931                }
932            } else if let Some(ext_tun) = &ext_general.tun {
933                tun["name"] = json!(ext_tun.name);
934                tun["address"] = json!(ext_tun.address);
935                tun["gateway"] = json!(ext_tun.gateway);
936                tun["netmask"] = json!(ext_tun.netmask);
937                tun["mtu"] = json!(ext_tun.mtu);
938            }
939            config.inbounds.push(inbound("tun", "tun", None, tun));
940        }
941    }
942
943    let certificates = conf.certificates.as_ref();
944    let ech_configs = conf.ech_configs.as_ref();
945    let resolve_cert = |value: &Option<String>| -> Option<String> {
946        let value = value.as_ref()?;
947        if let Some(certificates) = certificates {
948            if let Some(content) = certificates.get(value) {
949                return Some(content.clone());
950            }
951        }
952        Some(value.clone())
953    };
954    let resolve_ech = |value: &Option<String>| -> Option<String> {
955        let value = value.as_ref()?;
956        if let Some(ech_configs) = ech_configs {
957            if let Some(content) = ech_configs.get(value) {
958                return Some(content.clone());
959            }
960        }
961        Some(value.clone())
962    };
963
964    let outbounds = &mut config.outbounds;
965    for ext_proxy in conf.proxy.iter().flatten() {
966        let tag = ext_proxy.tag.as_str();
967        let first_new = outbounds.len();
968        match ext_proxy.protocol.as_str() {
969            "direct" => outbounds.push(outbound(tag, "direct", json!({}))),
970            "drop" => outbounds.push(outbound(tag, "block", json!({}))),
971            "redirect" => outbounds.push(outbound(
972                tag,
973                "redirect",
974                json!({ "server": ext_proxy.address, "server_port": ext_proxy.port }),
975            )),
976            "socks" => outbounds.push(outbound(
977                tag,
978                "socks",
979                json!({
980                    "server": ext_proxy.address,
981                    "server_port": ext_proxy.port,
982                    "username": ext_proxy.username,
983                    "password": ext_proxy.password,
984                }),
985            )),
986            "ss" | "shadowsocks" => {
987                let mut ss = json!({
988                    "server": ext_proxy.address,
989                    "server_port": ext_proxy.port,
990                    "method": ext_proxy.encrypt_method,
991                    "password": ext_proxy.password,
992                    "prefix": ext_proxy.prefix,
993                });
994                if let Some(obfs) = &ext_proxy.obfs_type {
995                    let mut opts = format!("obfs={}", obfs);
996                    if let Some(host) = &ext_proxy.obfs_host {
997                        opts.push_str(&format!(";obfs-host={}", host));
998                    }
999                    if let Some(path) = &ext_proxy.obfs_path {
1000                        opts.push_str(&format!(";obfs-uri={}", path));
1001                    }
1002                    ss["plugin"] = json!("obfs-local");
1003                    ss["plugin_opts"] = json!(opts);
1004                }
1005                outbounds.push(outbound(tag, "shadowsocks", ss));
1006            }
1007            "vless" => {
1008                let mut vless = json!({
1009                    "server": ext_proxy.address,
1010                    "server_port": ext_proxy.port,
1011                    // prioritize uuid, then password
1012                    "uuid": ext_proxy.uuid.as_ref().or(ext_proxy.password.as_ref()),
1013                });
1014                if let Some(flow) = &ext_proxy.flow {
1015                    vless["flow"] = json!(flow);
1016                }
1017                if ext_proxy.reality.unwrap_or(false) {
1018                    vless["tls"] = json!({
1019                        "enabled": true,
1020                        "server_name": ext_proxy.sni,
1021                        "reality": {
1022                            "enabled": true,
1023                            "public_key": ext_proxy.reality_public_key,
1024                            "short_id": ext_proxy.reality_short_id,
1025                        },
1026                        "utls": utls(&ext_proxy.client_fingerprint),
1027                    });
1028                }
1029                outbounds.push(outbound(tag, "vless", vless));
1030            }
1031            protocol @ ("trojan" | "vmess") => {
1032                let mut proxy = if protocol == "trojan" {
1033                    json!({
1034                        "server": ext_proxy.address,
1035                        "server_port": ext_proxy.port,
1036                        "password": ext_proxy.password,
1037                    })
1038                } else {
1039                    json!({
1040                        "server": ext_proxy.address,
1041                        "server_port": ext_proxy.port,
1042                        "uuid": ext_proxy.uuid.as_ref().or(ext_proxy.username.as_ref()),
1043                        "security": ext_proxy
1044                            .encrypt_method
1045                            .as_deref()
1046                            .unwrap_or("chacha20-ietf-poly1305"),
1047                    })
1048                };
1049                let quic = ext_proxy.quic.unwrap_or(false);
1050                let certificate = resolve_cert(&ext_proxy.tls_cert);
1051                // A certificate from a [Certificate] section is inline.
1052                let (certificate, certificate_path) = match certificate {
1053                    Some(c) if c.contains("-----BEGIN") => (Some(c), None),
1054                    other => (None, other),
1055                };
1056                let ech = ext_proxy.tls_ech.unwrap_or(false).then(|| {
1057                    json!({
1058                        "enabled": true,
1059                        "config": resolve_ech(&ext_proxy.tls_ech_config_list),
1060                        "disable_dns_lookup": ext_proxy.tls_ech_disable_dns_lookup,
1061                    })
1062                });
1063                proxy["tls"] = json!({
1064                    "enabled": true,
1065                    "server_name": ext_proxy.sni,
1066                    "insecure": ext_proxy.tls_insecure,
1067                    "alpn": if protocol == "trojan" && !quic { None } else { Some(["http/1.1"]) },
1068                    "certificate": certificate,
1069                    "certificate_path": certificate_path,
1070                    "ech": ech,
1071                    "utls": utls(&ext_proxy.client_fingerprint),
1072                });
1073                if quic {
1074                    proxy["transport"] = json!({ "type": "quic" });
1075                } else if ext_proxy.ws.unwrap_or(false) {
1076                    let headers = ext_proxy
1077                        .ws_host
1078                        .as_ref()
1079                        .map(|host| json!({ "Host": host }));
1080                    proxy["transport"] = json!({
1081                        "type": "ws",
1082                        "path": ext_proxy.ws_path.as_deref().unwrap_or("/"),
1083                        "headers": headers,
1084                    });
1085                }
1086                if ext_proxy.amux.unwrap_or(false) && !quic {
1087                    proxy["multiplex"] = json!({
1088                        "enabled": true,
1089                        "protocol": "amux",
1090                        "max_accepts": ext_proxy.amux_max,
1091                        "concurrency": ext_proxy.amux_con,
1092                        "max_recv_bytes": ext_proxy.amux_max_recv,
1093                        "max_lifetime": ext_proxy.amux_max_lifetime,
1094                    });
1095                }
1096                outbounds.push(outbound(tag, protocol, proxy));
1097            }
1098            other => {
1099                return Err(anyhow!(
1100                    "[Proxy] {}: unsupported proxy type \"{}\"",
1101                    tag,
1102                    other
1103                ))
1104            }
1105        }
1106        if let Some(interface) = &ext_proxy.interface {
1107            if ext_proxy.protocol == "drop" {
1108                return Err(anyhow!(
1109                    "[Proxy] {}: interface: a drop proxy sends nothing",
1110                    tag
1111                ));
1112            }
1113            let (field, value) = match interface.parse::<std::net::IpAddr>() {
1114                Ok(std::net::IpAddr::V4(_)) => ("inet4_bind_address", interface),
1115                Ok(std::net::IpAddr::V6(_)) => ("inet6_bind_address", interface),
1116                Err(_) => ("bind_interface", interface),
1117            };
1118            for outbound in &mut outbounds[first_new..] {
1119                outbound.options.insert(field.into(), json!(value));
1120            }
1121        }
1122    }
1123
1124    for ext_proxy_group in conf.proxy_group.iter().flatten() {
1125        let tag = ext_proxy_group.tag.as_str();
1126        let members = &ext_proxy_group.actors;
1127        let mut params = GroupParams::new(ext_proxy_group);
1128        let group = match ext_proxy_group.protocol.as_str() {
1129            "chain" => {
1130                // Each hop after the first is a copy of its proxy that dials
1131                // through the hop before; the last copy is the group.
1132                let hops = members.as_deref().unwrap_or_default();
1133                if hops.len() < 2 {
1134                    return Err(anyhow!(
1135                        "[Proxy Group] {}: a chain needs at least two proxies",
1136                        tag
1137                    ));
1138                }
1139                let mut previous = hops[0].clone();
1140                for (i, hop) in hops.iter().enumerate().skip(1) {
1141                    let proxy = config
1142                        .outbounds
1143                        .iter()
1144                        .find(|o| &o.tag == hop)
1145                        .filter(|o| o.options.contains_key("server"))
1146                        .ok_or_else(|| {
1147                            anyhow!(
1148                                "[Proxy Group] {}: [{}] is not a proxy defined in [Proxy]",
1149                                tag,
1150                                hop
1151                            )
1152                        })?;
1153                    let mut copy = proxy.clone();
1154                    if copy.options.contains_key("detour") {
1155                        return Err(anyhow!(
1156                            "[Proxy Group] {}: [{}] already dials through another proxy",
1157                            tag,
1158                            hop
1159                        ));
1160                    }
1161                    copy.tag = if i + 1 == hops.len() {
1162                        tag.to_string()
1163                    } else {
1164                        format!("{}/{}", tag, hop)
1165                    };
1166                    copy.options.insert("detour".into(), json!(previous));
1167                    previous = copy.tag.clone();
1168                    config.outbounds.push(copy);
1169                }
1170                params.finish()?;
1171                continue;
1172            }
1173            "tryall" => outbound(
1174                tag,
1175                "tryall",
1176                json!({ "outbounds": members, "delay_base": params.number("delay-base")? }),
1177            ),
1178            "static" => {
1179                return Err(anyhow!(
1180                    "[Proxy Group] {}: there is no static group; use load-balance",
1181                    tag
1182                ))
1183            }
1184            "select" => outbound(tag, "selector", json!({ "outbounds": members })),
1185            "url-test" => outbound(
1186                tag,
1187                "urltest",
1188                json!({
1189                    "outbounds": members,
1190                    "url": params.take("url"),
1191                    "interval": params.seconds("interval"),
1192                    "tolerance": params.number("tolerance")?,
1193                    "idle_timeout": params.seconds("idle-timeout"),
1194                    "interrupt_exist_connections": params.flag("interrupt-exist-connections")?,
1195                }),
1196            ),
1197            "fallback" => outbound(
1198                tag,
1199                "fallback",
1200                json!({
1201                    "outbounds": members,
1202                    "url": params.take("url"),
1203                    "interval": params.seconds("interval"),
1204                    "timeout": params.seconds("timeout"),
1205                    "lazy": params.flag("lazy")?,
1206                    "interrupt_exist_connections": params.flag("interrupt-exist-connections")?,
1207                }),
1208            ),
1209            "load-balance" => outbound(
1210                tag,
1211                "load-balance",
1212                json!({
1213                    "outbounds": members,
1214                    "strategy": params.take("strategy"),
1215                    "url": params.take("url"),
1216                    "interval": params.seconds("interval"),
1217                    "lazy": params.flag("lazy")?,
1218                }),
1219            ),
1220            "mptp" => outbound(
1221                tag,
1222                "mptp",
1223                json!({
1224                    "outbounds": members,
1225                    "server": params.take("address"),
1226                    "server_port": params.number("port")?,
1227                }),
1228            ),
1229            other => {
1230                return Err(anyhow!(
1231                    "[Proxy Group] {}: unsupported group type \"{}\"",
1232                    tag,
1233                    other
1234                ))
1235            }
1236        };
1237        params.finish()?;
1238        config.outbounds.push(group);
1239    }
1240
1241    if conf.general.as_ref().and_then(|g| g.sniffing) == Some(true) {
1242        config.route.rules.push(model::Rule {
1243            action: model::RuleAction::Sniff,
1244            ..Default::default()
1245        });
1246    }
1247    if let Some(reverse_mapping) = conf.general.as_ref().and_then(|g| g.dns_reverse_mapping) {
1248        config.dns.reverse_mapping = reverse_mapping;
1249    }
1250    for ext_rule in conf.rule.iter().flatten() {
1251        if ext_rule.type_field == "FINAL" {
1252            config.route.final_outbound = Some(ext_rule.target.clone());
1253            continue;
1254        }
1255        let mut rule = model::Rule {
1256            outbound: Some(ext_rule.target.clone()),
1257            ..Default::default()
1258        };
1259        let filter = ext_rule.filter.clone().ok_or_else(|| {
1260            anyhow!(
1261                "[Rule] {},{}: missing the value to match",
1262                ext_rule.type_field,
1263                ext_rule.target
1264            )
1265        })?;
1266        let condition = match ext_rule.type_field.as_str() {
1267            "IP-CIDR" => &mut rule.ip_cidr,
1268            "DOMAIN" => &mut rule.domain,
1269            "DOMAIN-KEYWORD" => &mut rule.domain_keyword,
1270            "DOMAIN-SUFFIX" => &mut rule.domain_suffix,
1271            "GEOIP" => &mut rule.geoip,
1272            "EXTERNAL" => &mut rule.external,
1273            "PORT-RANGE" => &mut rule.port_range,
1274            "NETWORK" => &mut rule.network,
1275            "INBOUND-TAG" => &mut rule.inbound,
1276            "PROCESS-NAME" => &mut rule.process_name,
1277            other => return Err(anyhow!("[Rule] unsupported rule type \"{}\"", other)),
1278        };
1279        condition.push(filter);
1280        config.route.rules.push(rule);
1281    }
1282    // A domain no rule matches is resolved, and the address rules are
1283    // tried again with its addresses.
1284    if conf
1285        .general
1286        .as_ref()
1287        .and_then(|g| g.routing_domain_resolve)
1288        .unwrap_or(false)
1289    {
1290        let by_address: Vec<model::Rule> = config
1291            .route
1292            .rules
1293            .iter()
1294            .filter(|r| {
1295                !r.ip_cidr.is_empty()
1296                    || !r.geoip.is_empty()
1297                    || r.external.iter().any(|e| e.starts_with("mmdb:"))
1298            })
1299            .cloned()
1300            .collect();
1301        config.route.rules.push(model::Rule {
1302            action: model::RuleAction::Resolve,
1303            ..Default::default()
1304        });
1305        config.route.rules.extend(by_address);
1306    }
1307
1308    if let Some(servers) = conf.general.as_ref().and_then(|g| g.dns_server.clone()) {
1309        config.dns.servers = servers;
1310    }
1311    if let Some(ipv6) = conf.general.as_ref().and_then(|g| g.ipv6) {
1312        config.dns.strategy = if ipv6 {
1313            model::DnsStrategy::PreferIpv4
1314        } else {
1315            model::DnsStrategy::Ipv4Only
1316        };
1317    }
1318    if let Some(general) = &conf.general {
1319        match (&general.api_interface, general.api_port) {
1320            (Some(interface), Some(port)) => {
1321                let ip = interface.parse::<std::net::IpAddr>().map_err(|_| {
1322                    anyhow!("api-interface: \"{}\" is not an IP address", interface)
1323                })?;
1324                config.api.listen = Some((ip, port).into());
1325            }
1326            (None, None) => {}
1327            _ => return Err(anyhow!("api-interface and api-port go together")),
1328        }
1329    }
1330    if let Some(hosts) = &conf.host {
1331        config.dns.hosts = hosts.clone();
1332    }
1333
1334    config.validate()?;
1335    Ok(config)
1336}

1338fn to_log(general: &General) -> Result<model::Log> {
1339    let level = match general.loglevel.as_deref() {
1340        None => model::LogLevel::default(),
1341        Some(level) => serde_json::from_value(json!(level.to_lowercase()))
1342            .map_err(|_| anyhow!("[General] loglevel: unknown level \"{}\"", level))?,
1343    };
1344    let format = match general.logformat.as_deref() {
1345        None => model::LogFormat::default(),
1346        Some(format) => serde_json::from_value(json!(format.to_lowercase()))
1347            .map_err(|_| anyhow!("[General] logformat: unknown format \"{}\"", format))?,
1348    };
1349    let output = general
1350        .logoutput
1351        .clone()
1352        .filter(|output| output != "console");
1353    Ok(model::Log {
1354        level,
1355        output,
1356        format,
1357    })
1358}

1360/// The options in `value`, less the ones the `.conf` left out, at any depth.
1361fn options(value: serde_json::Value) -> model::Options {
1362    match options_value(value) {
1363        serde_json::Value::Object(map) => map,
1364        _ => unreachable!("options are always an object"),
1365    }
1366}
1367
1368fn options_value(value: serde_json::Value) -> serde_json::Value {
1369    match value {
1370        serde_json::Value::Object(map) => serde_json::Value::Object(
1371            map.into_iter()
1372                .filter(|(_, v)| !v.is_null())
1373                .map(|(k, v)| (k, options_value(v)))
1374                .collect(),
1375        ),
1376        other => other,
1377    }
1378}

1396/// The `key=value` parameters of a `[Proxy Group]` line, taken one by
1397/// one by what its type knows: any left over is a mistake.
1398struct GroupParams<'a> {
1399    tag: &'a str,
1400    params: Vec<(&'a str, &'a str)>,
1401}
1402
1403impl<'a> GroupParams<'a> {
1404    fn new(group: &'a ProxyGroup) -> Self {
1405        Self {
1406            tag: &group.tag,
1407            params: group
1408                .params
1409                .iter()
1410                .map(|(k, v)| (k.as_str(), v.as_str()))
1411                .collect(),
1412        }
1413    }
1414
1415    fn take(&mut self, key: &str) -> Option<&'a str> {
1416        let i = self.params.iter().position(|(k, _)| *k == key)?;
1417        Some(self.params.remove(i).1)
1418    }
1419
1420    fn number(&mut self, key: &str) -> Result<Option<u64>> {
1421        self.take(key)
1422            .map(|v| {
1423                v.parse().map_err(|_| {
1424                    anyhow!("[Proxy Group] {}: {}: not a number: {}", self.tag, key, v)
1425                })
1426            })
1427            .transpose()
1428    }
1429
1430    fn flag(&mut self, key: &str) -> Result<Option<bool>> {
1431        self.take(key)
1432            .map(|v| match v {
1433                "true" => Ok(true),
1434                "false" => Ok(false),
1435                _ => Err(anyhow!(
1436                    "[Proxy Group] {}: {}: expected true or false: {}",
1437                    self.tag,
1438                    key,
1439                    v
1440                )),
1441            })
1442            .transpose()
1443    }
1444
1445    /// A duration: a bare number is seconds, as Surge writes them; one
1446    /// with a unit (`500ms`) is passed on as it is.
1447    fn seconds(&mut self, key: &str) -> Option<String> {
1448        self.take(key).map(|v| match v.parse::<u64>() {
1449            Ok(n) => format!("{}s", n),
1450            Err(_) => v.to_string(),
1451        })
1452    }
1453
1454    fn finish(self) -> Result<()> {
1455        match self.params.first() {
1456            Some((k, _)) => Err(anyhow!(
1457                "[Proxy Group] {}: unknown parameter \"{}\"",
1458                self.tag,
1459                k
1460            )),
1461            None => Ok(()),
1462        }
1463    }
1464}

1467    model::Outbound {
1468        protocol: protocol.to_string(),
1469        tag: tag.to_string(),
1470        options: options(value),
1471    }
1472}
