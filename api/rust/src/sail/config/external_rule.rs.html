12/// How a domain condition compares against a destination.
13#[derive(Debug, Clone, Copy, PartialEq, Eq)]
14pub enum DomainKind {
15    /// The domain contains the value.
16    Keyword,
17    /// The domain is the value or a subdomain of it.
18    Suffix,
19    /// The domain is exactly the value.
20    Full,
21}
22
23/// A country in a GeoIP database.
24#[derive(Debug, Clone, PartialEq, Eq)]
25pub struct Mmdb {
26    pub file: String,
27    pub country_code: String,
28}
29
30/// What an external rule adds to a rule.
31pub enum External {
32    Mmdb(Mmdb),
33    Domains(Vec<(DomainKind, String)>),
34}
35
36/// `code` in the default GeoIP database.
37pub fn geoip(code: &str, env: &RuntimeEnv) -> Mmdb {
38    Mmdb {
39        file: env.data_path("geo.mmdb"),
40        country_code: code.to_string(),
41    }
42}

44/// The site group `code` in the default site list.
45pub fn geosite(code: &str, env: &RuntimeEnv) -> Result<Vec<(DomainKind, String)>> {
46    load_site_group(&env.data_path("site.dat"), code)
47}

49/// `mmdb:<code>`, `mmdb:<file>:<code>`, `site:<code>` or `site:<file>:<code>`.
50/// A relative file is looked up in the data directory.
51pub fn load(filter: &str, env: &RuntimeEnv) -> Result<External> {
52    let parts: Vec<&str> = filter.split(':').collect();
53    let (kind, file, code) = match parts.as_slice() {
54        [kind, code] => (*kind, None, *code),
55        [kind, file, code] => (*kind, Some(*file), *code),
56        _ => return Err(anyhow!("invalid external rule \"{}\"", filter)),
57    };
58    match kind {
59        "mmdb" => Ok(External::Mmdb(Mmdb {
60            file: env.data_path(file.unwrap_or("geo.mmdb")),
61            country_code: code.to_string(),
62        })),
63        "site" => Ok(External::Domains(load_site_group(
64            &env.data_path(file.unwrap_or("site.dat")),
65            code,
66        )?)),
67        _ => Err(anyhow!(
68            "invalid external rule \"{}\": expected mmdb:... or site:...",
69            filter
70        )),
71    }
72}

74/// The domains of the site group `code` in `file`.
75fn load_site_group(file: &str, code: &str) -> Result<Vec<(DomainKind, String)>> {
76    // Loads SiteGroup objects one by one instead of loading the whole list.
77    let mut reader = BufReader::with_capacity(
78        2048,
79        File::open(file).map_err(|e| anyhow!("open site list {} failed: {}", file, e))?,
80    );
81    let mut input = protobuf::CodedInputStream::new(&mut reader);
82    while !input.eof()? {
83        let _ = input.read_raw_byte()?; // skip
84        let mut site_group = input.read_message::<geosite::SiteGroup>()?;
85        if site_group.tag != code.to_uppercase() {
86            continue;
87        }
88        let mut domains = Vec::with_capacity(site_group.domain.len());
89        for domain in site_group.domain.iter_mut() {
90            let kind = match domain.type_.enum_value() {
91                Ok(geosite::domain::Type::Plain) => DomainKind::Keyword,
92                Ok(geosite::domain::Type::Domain) => DomainKind::Suffix,
93                Ok(geosite::domain::Type::Full) => DomainKind::Full,
94                // Regular expressions are not supported.
95                _ => continue,
96            };
97            domains.push((kind, std::mem::take(&mut domain.value)));
98        }
99        tracing::debug!(
100            "loaded {} domain rules from [{}] for tag [{}]",
101            domains.len(),
102            file,
103            code
104        );
105        return Ok(domains);
106    }
107    Err(anyhow!("no site group [{}] in {}", code, file))
108}
