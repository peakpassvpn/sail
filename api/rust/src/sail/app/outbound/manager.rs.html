23/// The outbounds of an instance. It is not changed in place: a change
24/// makes a new manager, which replaces this one in the instance's
25/// snapshot.
26#[derive(Clone)]
27pub struct OutboundManager {
28    handlers: HashMap<String, AnyOutboundHandler>,
29    /// Keeps the plugin libraries the handlers come from loaded.
30    #[cfg(feature = "plugin")]
31    external_handlers: Vec<Arc<super::plugin::ExternalHandlers>>,
32    #[cfg(feature = "outbound-select")]
33    selectors: Arc<super::Selectors>,
34    default_handler: Option<String>,
35    /// The tasks each outbound's handler spawned.
36    abort_handles: HashMap<String, Vec<AbortHandle>>,
37    /// The outbounds each outbound is built on.
38    dependencies: HashMap<String, Vec<String>>,
39}
40
41impl OutboundManager {
42    /// Builds `outbounds`; their sockets are opened with `dial_defaults`
43    /// where their own dial fields leave off.
44    pub fn new(
45        outbounds: &[Outbound],
46        dial_defaults: &DialOptions,
47        env: &RuntimeEnv,
48        dns_client: SyncDnsClient,
49    ) -> Result<Self> {
50        let empty = OutboundManager {
51            handlers: HashMap::new(),
52            #[cfg(feature = "plugin")]
53            external_handlers: Vec::new(),
54            #[cfg(feature = "outbound-select")]
55            selectors: Arc::new(HashMap::new()),
56            // The first outbound in the configuration is the default one.
57            default_handler: outbounds.first().map(|o| o.tag.clone()),
58            abort_handles: HashMap::new(),
59            dependencies: HashMap::new(),
60        };
61        if let Some(tag) = &empty.default_handler {
62            tracing::debug!("default handler [{}]", tag);
63        }
64        empty.with(outbounds, dial_defaults, env, dns_client)
65    }
66
67    /// This manager with `outbounds` built onto it: they may be built on
68    /// the outbounds here, but not take their tags.
69    fn with(
70        &self,
71        outbounds: &[Outbound],
72        dial_defaults: &DialOptions,
73        env: &RuntimeEnv,
74        dns_client: SyncDnsClient,
75    ) -> Result<Self> {
76        let mut next = self.clone();
77        #[cfg(feature = "plugin")]
78        let mut external_handlers = super::plugin::ExternalHandlers::new();
79        #[cfg(feature = "outbound-select")]
80        let mut selectors = (*self.selectors).clone();
81        registry::build_outbounds(
82            &include::OUTBOUNDS,
83            outbounds,
84            OutboundBuildState {
85                dns_client: &dns_client,
86                dial_defaults,
87                env,
88                handlers: &mut next.handlers,
89                abort_handles: &mut next.abort_handles,
90                dependencies: &mut next.dependencies,
91                #[cfg(feature = "outbound-select")]
92                selectors: &mut selectors,
93                #[cfg(feature = "plugin")]
94                external_handlers: &mut external_handlers,
95            },
96        )?;
97        #[cfg(feature = "plugin")]
98        next.external_handlers.push(Arc::new(external_handlers));
99        #[cfg(feature = "outbound-select")]
100        {
101            next.selectors = Arc::new(selectors);
102        }
103        Ok(next)
104    }

106    /// This manager with `outbound` added.
107    pub fn with_outbound(
108        &self,
109        outbound: &Outbound,
110        dial_defaults: &DialOptions,
111        env: &RuntimeEnv,
112        dns_client: SyncDnsClient,
113    ) -> Result<Self> {
114        self.with(
115            std::slice::from_ref(outbound),
116            dial_defaults,
117            env,
118            dns_client,
119        )
120    }

122    /// This manager without the outbound `tag`, which nothing else may be
123    /// built on, and the tasks to stop once it is replaced.
124    pub fn without_outbound(&self, tag: &str) -> Result<(Self, Vec<AbortHandle>)> {
125        let Some(removed) = self.handlers.get(tag) else {
126            return Err(anyhow!("[{}] outbound: does not exist", tag));
127        };
128        if let Some((user, _)) = self
129            .dependencies
130            .iter()
131            .find(|(user, deps)| user.as_str() != tag && deps.iter().any(|d| d == tag))
132        {
133            return Err(anyhow!("[{}] outbound: [{}] is built on it", tag, user));
134        }
135        if self.default_handler.as_deref() == Some(tag) {
136            return Err(anyhow!(
137                "[{}] outbound: it is the default outbound, the first configured",
138                tag
139            ));
140        }
141        let mut next = self.clone();
142        next.handlers.remove(tag);
143        next.dependencies.remove(tag);
144        let mut tasks = next.abort_handles.remove(tag).unwrap_or_default();
145        // An outbound identical to another shares its handler, and the
146        // tasks go on for the one left.
147        if let Some((twin, _)) = next.handlers.iter().find(|(_, h)| Arc::ptr_eq(h, removed)) {
148            next.abort_handles
149                .entry(twin.clone())
150                .or_default()
151                .append(&mut tasks);
152        }
153        #[cfg(feature = "outbound-select")]
154        if next.selectors.contains_key(tag) {
155            let mut selectors = (*next.selectors).clone();
156            selectors.remove(tag);
157            next.selectors = Arc::new(selectors);
158        }
159        Ok((next, tasks))
160    }

162    /// Selects what `previous`, the manager this one replaces, had
163    /// selected, where the same selector still offers it.
164    #[cfg(feature = "outbound-select")]
165    pub async fn restore_selected(&self, previous: &OutboundManager) {
166        for (tag, selector) in self.selectors.iter() {
167            if let Some(old) = previous.selectors.get(tag) {
168                let selected = old.read().await.get_selected_tag();
169                let _ = selector.write().await.set_selected(&selected);
170            }
171        }
172    }
173
174    /// Stops the tasks the handlers started, once they are replaced.
175    pub fn abort_tasks(&self) {
176        for abort_handle in self.abort_handles.values().flatten() {
177            abort_handle.abort();
178        }
179    }
180
181    pub fn get(&self, tag: &str) -> Option<AnyOutboundHandler> {
182        self.handlers.get(tag).map(Clone::clone)
183    }
184
185    pub fn default_handler(&self) -> Option<String> {
186        self.default_handler.clone()
187    }
188
189    pub fn handlers(&self) -> Handlers<'_> {
190        Handlers {
191            inner: self.handlers.values(),
192        }
193    }
194
195    #[cfg(feature = "outbound-select")]
196    pub fn get_selector(&self, tag: &str) -> Option<Arc<RwLock<OutboundSelector>>> {
197        self.selectors.get(tag).map(Clone::clone)
198    }
199}

201pub struct Handlers<'a> {
202    inner: hash_map::Values<'a, String, AnyOutboundHandler>,
203}
204
205impl<'a> Iterator for Handlers<'a> {
206    type Item = &'a AnyOutboundHandler;
207
208    fn next(&mut self) -> Option<Self::Item> {
209        self.inner.next()
210    }
