1//! Routing: the rules of `route`, compiled, and matched in order against
2//! what is known about a connection. A rule's action either decides where
3//! the connection goes (`route`, `reject`) or learns more about it
4//! (`sniff`, `resolve`) and lets the next rules decide.

23/// Where a connection goes.
24#[derive(Debug, Clone, PartialEq, Eq)]
25pub enum Decision {
26    /// To this outbound; to the default one when `None`.
27    Route(Option<String>),
28    /// Nowhere: it is closed.
29    Reject,
30}
31
32/// What a `sniff` rule asks for.
33#[derive(Debug, Clone, PartialEq, Eq)]
34pub struct SniffAction {
35    pub tls: bool,
36    pub http: bool,
37    /// How long to wait for the first bytes.
38    pub timeout: Duration,
39    /// Connects to the sniffed domain rather than to the address asked for.
40    pub override_destination: bool,
41}
42
43/// Reads what a `sniff` rule asks for from a connection, into its session.
44/// Only the dispatcher, which holds the connection, can.
45#[async_trait]
46pub trait Sniffer: Send {
47    async fn sniff(&mut self, sess: &mut Session, action: &SniffAction) -> io::Result<()>;
48}
49
50/// For connections nothing can be read from, such as UDP.
51pub struct NoSniffer;
52
53#[async_trait]
54impl Sniffer for NoSniffer {
55    async fn sniff(&mut self, _sess: &mut Session, _action: &SniffAction) -> io::Result<()> {
56        Ok(())
57    }
58}
59
60enum Action {
61    Route(String),
62    Reject,
63    Sniff(SniffAction),
64    Resolve,
65}
66
67struct Rule {
68    matcher: Matcher,
69    action: Action,
70}
71
72impl Rule {
73    fn new(rule: &model::Rule, readers: &mut Readers, env: &RuntimeEnv) -> Result<Self> {
74        let action = match rule.action {
75            RuleAction::Route => Action::Route(
76                rule.outbound
77                    .clone()
78                    .ok_or_else(|| anyhow!("outbound: a route rule needs one"))?,
79            ),
80            RuleAction::Reject => Action::Reject,
81            RuleAction::Resolve => Action::Resolve,
82            RuleAction::Sniff => {
83                let all = rule.sniffer.is_empty();
84                Action::Sniff(SniffAction {
85                    tls: all || rule.sniffer.contains(&model::Sniffer::Tls),
86                    http: all || rule.sniffer.contains(&model::Sniffer::Http),
87                    timeout: rule.timeout.unwrap_or(Duration::from_millis(300)),
88                    override_destination: rule.override_destination,
89                })
90            }
91        };
92        Ok(Rule {
93            matcher: Matcher::new(rule, readers, env)?,
94            action,
95        })
96    }
97}

99pub struct Router {
100    rules: Vec<Rule>,
101    final_outbound: Option<String>,
102    dns_client: SyncDnsClient,
103}
104
105impl Router {
106    fn load_rules(route: &model::Route, env: &RuntimeEnv) -> Result<Vec<Rule>> {
107        let mut readers = Readers::new();
108        route
109            .rules
110            .iter()
111            .enumerate()
112            .map(|(i, rule)| {
113                Rule::new(rule, &mut readers, env).map_err(|e| anyhow!("route.rules[{}]: {}", i, e))
114            })
115            .collect()
116    }
117
118    pub fn new(route: &model::Route, dns_client: SyncDnsClient, env: &RuntimeEnv) -> Result<Self> {
119        Ok(Router {
120            rules: Self::load_rules(route, env)?,
121            final_outbound: route.final_outbound.clone(),
122            dns_client,
123        })
124    }
125
126    pub fn reload(&mut self, route: &model::Route, env: &RuntimeEnv) -> Result<()> {
127        self.rules = Self::load_rules(route, env)?;
128        self.final_outbound = route.final_outbound.clone();
129        Ok(())
130    }
131
132    /// Whether a rule, or `final`, routes to the outbound `tag`.
133    pub fn uses(&self, tag: &str) -> bool {
134        self.final_outbound.as_deref() == Some(tag)
135            || self
136                .rules
137                .iter()
138                .any(|rule| matches!(&rule.action, Action::Route(t) if t == tag))
139    }
140
141    /// Matches `sess` against the rules in order, sniffing through
142    /// `sniffer` and resolving as they say, until one decides.
143    pub async fn pick_route(
144        &self,
145        sess: &mut Session,
146        sniffer: &mut dyn Sniffer,
147    ) -> Result<Decision> {
148        let mut resolved: Vec<IpAddr> = Vec::new();
149        let mut facts = Facts::new(sess, &resolved);
150        for (i, rule) in self.rules.iter().enumerate() {
151            if !rule.matcher.matches(&facts) {
152                continue;
153            }
154            match &rule.action {
155                Action::Route(tag) => {
156                    debug!("rule {} routes to {}", i, tag);
157                    return Ok(Decision::Route(Some(tag.clone())));
158                }
159                Action::Reject => {
160                    debug!("rule {} rejects", i);
161                    return Ok(Decision::Reject);
162                }
163                Action::Sniff(action) => {
164                    sniffer
165                        .sniff(sess, action)
166                        .await
167                        .map_err(|e| anyhow!("sniff: {}", e))?;
168                }
169                Action::Resolve => {
170                    if resolved.is_empty() && !sess.skip_resolve {
171                        if let Some(domain) = facts.domain() {
172                            resolved = self.resolve(domain).await;
173                        }
174                    }
175                }
176            }
177            facts = Facts::new(sess, &resolved);
178        }
179        Ok(Decision::Route(self.final_outbound.clone()))
180    }
181
182    /// The addresses of `domain`, or none when it does not resolve: the
183    /// rules after a `resolve` then match without them.
184    async fn resolve(&self, domain: &str) -> Vec<IpAddr> {
185        match self
186            .dns_client
187            .load_full()
188            .lookup(&domain.to_string())
189            .await
190        {
191            Ok(ips) => {
192                debug!("resolved {} to {:?} for routing", domain, ips);
193                ips
194            }
195            Err(e) => {
196                debug!("resolving {} for routing failed: {}", domain, e);
197                Vec::new()
198            }
199        }
200    }
201}

225    /// Plays a connection whose first bytes carry `domain`.
226    struct FakeSniffer {
227        domain: &'static str,
228        calls: usize,
229    }
230
231    #[async_trait]
232    impl Sniffer for FakeSniffer {
233        async fn sniff(&mut self, sess: &mut Session, action: &SniffAction) -> io::Result<()> {
234            self.calls += 1;
235            if action.tls {
236                sess.set_sniffed_domain(crate::session::SniffedFrom::Tls, self.domain.to_string());
237            }
238            Ok(())
239        }
240    }

242    fn to_ip() -> Session {
243        Session {
244            destination: SocksAddr::from(("1.2.3.4".parse::<IpAddr>().unwrap(), 443)),
245            ..Default::default()
246        }
247    }

249    #[tokio::test]
250    async fn a_sniffed_domain_is_matched_by_the_rules_after_the_sniff() {
251        let router = router(serde_json::json!([
252            { "domain_suffix": ["example.com"], "outbound": "a" },
253            { "action": "sniff", "port_range": ["443"] },
254            { "domain_suffix": ["example.com"], "outbound": "a" },
255        ]));
256        let mut sniffer = FakeSniffer {
257            domain: "www.example.com",
258            calls: 0,
259        };
260        let mut sess = to_ip();
261        let decision = router.pick_route(&mut sess, &mut sniffer).await.unwrap();
262        assert_eq!(decision, Decision::Route(Some("a".into())));
263        assert_eq!(sniffer.calls, 1);
264        assert_eq!(sess.sniffed_domain(), Some("www.example.com"));
265    }

294    #[tokio::test]
295    async fn a_resolved_domain_is_matched_by_address() {
296        let router = router(serde_json::json!([
297            { "action": "resolve" },
298            { "ip_cidr": ["127.0.0.0/8"], "outbound": "a" },

300        let mut sess = Session {
301            destination: SocksAddr::Domain("test.sail".into(), 80),
302            ..Default::default()
303        };
304        let decision = router.pick_route(&mut sess, &mut NoSniffer).await.unwrap();
305        assert_eq!(decision, Decision::Route(Some("a".into())));
306
307        // Not for the lookups the DNS client makes for itself.
308        sess.skip_resolve = true;
309        let decision = router.pick_route(&mut sess, &mut NoSniffer).await.unwrap();
310        assert_eq!(decision, Decision::Route(Some("b".into())));
