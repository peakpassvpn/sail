17/// How long a connection without queries is kept. Neither side sends
18/// keep-alives: a connection that went idle is dropped, and the next query
19/// makes a new one.
20const IDLE_TIMEOUT: Duration = Duration::from_secs(30);
21
22#[derive(Clone, Copy, Debug, PartialEq, Eq)]
23pub(super) enum Kind {
24    /// DNS over QUIC.
25    Doq,
26    /// DNS over HTTP/3.
27    #[cfg(feature = "dns-h3")]
28    H3,
29}
30
31impl Kind {
32    fn alpn(self) -> &'static [u8] {
33        match self {
34            Self::Doq => b"doq",
35            #[cfg(feature = "dns-h3")]
36            Self::H3 => b"h3",
37        }
38    }
39}

41/// The connection of one upstream.
42pub(super) struct Pool {
43    kind: Kind,
44    /// Built on first use, with the certificates the client trusts.
45    client_config: OnceLock<std::result::Result<quinn::ClientConfig, String>>,
46    /// Locked while a connection is made, so that queries at once make one.
47    live: TokioMutex<Option<Live>>,
48}

50/// A QUIC connection, with what runs on it.
51struct Live {
52    /// Kept with its connection: the endpoint owns the socket.
53    _endpoint: quinn::Endpoint,
54    conn: quinn::Connection,
55    #[cfg(feature = "dns-h3")]
56    h3: Option<H3>,
57}

73impl Live {

79impl Pool {
80    pub fn new(kind: Kind) -> Self {
81        Self {
82            kind,
83            client_config: OnceLock::new(),
84            live: TokioMutex::new(None),
85        }
86    }

88    fn client_config(&self, certificate: Option<&str>) -> Result<quinn::ClientConfig> {
89        self.client_config
90            .get_or_init(|| build_client_config(self.kind, certificate).map_err(|e| e.to_string()))
91            .clone()
92            .map_err(|e| anyhow!("quic client config: {}", e))
93    }
94}

96fn build_client_config(kind: Kind, certificate: Option<&str>) -> Result<quinn::ClientConfig> {
97    // As for TLS: the bundled roots, or `certificate` instead, and the
98    // server verified.
99    let crypto = client_crypto(certificate, false, &[kind.alpn().to_vec()])?;
100    let mut client_config = quinn::ClientConfig::new(Arc::new(crypto));
101    let mut transport = quinn::TransportConfig::default();
102    transport.max_idle_timeout(quinn::IdleTimeout::try_from(IDLE_TIMEOUT).ok());
103    transport.keep_alive_interval(None);
104    client_config.transport_config(Arc::new(transport));
105    Ok(client_config)
106}

108impl DnsClient {
109    pub(super) async fn exchange_quic(
110        &self,
111        upstream: &Upstream,
112        pool: &Pool,
113        addr: SocketAddr,
114        is_direct: bool,
115        request: &[u8],
116    ) -> Result<Vec<u8>> {
117        // DoQ requires ID 0 (RFC 9250 §4.2.1), and DoH asks for it so that
118        // answers can be cached (RFC 8484 §4.1). The caller checks the ID
119        // of the answer against the query's, so it is given back here.
120        let id = [request[0], request[1]];
121        let mut request = request.to_vec();
122        request[..2].copy_from_slice(&[0, 0]);
123
124        let mut response = None;
125        // Once on the kept connection, which may be dead without being
126        // known to be, then once on a new one.
127        for fresh in [false, true] {
128            let live = self
129                .quic_connection(upstream, pool, addr, is_direct, fresh)
130                .await?;
131            let res = match pool.kind {
132                Kind::Doq => exchange_doq(&live.conn, &request).await,
133                #[cfg(feature = "dns-h3")]
134                Kind::H3 => match &live.h3 {
135                    Some(h3) => exchange_h3(upstream, h3, &request).await,
136                    None => Err(anyhow!("no http/3 connection")),
137                },
138            };
139            match res {
140                Ok(r) => {
141                    response = Some(r);
142                    break;
143                }
144                Err(e) if !fresh && live.reused => {
145                    debug!("{}: kept connection failed: {}", upstream, e);
146                    pool.invalidate(&live.conn).await;
147                }
148                Err(e) => {
149                    pool.invalidate(&live.conn).await;
150                    return Err(e);
151                }
152            }
153        }
154        let mut response = response.ok_or_else(|| anyhow!("no answer"))?;
155        if response.len() < 2 || response[..2] != [0, 0] {
156            return Err(anyhow!("dns response with a non-zero id"));
157        }
158        response[..2].copy_from_slice(&id);
159        Ok(response)
160    }
161
162    /// The upstream's connection, made now if there is none alive or if
163    /// `fresh`.
164    async fn quic_connection(
165        &self,
166        upstream: &Upstream,
167        pool: &Pool,
168        addr: SocketAddr,
169        is_direct: bool,
170        fresh: bool,
171    ) -> Result<Handle> {
172        let mut live = pool.live.lock().await;
173        if let Some(l) = live.as_ref() {
174            if l.is_alive() && !fresh {
175                return Ok(Handle::of(l, true));
176            }
177        }
178        *live = None;
179        let l = self.connect_quic(upstream, pool, addr, is_direct).await?;
180        let handle = Handle::of(&l, false);
181        *live = Some(l);
182        Ok(handle)
183    }
184
185    async fn connect_quic(
186        &self,
187        upstream: &Upstream,
188        pool: &Pool,
189        addr: SocketAddr,
190        is_direct: bool,
191    ) -> Result<Live> {
192        let client_config = pool.client_config(self.upstream_certificate.as_deref())?;
193        let mut endpoint = if is_direct {
194            endpoint(bind(addr.ip(), &self.dial).await?, None)?
195        } else {
196            // QUIC over the datagrams of the outbound the router picks.
197            let datagram = self.dial_datagram(addr).await?;
198            endpoint_on(
199                Arc::new(super::socket::DatagramSocket::new(datagram, addr)),
200                None,
201            )?
202        };
203        endpoint.set_default_client_config(client_config);
204        let connecting = endpoint
205            .connect(addr, &upstream.host)
206            .map_err(|e| anyhow!("connect quic failed: {}", e))?;
207        let conn = timeout(self.dial.connect_timeout, connecting)
208            .await
209            .map_err(|_| anyhow!("quic handshake timed out"))?
210            .map_err(|e| anyhow!("quic handshake failed: {}", e))?;
211        debug!("{}: connected over quic to {}", upstream, addr);
212        #[cfg(feature = "dns-h3")]
213        let h3 = match pool.kind {
214            Kind::H3 => Some(connect_h3(conn.clone()).await?),
215            Kind::Doq => None,
216        };
217        Ok(Live {
218            _endpoint: endpoint,
219            conn,
220            #[cfg(feature = "dns-h3")]
221            h3,
222        })
223    }
224}

226impl Pool {
227    /// Drops the kept connection if it is still `conn`.
228    async fn invalidate(&self, conn: &quinn::Connection) {
229        let mut live = self.live.lock().await;
230        if live
231            .as_ref()
232            .is_some_and(|l| l.conn.stable_id() == conn.stable_id())
233        {
234            if let Some(l) = live.take() {
235                l.conn.close(0u32.into(), b"");
236            }
237        }
238    }
239}

241/// What a query needs of the kept connection, taken without holding the
242/// pool's lock.
243struct Handle {
244    conn: quinn::Connection,
245    #[cfg(feature = "dns-h3")]
246    h3: Option<H3Handle>,
247    /// Whether the connection was made for an earlier query.
248    reused: bool,
249}

251#[cfg(feature = "dns-h3")]
252struct H3Handle {
253    send_request: h3::client::SendRequest<h3_quinn::OpenStreams, bytes::Bytes>,
254}
255
256impl Handle {
257    fn of(live: &Live, reused: bool) -> Self {
258        Self {
259            conn: live.conn.clone(),
260            #[cfg(feature = "dns-h3")]
261            h3: live.h3.as_ref().map(|h3| H3Handle {
262                send_request: h3.send_request.clone(),
263            }),
264            reused,
265        }
266    }
267}

269/// A query on its own stream, each message prefixed with its length
270/// (RFC 9250 §4.2).
271async fn exchange_doq(conn: &quinn::Connection, request: &[u8]) -> Result<Vec<u8>> {
272    let len = u16::try_from(request.len()).map_err(|_| anyhow!("dns query too long"))?;
273    let (mut send, mut recv) = conn
274        .open_bi()
275        .await
276        .map_err(|e| anyhow!("open quic stream failed: {}", e))?;
277    let mut buf = Vec::with_capacity(2 + request.len());
278    buf.extend_from_slice(&len.to_be_bytes());
279    buf.extend_from_slice(request);
280    send.write_all(&buf)
281        .await
282        .map_err(|e| anyhow!("write doq query failed: {}", e))?;
283    // The query ends the client's side of the stream (RFC 9250 §4.2).
284    send.finish()
285        .map_err(|e| anyhow!("finish doq query failed: {}", e))?;
286    let data = recv
287        .read_to_end(2 + MAX_MESSAGE_LEN)
288        .await
289        .map_err(|e| anyhow!("read doq answer failed: {}", e))?;
290    let (len, response) = match data.split_first_chunk::<2>() {
291        Some((len, response)) => (u16::from_be_bytes(*len) as usize, response),
292        None => return Err(anyhow!("doq answer too short")),
293    };
294    if response.len() != len {
295        return Err(anyhow!(
296            "doq answer of {} bytes, {} announced",
297            response.len(),
298            len
299        ));
300    }
301    Ok(response.to_vec())
302}

304#[cfg(feature = "dns-h3")]
305async fn connect_h3(conn: quinn::Connection) -> Result<H3> {
306    let (mut driver, send_request) = h3::client::new(h3_quinn::Connection::new(conn))
307        .await
308        .map_err(|e| anyhow!("http/3 connection failed: {}", e))?;
309    let driver = tokio::spawn(async move {
310        let e = futures::future::poll_fn(|cx| driver.poll_close(cx)).await;
311        debug!("http/3 connection closed: {}", e);
312    })
313    .abort_handle();
314    Ok(H3 {
315        send_request,
316        driver,
317    })
318}

320/// A POST of the query to the upstream's path (RFC 8484 §4.1).
321#[cfg(feature = "dns-h3")]
322async fn exchange_h3(upstream: &Upstream, h3: &H3Handle, request: &[u8]) -> Result<Vec<u8>> {
323    use bytes::Buf;
324    use http::header::{ACCEPT, CONTENT_LENGTH, CONTENT_TYPE};
325
326    const DNS_MESSAGE: &str = "application/dns-message";
327    let req = http::Request::post(upstream.uri())
328        .header(CONTENT_TYPE, DNS_MESSAGE)
329        .header(ACCEPT, DNS_MESSAGE)
330        .header(CONTENT_LENGTH, request.len())
331        .body(())
332        .map_err(|e| anyhow!("invalid http/3 request: {}", e))?;
333    let mut send_request = h3.send_request.clone();
334    let mut stream = send_request
335        .send_request(req)
336        .await
337        .map_err(|e| anyhow!("send http/3 request failed: {}", e))?;
338    stream
339        .send_data(bytes::Bytes::copy_from_slice(request))
340        .await
341        .map_err(|e| anyhow!("send http/3 body failed: {}", e))?;
342    stream
343        .finish()
344        .await
345        .map_err(|e| anyhow!("finish http/3 request failed: {}", e))?;
346    let resp = stream
347        .recv_response()
348        .await
349        .map_err(|e| anyhow!("read http/3 response failed: {}", e))?;
350    if resp.status() != http::StatusCode::OK {
351        return Err(anyhow!(
352            "doh3 server returned http status {}",
353            resp.status().as_u16()
354        ));
355    }
356    let mut body = Vec::new();
357    while let Some(mut chunk) = stream
358        .recv_data()
359        .await
360        .map_err(|e| anyhow!("read http/3 body failed: {}", e))?
361    {
362        if body.len() + chunk.remaining() > MAX_MESSAGE_LEN {
363            return Err(anyhow!("doh3 answer too long"));
364        }
365        while chunk.has_remaining() {
366            let part = chunk.chunk();
367            body.extend_from_slice(part);
368            let n = part.len();
369            chunk.advance(n);
370        }
371    }
372    Ok(body)
