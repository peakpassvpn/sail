17/// Packets queued each way. Past that, packets are dropped, as a socket
18/// buffer would drop them, and QUIC sends them again.
19const QUEUE: usize = 256;
20/// The largest datagram.
21const MAX_DATAGRAM: usize = 65535;
22
23/// Talks to one peer, `peer`, whatever address the outbound reports its
24/// packets from.
25pub(super) struct DatagramSocket {
26    peer: SocketAddr,
27    tx: mpsc::Sender<Vec<u8>>,
28    rx: Mutex<mpsc::Receiver<Vec<u8>>>,
29    tasks: [tokio::task::AbortHandle; 2],
30}
31
32impl DatagramSocket {
33    pub fn new(datagram: AnyOutboundDatagram, peer: SocketAddr) -> Self {
34        let (mut recv_half, mut send_half) = datagram.split();
35        let (tx, mut send_rx) = mpsc::channel::<Vec<u8>>(QUEUE);
36        let (recv_tx, rx) = mpsc::channel::<Vec<u8>>(QUEUE);
37        let dst = SocksAddr::from(peer);
38        let sender = tokio::spawn(async move {
39            while let Some(packet) = send_rx.recv().await {
40                if let Err(e) = send_half.send_to(&packet, &dst).await {
41                    debug!("send quic packet through outbound failed: {}", e);
42                    break;
43                }
44            }
45            let _ = send_half.close().await;
46        });
47        let receiver = tokio::spawn(async move {
48            let mut buf = vec![0u8; MAX_DATAGRAM];
49            loop {
50                match recv_half.recv_from(&mut buf).await {
51                    Ok((n, _)) => match recv_tx.try_send(buf[..n].to_vec()) {
52                        Ok(()) | Err(mpsc::error::TrySendError::Full(_)) => {}
53                        Err(mpsc::error::TrySendError::Closed(_)) => break,
54                    },
55                    Err(e) => {
56                        debug!("receive quic packet through outbound failed: {}", e);
57                        break;
58                    }
59                }
60            }
61        });
62        Self {
63            peer,
64            tx,
65            rx: Mutex::new(rx),
66            tasks: [sender.abort_handle(), receiver.abort_handle()],
67        }
68    }
69}
70
71impl Drop for DatagramSocket {
72    fn drop(&mut self) {
73        for task in &self.tasks {
74            task.abort();
75        }
76    }
77}
78
79impl fmt::Debug for DatagramSocket {
80    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
81        f.debug_struct("DatagramSocket")
82            .field("peer", &self.peer)
83            .finish()
84    }
85}

87/// Sending never waits: a full queue drops the packet.
88#[derive(Debug)]
89struct AlwaysWritable;
90
91impl quinn::UdpPoller for AlwaysWritable {
92    fn poll_writable(self: Pin<&mut Self>, _cx: &mut Context) -> Poll<io::Result<()>> {
93        Poll::Ready(Ok(()))
94    }
95}
96
97impl quinn::AsyncUdpSocket for DatagramSocket {
98    fn create_io_poller(self: Arc<Self>) -> Pin<Box<dyn quinn::UdpPoller>> {
99        Box::pin(AlwaysWritable)
100    }
101
102    fn try_send(&self, transmit: &quinn::udp::Transmit) -> io::Result<()> {
103        match self.tx.try_send(transmit.contents.to_vec()) {
104            Ok(()) | Err(mpsc::error::TrySendError::Full(_)) => Ok(()),
105            Err(mpsc::error::TrySendError::Closed(_)) => Err(io::Error::new(
106                io::ErrorKind::BrokenPipe,
107                "outbound datagram closed",
108            )),
109        }
110    }
111
112    fn poll_recv(
113        &self,
114        cx: &mut Context,
115        bufs: &mut [IoSliceMut<'_>],
116        meta: &mut [quinn::udp::RecvMeta],
117    ) -> Poll<io::Result<usize>> {
118        let (Some(buf), Some(meta)) = (bufs.first_mut(), meta.first_mut()) else {
119            return Poll::Ready(Ok(0));
120        };
121        let mut rx = self.rx.lock().unwrap_or_else(|e| e.into_inner());
122        match rx.poll_recv(cx) {
123            Poll::Ready(Some(packet)) => {
124                let n = packet.len().min(buf.len());
125                buf[..n].copy_from_slice(&packet[..n]);
126                meta.addr = self.peer;
127                meta.len = n;
128                meta.stride = n;
129                meta.ecn = None;
130                meta.dst_ip = None;
131                Poll::Ready(Ok(1))
132            }
133            Poll::Ready(None) => Poll::Ready(Err(io::Error::new(
134                io::ErrorKind::BrokenPipe,
135                "outbound datagram closed",
136            ))),
137            Poll::Pending => Poll::Pending,
138        }
139    }
140
141    fn local_addr(&self) -> io::Result<SocketAddr> {
142        Ok(match self.peer {
143            SocketAddr::V4(_) => SocketAddr::new(IpAddr::V4(Ipv4Addr::UNSPECIFIED), 0),
144            SocketAddr::V6(_) => SocketAddr::new(IpAddr::V6(Ipv6Addr::UNSPECIFIED), 0),
145        })
146    }
147}
