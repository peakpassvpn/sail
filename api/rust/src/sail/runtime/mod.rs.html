16/// Where the host keeps things, and what it does for the instance.
17#[derive(Debug, Clone, Default, PartialEq, Eq)]
18pub struct Host {
19    /// Data files (`geo.mmdb`, `site.dat`), certificates given by relative
20    /// path, and the cache live here. Defaults to the executable's directory.
21    pub data_dir: Option<PathBuf>,
22    /// Keeps state across restarts, such as selected outbounds, when set.
23    pub cache_dir: Option<PathBuf>,
24    /// Sends logs to the platform's system log rather than to standard
25    /// output; needs a platform.
26    pub log_to_system: bool,
27    /// How outbound sockets are kept out of the VPN (Android), when the
28    /// platform does not do it.
29    pub socket_protect: Option<crate::net::dial::SocketProtect>,
30    /// What the embedding host does for the instance.
31    pub platform: Option<PlatformRef>,
32}
33
34/// Everything an instance runs with that is not its configuration.
35#[derive(Debug, Clone, Default)]
36pub struct RuntimeEnv {
37    pub options: RuntimeOptions,
38    pub host: Host,
39}
40
41pub type SyncRuntimeEnv = Arc<RuntimeEnv>;
42
43impl RuntimeEnv {
44    /// The directory data files are looked up in.
45    pub fn data_dir(&self) -> PathBuf {
46        self.host.data_dir.clone().unwrap_or_else(|| {
47            std::env::current_exe()
48                .ok()
49                .and_then(|exe| exe.parent().map(Path::to_path_buf))
50                .unwrap_or_default()
51        })
52    }
53
54    /// `path` in the data directory, unless it is absolute.
55    pub fn data_path(&self, path: &str) -> String {
56        let p = Path::new(path);
57        if p.is_absolute() {
58            return path.to_string();
59        }
60        self.data_dir().join(p).to_string_lossy().to_string()
61    }
62
63    /// A certificate or key given inline or as a path; a relative path is
64    /// looked up in the data directory.
65    ///
66    /// The two halves of a keypair are configured the same way and have to
67    /// be read the same way. They were not: a certificate was recognised
68    /// inline and a key never was, so an inline key became a path under the
69    /// data directory made of PEM, and what the operator saw was "no private
70    /// keys found" about a key that was right there in the configuration.
71    pub fn certificate(&self, value: &str) -> String {
72        if value.contains("-----BEGIN") {
73            return value.to_string();
74        }
75        self.data_path(value)
76    }
77}

79/// How a host describes the tuning and host options of an instance in one
80/// piece, for hosts that pass them as text (FFI):
81///
82/// ```json
83/// { "profile": "mobile", "set": ["relay.buffer_size=32"],
84///   "data_dir": "/var/lib/sail", "cache_dir": "/var/cache/sail",
85///   "log_to_system": true, "socket_protect": "/data/protect.sock" }
86/// ```
87#[derive(Deserialize, Debug, Default)]
88#[serde(deny_unknown_fields)]
89pub struct StartSettings {
90    #[serde(default)]
91    pub profile: Option<String>,
92    #[serde(default)]
93    pub set: Vec<String>,
94    #[serde(default)]
95    pub data_dir: Option<PathBuf>,
96    #[serde(default)]
97    pub cache_dir: Option<PathBuf>,
98    /// Whether to log to the platform's system log; the host decides when
99    /// unset.
100    #[serde(default)]
101    pub log_to_system: Option<bool>,
102    /// A Unix socket path, or an `address:port` to connect to over TCP.
103    #[serde(default)]
104    pub socket_protect: Option<String>,
105}
106
107impl StartSettings {
108    pub fn from_json(json: &str) -> Result<Self> {
109        let de = &mut serde_json::Deserializer::from_str(json);
110        serde_path_to_error::deserialize(de)
111            .map_err(|e| anyhow!("start settings: {}: {}", e.path(), e.inner()))
112    }
113
114    /// The tuning and host these settings describe.
115    pub fn resolve(self) -> Result<(RuntimeOptions, Host)> {
116        let profile = match &self.profile {
117            Some(p) => p.parse()?,
118            None => Profile::default(),
119        };
120        let mut options = RuntimeOptions::profile(profile);
121        options.set_all(self.set.iter().map(String::as_str))?;
122        let socket_protect = self.socket_protect.map(|p| match p.parse() {
123            Ok(addr) => crate::net::dial::SocketProtect::Tcp(addr),
124            Err(_) => crate::net::dial::SocketProtect::Unix(p),
125        });
126        Ok((
127            options,
128            Host {
129                data_dir: self.data_dir,
130                cache_dir: self.cache_dir,
131                log_to_system: self.log_to_system.unwrap_or(false),
132                socket_protect,
133                platform: None,
134            },
135        ))
136    }
137}

139#[cfg(test)]
140mod tests {
141    use super::*;
142
143    const INLINE_KEY: &str = "-----BEGIN PRIVATE KEY-----\nMIGH\n-----END PRIVATE KEY-----\n";
144
145    fn env() -> RuntimeEnv {
146        RuntimeEnv {
147            host: Host {
148                data_dir: Some(PathBuf::from("/data")),
149                ..Default::default()
150            },
151            ..Default::default()
152        }
153    }
154
155    #[test]
156    fn an_inline_key_is_not_mistaken_for_a_path() {
157        assert_eq!(env().certificate(INLINE_KEY), INLINE_KEY);
158    }
159
160    /// What counts as absolute is the platform's business, and the test has
161    /// to ask the same question the code does.
162    #[test]
163    fn an_absolute_path_is_left_alone() {
164        let absolute = if cfg!(windows) {
165            r"C:\sail\cert.pem"
166        } else {
167            "/etc/sail/cert.pem"
168        };
169        assert_eq!(env().certificate(absolute), absolute);
170    }
171
172    #[test]
173    fn a_relative_path_is_in_the_data_directory() {
174        let resolved = env().certificate("cert.pem");
175        assert_eq!(
176            PathBuf::from(resolved),
177            PathBuf::from("/data").join("cert.pem")
178        );
179    }
180
181    #[test]
182    fn start_settings_resolve_to_tuning_and_host() {
183        let (options, host) = StartSettings::from_json(
184            r#"{ "profile": "router", "set": ["relay.buffer_size=2"],
185                 "data_dir": "/d", "socket_protect": "127.0.0.1:9000" }"#,
186        )
187        .unwrap()
188        .resolve()
189        .unwrap();
190        assert_eq!(options.relay.buffer_size, 2);
191        assert_eq!(
192            options.relay.buffer_max_size,
193            RuntimeOptions::profile(Profile::Router)
194                .relay
195                .buffer_max_size
196        );
197        assert_eq!(host.data_dir, Some(PathBuf::from("/d")));
198        assert_eq!(
199            host.socket_protect,
200            Some(crate::net::dial::SocketProtect::Tcp(
201                "127.0.0.1:9000".parse().unwrap()
202            ))
203        );
204        let err = StartSettings::from_json(r#"{ "profile": "mobile", "sett": [] }"#).unwrap_err();
205        assert!(err.to_string().contains("sett"), "{}", err);
206    }
