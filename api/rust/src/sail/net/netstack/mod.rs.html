1//! The userspace TCP/IP stack as the rest of sail uses it: a runtime that
2//! drives `sail-netstack` over any [`sail_netstack::PacketIo`], and hands
3//! out the connections and datagrams it carries. The TUN inbound runs it
4//! over the device; an outbound that carries IP packets runs it over
5//! [`ChannelPacketIo`].
6// The stack serves the protocols that carry IP packets; built without any
7// of them, as `netstack` alone, it has no user.

9    not(feature = "inbound-tun"),
10    allow(
11        dead_code,
12        unused_imports,
13        reason = "no protocol that uses it is enabled"
14    )
15)]

18mod runtime;
19mod stream;
