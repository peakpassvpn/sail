1//! The client: mux connections made by an outbound, and the streams handed
2//! out over them.
3//!
4//! A mux connection is a connection of the outbound's whole stack -- its
5//! protocol over its transport and TLS, through its detour -- to the magic
6//! destination, which is why the outbound hands it over as a `Connector`
7//! around the stack instead of being one more layer in it.
8//!
9//! Streams go to the connection with the fewest, as in sing-mux, and a new
10//! connection is made when those are too busy: with `max_connections`,
11//! while there are fewer than that many and the least busy carries at
12//! least `min_streams`; with `max_streams`, when every connection carries
13//! that many. Hard limits stand behind both: `MAX_CONNECTIONS` connections,
14//! `session::MAX_STREAMS` streams on one.
15

39/// Connections one outbound keeps at most.
40pub const MAX_CONNECTIONS: usize = 32;
41/// How long making a mux connection may take.
42const CONNECT_TIMEOUT: Duration = Duration::from_secs(10);
43/// How often idle connections are looked for.
44const CHECK_INTERVAL: Duration = Duration::from_secs(30);
45/// How long a connection without streams is kept.
46const IDLE_TIMEOUT: Duration = Duration::from_secs(60);
47
48#[derive(Debug, Clone, PartialEq, Eq)]
49pub struct ClientOptions {
50    pub protocol: Protocol,
51    pub padding: bool,
52    pub max_connections: usize,
53    pub min_streams: usize,
54    pub max_streams: usize,
55}
56
57impl ClientOptions {
58    /// Checks the limits as sing-box does: `max_streams` excludes the
59    /// other two. With none set, up to 4 connections, a new one while the
60    /// least busy carries 4 streams or more.
61    pub fn new(
62        protocol: Protocol,
63        padding: bool,
64        max_connections: Option<usize>,
65        min_streams: Option<usize>,
66        max_streams: Option<usize>,
67    ) -> Result<Self, String> {
68        let max_streams = max_streams.filter(|n| *n > 0);
69        let max_connections = max_connections.filter(|n| *n > 0);
70        let min_streams = min_streams.filter(|n| *n > 0);
71        if max_streams.is_some() && (max_connections.is_some() || min_streams.is_some()) {
72            return Err(
73                "max_streams cannot be set with max_connections or min_streams".to_string(),
74            );
75        }
76        if max_connections.is_some_and(|n| n > MAX_CONNECTIONS) {
77            return Err(format!("max_connections cannot exceed {}", MAX_CONNECTIONS));
78        }
79        if max_streams.is_some_and(|n| n > super::session::MAX_STREAMS) {
80            return Err(format!(
81                "max_streams cannot exceed {}",
82                super::session::MAX_STREAMS
83            ));
84        }
85        let (max_connections, min_streams) = match max_streams {
86            Some(_) => (0, 0),
87            None => (max_connections.unwrap_or(4), min_streams.unwrap_or(4)),
88        };
89        Ok(ClientOptions {
90            protocol,
91            padding,
92            max_connections,
93            min_streams,
94            max_streams: max_streams.unwrap_or(0),
95        })
96    }
97}

99/// One mux connection.
100enum Conn {
101    Frames(FrameSession),
102    H2(H2Client),
103}
104
105impl Conn {
106    async fn open(&self) -> io::Result<AnyStream> {
107        match self {
108            Conn::Frames(session) => Ok(Box::new(session.open()?)),
109            Conn::H2(client) => Ok(Box::new(client.open().await?)),
110        }
111    }
112
113    fn num_streams(&self) -> usize {
114        match self {
115            Conn::Frames(session) => session.num_streams(),
116            Conn::H2(client) => client.num_streams(),
117        }
118    }
119
120    fn is_closed(&self) -> bool {
121        match self {
122            Conn::Frames(session) => session.is_closed(),
123            Conn::H2(client) => client.is_closed(),
124        }
125    }
126
127    fn can_take_new_request(&self) -> bool {
128        match self {
129            Conn::Frames(session) => session.num_streams() < super::session::MAX_STREAMS,
130            Conn::H2(client) => client.can_take_new_request(),
131        }
132    }
133
134    fn close(&self) {
135        match self {
136            Conn::Frames(session) => session.close(),
137            Conn::H2(client) => client.close(),
138        }
139    }
140}

142struct Entry {
143    conn: Arc<Conn>,
144    /// Since when it has carried no streams, as last checked.
145    idle_since: Option<Instant>,
146}
147
148pub struct Client {
149    connector: Connector,
150    options: ClientOptions,
151    /// Held while a connection is made, so that streams asking at once
152    /// share it rather than each making their own, as in sing-mux.
153    conns: tokio::sync::Mutex<Vec<Entry>>,
154    /// The idle check, until the first connection starts it: outbounds are
155    /// built outside a runtime.
156    cleanup: SyncMutex<Option<BoxFuture<'static, ()>>>,
157}
158
159impl Client {
160    /// The client, and the handle that stops its idle check.
161    pub fn new(connector: Connector, options: ClientOptions) -> (Arc<Client>, AbortHandle) {
162        let client = Arc::new(Client {
163            connector,
164            options,
165            conns: tokio::sync::Mutex::new(Vec::new()),
166            cleanup: SyncMutex::new(None),
167        });
168        let weak: Weak<Client> = Arc::downgrade(&client);
169        let (check, handle) = abortable(async move {
170            loop {
171                tokio::time::sleep(CHECK_INTERVAL).await;
172                let Some(client) = weak.upgrade() else {
173                    return;
174                };
175                client.check_idle();
176            }
177        });
178        if let Ok(mut cleanup) = client.cleanup.lock() {
179            *cleanup = Some(check.map(|_| ()).boxed());
180        }
181        (client, handle)
182    }
183
184    fn check_idle(&self) {
185        // Not while a connection is being made; the next check will do.
186        let Ok(mut conns) = self.conns.try_lock() else {
187            return;
188        };
189        let now = Instant::now();
190        conns.retain_mut(|entry| {
191            if entry.conn.is_closed() {
192                return false;
193            }
194            if entry.conn.num_streams() > 0 {
195                entry.idle_since = None;
196                return true;
197            }
198            match entry.idle_since {
199                None => {
200                    entry.idle_since = Some(now);
201                    true
202                }
203                Some(since) if now.duration_since(since) >= IDLE_TIMEOUT => {
204                    entry.conn.close();
205                    false
206                }
207                Some(_) => true,
208            }
209        });
210    }
211
212    /// A new stream, before its request.
213    async fn open_stream(&self, sess: &Session) -> io::Result<AnyStream> {
214        if let Some(check) = self.cleanup.lock().ok().and_then(|mut c| c.take()) {
215            tokio::spawn(check);
216        }
217        let mut last = io::Error::other("mux: no connection");
218        for _ in 0..2 {
219            let conn = match self.offer(sess).await {
220                Ok(conn) => conn,
221                Err(e) => {
222                    last = e;
223                    continue;
224                }
225            };
226            match conn.open().await {
227                Ok(stream) => return Ok(stream),
228                Err(e) => {
229                    debug!("mux open stream: {}", e);
230                    last = e;
231                }
232            }
233        }
234        Err(last)
235    }
236
237    /// The connection a new stream goes on, made if need be.
238    async fn offer(&self, sess: &Session) -> io::Result<Arc<Conn>> {
239        let mut conns = self.conns.lock().await;
240        conns.retain(|entry| {
241            if entry.conn.is_closed() {
242                entry.conn.close();
243                return false;
244            }
245            true
246        });
247        let least = conns
248            .iter()
249            .filter(|e| e.conn.can_take_new_request())
250            .min_by_key(|e| e.conn.num_streams())
251            .map(|e| e.conn.clone());
252        if let Some(conn) = &least {
253            if reuse(&self.options, conn.num_streams(), conns.len()) {
254                return Ok(conn.clone());
255            }
256        }
257        if conns.len() >= MAX_CONNECTIONS {
258            return least.ok_or_else(|| io::Error::other("mux: every connection is full"));
259        }
260        let conn = Arc::new(
261            tokio::time::timeout(CONNECT_TIMEOUT, self.connect(sess))
262                .await
263                .map_err(|_| io::Error::new(io::ErrorKind::TimedOut, "mux: connect timed out"))??,
264        );
265        conns.push(Entry {
266            conn: conn.clone(),
267            idle_since: None,
268        });
269        Ok(conn)
270    }
271
272    async fn connect(&self, sess: &Session) -> io::Result<Conn> {
273        let mut sess = sess.clone();
274        sess.network = Network::Tcp;
275        sess.destination = SocksAddr::Domain(MAGIC_DOMAIN.to_string(), MAGIC_PORT);
276        sess.sniffed = None;
277        let mut conn = self
278            .connector
279            .connect(&sess)
280            .instrument(tracing::Span::current())
281            .await?;
282        conn.write_all(&encode_request(self.options.protocol, self.options.padding))
283            .await?;
284        let conn: AnyStream = if self.options.padding {
285            Box::new(PaddingStream::new(conn))
286        } else {
287            conn
288        };
289        debug!("mux connection ({:?})", self.options.protocol);
290        Ok(match self.options.protocol {
291            Protocol::Smux => Conn::Frames(FrameSession::new(conn, Flavor::Smux, false).0),
292            Protocol::Yamux => Conn::Frames(FrameSession::new(conn, Flavor::Yamux, false).0),
293            Protocol::H2Mux => Conn::H2(H2Client::new(conn).await?),
294        })
295    }
296
297    /// Opens a stream and sends `request` on it.
298    async fn request(&self, sess: &Session, request: StreamRequest) -> io::Result<AnyStream> {
299        let mut stream = self.open_stream(sess).await?;
300        let mut buf = BytesMut::new();
301        request.encode(&mut buf);
302        stream.write_all(&buf).await?;
303        Ok(stream)
304    }
305}
306
307/// Whether a stream goes on the least busy connection, which carries
308/// `streams`, rather than a new one, with `conns` connections open.
309fn reuse(o: &ClientOptions, streams: usize, conns: usize) -> bool {
310    if streams == 0 {
311        return true;
312    }
313    if o.max_connections > 0 {
314        conns >= o.max_connections || streams < o.min_streams
315    } else {
316        o.max_streams > 0 && streams < o.max_streams
317    }
318}
319
320impl Drop for Client {
321    fn drop(&mut self) {
322        if let Ok(conns) = self.conns.try_lock() {
323            for entry in conns.iter() {
324                entry.conn.close();
325            }
326        }
327    }
328}

335#[async_trait]
336impl OutboundStreamHandler for StreamHandler {
337    fn connect_addr(&self) -> OutboundConnect {
338        OutboundConnect::Unknown
339    }
340
341    async fn handle<'a>(
342        &'a self,
343        sess: &'a Session,
344        _lhs: Option<&mut AnyStream>,
345        _stream: Option<AnyStream>,
346    ) -> io::Result<AnyStream> {
347        let stream = self
348            .client
349            .request(sess, StreamRequest::Tcp(sess.destination.clone()))
350            .await?;
351        Ok(Box::new(ClientStream {
352            inner: stream,
353            status: [0],
354            status_read: false,
355        }))
356    }
357}

387/// A TCP stream on the client, which reads the server's status before its
388/// first data.
389struct ClientStream {
390    inner: AnyStream,
391    status: [u8; 1],
392    status_read: bool,
393}
394
395impl AsyncRead for ClientStream {
396    fn poll_read(
397        self: Pin<&mut Self>,
398        cx: &mut Context<'_>,
399        buf: &mut ReadBuf<'_>,
400    ) -> Poll<io::Result<()>> {
401        let me = self.get_mut();
402        if !me.status_read {
403            let mut status = ReadBuf::new(&mut me.status);
404            ready!(Pin::new(&mut me.inner).poll_read(cx, &mut status))?;
405            if status.filled().is_empty() {
406                return Poll::Ready(Err(io::Error::new(
407                    io::ErrorKind::UnexpectedEof,
408                    "mux: stream closed before its status",
409                )));
410            }
411            if me.status[0] != STATUS_SUCCESS {
412                return Poll::Ready(Err(io::Error::other("mux: remote error")));
413            }
414            me.status_read = true;
415        }
416        Pin::new(&mut me.inner).poll_read(cx, buf)
417    }
418}
419
420impl AsyncWrite for ClientStream {
421    fn poll_write(
422        self: Pin<&mut Self>,
423        cx: &mut Context<'_>,
424        buf: &[u8],
425    ) -> Poll<io::Result<usize>> {
426        Pin::new(&mut self.get_mut().inner).poll_write(cx, buf)
427    }
428
429    fn poll_flush(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<io::Result<()>> {
430        Pin::new(&mut self.get_mut().inner).poll_flush(cx)
431    }
432
433    fn poll_shutdown(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<io::Result<()>> {
434        Pin::new(&mut self.get_mut().inner).poll_shutdown(cx)
435    }
436}

438/// `layered`, the outbound's whole stack, with its TCP and UDP carried
439/// over mux connections it makes.
440pub fn outbound(
441    tag: &str,
442    layered: AnyOutboundHandler,
443    dns_client: crate::app::SyncDnsClient,
444    options: ClientOptions,
445    abort_handles: &mut Vec<AbortHandle>,
446) -> AnyOutboundHandler {
447    let (client, cleanup) = Client::new(Connector::around(layered, dns_client), options);
448    abort_handles.push(cleanup);
449    crate::adapter::outbound::HandlerBuilder::default()
450        .tag(tag.to_owned())
451        .stream_handler(Arc::new(StreamHandler {
452            client: client.clone(),
453        }))
454        .datagram_handler(Arc::new(DatagramHandler { client }))
455        .build()
456}

458#[cfg(test)]
459mod tests {
460    use super::*;
461
462    fn options(
463        max_connections: Option<usize>,
464        min_streams: Option<usize>,
465        max_streams: Option<usize>,
466    ) -> Result<ClientOptions, String> {
467        ClientOptions::new(
468            Protocol::Smux,
469            false,
470            max_connections,
471            min_streams,
472            max_streams,
473        )
474    }
475
476    #[test]
477    fn limits_are_checked_as_sing_box_does() {
478        let o = options(None, None, None).unwrap();
479        assert_eq!((o.max_connections, o.min_streams, o.max_streams), (4, 4, 0));
480        let o = options(None, None, Some(8)).unwrap();
481        assert_eq!((o.max_connections, o.min_streams, o.max_streams), (0, 0, 8));
482        assert!(options(Some(2), None, Some(8)).is_err());
483        assert!(options(None, Some(2), Some(8)).is_err());
484        assert!(options(Some(MAX_CONNECTIONS + 1), None, None).is_err());
485        assert!(options(None, None, Some(100_000)).is_err());
486    }
487
488    #[test]
489    fn a_new_connection_only_when_the_least_busy_is_busy_enough() {
490        let o = options(Some(2), Some(3), None).unwrap();
491        assert!(reuse(&o, 0, 1));
492        assert!(reuse(&o, 2, 1));
493        assert!(!reuse(&o, 3, 1));
494        assert!(reuse(&o, 50, 2));
495        let o = options(None, None, Some(2)).unwrap();
496        assert!(reuse(&o, 1, 10));
497        assert!(!reuse(&o, 2, 1));
498    }
