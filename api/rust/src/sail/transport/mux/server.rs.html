19enum Inner {
20    /// The session is kept for as long as streams may come.
21    Frames {
22        _session: FrameSession,
23        accept: mpsc::Receiver<MuxStream>,
24    },
25    H2(AbortHandle, mpsc::Receiver<H2Stream>),
26}
27
28/// A mux connection being served. Dropping it closes the connection and
29/// every stream on it.
30pub struct Server {
31    inner: Inner,
32}
33
34impl Server {
35    /// Reads the request that opens `conn`, and starts serving it.
36    pub async fn start(mut conn: AnyStream) -> io::Result<Server> {
37        let (protocol, padding) = read_request(&mut conn).await?;
38        let conn: AnyStream = if padding {
39            Box::new(PaddingStream::new(conn))
40        } else {
41            conn
42        };
43        let inner = match protocol {
44            Protocol::Smux | Protocol::Yamux => {
45                let flavor = if protocol == Protocol::Smux {
46                    Flavor::Smux
47                } else {
48                    Flavor::Yamux
49                };
50                let (session, accept) = FrameSession::new(conn, flavor, true);
51                let accept = accept.ok_or_else(|| io::Error::other("mux: not a server"))?;
52                Inner::Frames {
53                    _session: session,
54                    accept,
55                }
56            }
57            Protocol::H2Mux => {
58                let (handle, accept) = h2mux::serve(conn);
59                Inner::H2(handle, accept)
60            }
61        };
62        Ok(Server { inner })
63    }
64
65    /// The next stream the client opens; `None` once the connection is
66    /// done.
67    pub async fn accept(&mut self) -> Option<AnyStream> {
68        match &mut self.inner {
69            Inner::Frames { accept, .. } => accept.recv().await.map(|s| Box::new(s) as AnyStream),
70            Inner::H2(_, accept) => accept.recv().await.map(|s| Box::new(s) as AnyStream),
71        }
72    }
73}
74
75impl Drop for Server {
76    fn drop(&mut self) {
77        if let Inner::H2(handle, _) = &self.inner {
78            handle.abort();
79        }
80    }
81}

83/// Reads what a stream asks for. A TCP stream comes back ready to relay;
84/// a UDP one is for `packet::ServerDatagram`.
85pub async fn read_stream(mut stream: AnyStream) -> io::Result<(StreamRequest, AnyStream)> {
86    let request = StreamRequest::read(&mut stream).await?;
87    let stream: AnyStream = match request {
88        StreamRequest::Tcp(_) => Box::new(ServerStream {
89            inner: stream,
90            status_written: false,
91        }),
92        _ => stream,
93    };
94    Ok((request, stream))
95}

97/// A TCP stream on the server, which sends its status before its first
98/// data, as sing-mux does: had the destination failed, the stream would
99/// have been closed before.
100struct ServerStream {
101    inner: AnyStream,
102    status_written: bool,
103}
104
105impl ServerStream {
106    fn poll_status(&mut self, cx: &mut Context<'_>) -> Poll<io::Result<()>> {
107        while !self.status_written {
108            let n = ready!(Pin::new(&mut self.inner).poll_write(cx, &[STATUS_SUCCESS]))?;
109            if n == 0 {
110                return Poll::Ready(Err(io::ErrorKind::WriteZero.into()));
111            }
112            self.status_written = true;
113        }
114        Poll::Ready(Ok(()))
115    }
116}
117
118impl AsyncRead for ServerStream {
119    fn poll_read(
120        self: Pin<&mut Self>,
121        cx: &mut Context<'_>,
122        buf: &mut ReadBuf<'_>,
123    ) -> Poll<io::Result<()>> {
124        Pin::new(&mut self.get_mut().inner).poll_read(cx, buf)
125    }
126}
127
128impl AsyncWrite for ServerStream {
129    fn poll_write(
130        self: Pin<&mut Self>,
131        cx: &mut Context<'_>,
132        buf: &[u8],
133    ) -> Poll<io::Result<usize>> {
134        let me = self.get_mut();
135        ready!(me.poll_status(cx))?;
136        Pin::new(&mut me.inner).poll_write(cx, buf)
137    }
138
139    fn poll_flush(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<io::Result<()>> {
140        Pin::new(&mut self.get_mut().inner).poll_flush(cx)
141    }
142
143    fn poll_shutdown(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<io::Result<()>> {
144        let me = self.get_mut();
145        // A client reads the status before it can see the end.
146        ready!(me.poll_status(cx))?;
147        Pin::new(&mut me.inner).poll_shutdown(cx)
148    }
149}
