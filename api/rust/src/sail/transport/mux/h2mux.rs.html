1//! h2mux: every stream is an HTTP/2 CONNECT request, its body one way and
2//! the response's the other, as sing-mux carries streams over HTTP/2.
3//!
4//! HTTP/2 flow control bounds what waits unread: `STREAM_WINDOW` a stream,
5//! `CONNECTION_WINDOW` a connection.
6

24const STREAM_WINDOW: u32 = 1 << 20;
25const CONNECTION_WINDOW: u32 = 4 << 20;
26/// Written into one stream's send buffer at once.
27const MAX_WRITE: usize = 32 << 10;
28/// Streams accepted and not yet taken by the server.
29const ACCEPT_QUEUE: usize = 256;
30
31fn h2_error(e: h2::Error) -> io::Error {
32    if e.is_io() {
33        return e.into_io().unwrap_or_else(|| io::Error::other("h2mux: io"));
34    }
35    io::Error::other(format!("h2mux: {}", e))
36}
37
38/// Counts the streams open on a connection.
39struct Active(Arc<AtomicUsize>);
40
41impl Active {
42    fn new(count: &Arc<AtomicUsize>) -> Self {
43        count.fetch_add(1, Ordering::Relaxed);
44        Active(count.clone())
45    }
46}
47
48impl Drop for Active {
49    fn drop(&mut self) {
50        self.0.fetch_sub(1, Ordering::Relaxed);
51    }
52}

54pub struct H2Client {
55    send: h2::client::SendRequest<Bytes>,
56    closed: Arc<AtomicBool>,
57    active: Arc<AtomicUsize>,
58    task: AbortHandle,
59}
60
61impl H2Client {
62    pub async fn new<S>(conn: S) -> io::Result<Self>
63    where
64        S: AsyncRead + AsyncWrite + Unpin + Send + 'static,
65    {
66        let (send, connection) = h2::client::Builder::new()
67            .initial_window_size(STREAM_WINDOW)
68            .initial_connection_window_size(CONNECTION_WINDOW)
69            .max_concurrent_streams(0)
70            .handshake(conn)
71            .await
72            .map_err(h2_error)?;
73        let closed = Arc::new(AtomicBool::new(false));
74        let (driver, task) = abortable({
75            let closed = closed.clone();
76            connection.map(move |result| {
77                if let Err(e) = result {
78                    debug!("h2mux connection: {}", e);
79                }
80                closed.store(true, Ordering::Relaxed);
81            })
82        });
83        tokio::spawn(driver);
84        Ok(H2Client {
85            send,
86            closed,
87            active: Arc::new(AtomicUsize::new(0)),
88            task,
89        })
90    }
91
92    pub async fn open(&self) -> io::Result<H2Stream> {
93        let mut send = self.send.clone().ready().await.map_err(h2_error)?;
94        let request = Request::builder()
95            .method(Method::CONNECT)
96            .uri(Uri::from_static("localhost:443"))
97            .body(())
98            .map_err(io::Error::other)?;
99        let (response, stream) = send.send_request(request, false).map_err(h2_error)?;
100        Ok(H2Stream {
101            response: Some(response),
102            recv: None,
103            send: stream,
104            received: Bytes::new(),
105            ended: false,
106            _active: Active::new(&self.active),
107        })
108    }
109
110    pub fn num_streams(&self) -> usize {
111        self.active.load(Ordering::Relaxed)
112    }
113
114    pub fn is_closed(&self) -> bool {
115        self.closed.load(Ordering::Relaxed)
116    }
117
118    /// Whether the server takes another stream now.
119    pub fn can_take_new_request(&self) -> bool {
120        self.num_streams() < self.send.current_max_send_streams()
121    }
122
123    pub fn close(&self) {
124        self.closed.store(true, Ordering::Relaxed);
125        self.task.abort();
126    }
127}
128
129impl Drop for H2Client {
130    fn drop(&mut self) {
131        self.task.abort();
132    }
133}

135/// Serves HTTP/2 on `conn`: the streams come out of the receiver. The
136/// handle stops serving.
137pub fn serve<S>(conn: S) -> (AbortHandle, mpsc::Receiver<H2Stream>)
138where
139    S: AsyncRead + AsyncWrite + Unpin + Send + 'static,
140{
141    let (tx, rx) = mpsc::channel(ACCEPT_QUEUE);
142    let (task, handle) = abortable(async move {
143        let mut connection = match h2::server::Builder::new()
144            .initial_window_size(STREAM_WINDOW)
145            .initial_connection_window_size(CONNECTION_WINDOW)
146            .max_concurrent_streams(MAX_STREAMS as u32)
147            .handshake::<_, Bytes>(conn)
148            .await
149        {
150            Ok(c) => c,
151            Err(e) => {
152                debug!("h2mux handshake: {}", e);
153                return;
154            }
155        };
156        let active = Arc::new(AtomicUsize::new(0));
157        // Accepting drives the connection too, so it never waits on the
158        // receiver: a stream that finds the queue full is refused.
159        while let Some(accepted) = connection.accept().await {
160            let (request, mut respond) = match accepted {
161                Ok(v) => v,
162                Err(e) => {
163                    debug!("h2mux accept: {}", e);
164                    break;
165                }
166            };
167            let status = if request.method() == Method::CONNECT {
168                StatusCode::OK
169            } else {
170                StatusCode::NOT_FOUND
171            };
172            let response = Response::builder()
173                .status(status)
174                .body(())
175                .unwrap_or_default();
176            let send = match respond.send_response(response, status != StatusCode::OK) {
177                Ok(send) => send,
178                Err(e) => {
179                    debug!("h2mux respond: {}", e);
180                    continue;
181                }
182            };
183            if status != StatusCode::OK {
184                continue;
185            }
186            let stream = H2Stream {
187                response: None,
188                recv: Some(request.into_body()),
189                send,
190                received: Bytes::new(),
191                ended: false,
192                _active: Active::new(&active),
193            };
194            if tx.try_send(stream).is_err() {
195                debug!("h2mux: too many streams waiting, one refused");
196            }
197        }
198    });
199    tokio::spawn(task);
200    (handle, rx)
201}

203/// One stream: a CONNECT request, from either end.
204pub struct H2Stream {
205    /// On a client, until the response comes.
206    response: Option<ResponseFuture>,
207    recv: Option<RecvStream>,
208    send: SendStream<Bytes>,
209    /// Received and not yet read.
210    received: Bytes,
211    ended: bool,
212    _active: Active,
213}
214
215impl AsyncRead for H2Stream {
216    fn poll_read(
217        self: Pin<&mut Self>,
218        cx: &mut Context<'_>,
219        buf: &mut ReadBuf<'_>,
220    ) -> Poll<io::Result<()>> {
221        let me = self.get_mut();
222        loop {
223            if !me.received.is_empty() {
224                let n = me.received.len().min(buf.remaining());
225                buf.put_slice(&me.received[..n]);
226                me.received.advance(n);
227                return Poll::Ready(Ok(()));
228            }
229            if let Some(response) = me.response.as_mut() {
230                let response = ready!(response.poll_unpin(cx)).map_err(h2_error)?;
231                me.response = None;
232                if response.status() != StatusCode::OK {
233                    return Poll::Ready(Err(io::Error::other(format!(
234                        "h2mux: unexpected status {}",
235                        response.status()
236                    ))));
237                }
238                me.recv = Some(response.into_body());
239            }
240            let Some(recv) = me.recv.as_mut() else {
241                return Poll::Ready(Ok(()));
242            };
243            match ready!(recv.poll_data(cx)) {
244                Some(Ok(data)) => {
245                    let _ = recv.flow_control().release_capacity(data.len());
246                    me.received = data;
247                }
248                Some(Err(e)) if e.reason() == Some(h2::Reason::NO_ERROR) => {
249                    return Poll::Ready(Ok(()))
250                }
251                Some(Err(e)) => return Poll::Ready(Err(h2_error(e))),
252                None => return Poll::Ready(Ok(())),
253            }
254        }
255    }
256}

258impl AsyncWrite for H2Stream {
259    fn poll_write(
260        self: Pin<&mut Self>,
261        cx: &mut Context<'_>,
262        buf: &[u8],
263    ) -> Poll<io::Result<usize>> {
264        let me = self.get_mut();
265        if me.ended {
266            return Poll::Ready(Err(io::ErrorKind::BrokenPipe.into()));
267        }
268        if buf.is_empty() {
269            return Poll::Ready(Ok(0));
270        }
271        me.send.reserve_capacity(buf.len().min(MAX_WRITE));
272        match ready!(me.send.poll_capacity(cx)) {
273            Some(Ok(n)) => {
274                let n = n.min(buf.len());
275                me.send
276                    .send_data(Bytes::copy_from_slice(&buf[..n]), false)
277                    .map_err(h2_error)?;
278                Poll::Ready(Ok(n))
279            }
280            Some(Err(e)) => Poll::Ready(Err(h2_error(e))),
281            None => Poll::Ready(Err(io::ErrorKind::BrokenPipe.into())),
282        }
283    }
284
285    fn poll_flush(self: Pin<&mut Self>, _cx: &mut Context<'_>) -> Poll<io::Result<()>> {
286        Poll::Ready(Ok(()))
287    }
288
289    fn poll_shutdown(self: Pin<&mut Self>, _cx: &mut Context<'_>) -> Poll<io::Result<()>> {
290        let me = self.get_mut();
291        if !me.ended {
292            me.ended = true;
293            me.send.reserve_capacity(0);
294            me.send.send_data(Bytes::new(), true).map_err(h2_error)?;
295        }
296        Poll::Ready(Ok(()))
297    }
