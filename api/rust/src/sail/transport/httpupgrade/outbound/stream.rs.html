11pub struct Handler {
12    path: String,
13    /// `Host`; the server's address when not set.
14    host: Option<String>,
15    headers: Vec<(HeaderName, HeaderValue)>,
16}
17
18impl Handler {
19    /// `headers` are checked here, so that a mistake in them is a
20    /// configuration error.
21    pub fn new(
22        host: Option<String>,
23        path: String,
24        headers: &HashMap<String, String>,
25    ) -> anyhow::Result<Self> {
26        if !path.starts_with('/') {
27            return Err(anyhow::anyhow!("path: must start with /"));
28        }
29        // `host` is where sing-box sets it; a `Host` among the headers is
30        // one more place, and two that disagree are a mistake.
31        let header_host = headers
32            .iter()
33            .find(|(name, _)| name.eq_ignore_ascii_case("host"))
34            .map(|(_, host)| host.clone());
35        let host = match (host, header_host) {
36            (Some(a), Some(b)) if a != b => {
37                return Err(anyhow::anyhow!("host: differs from the Host header"))
38            }
39            (a, b) => a.or(b),
40        };
41        if let Some(host) = &host {
42            HeaderValue::from_str(host).map_err(|_| anyhow::anyhow!("host: invalid"))?;
43        }
44        let mut headers = upgrade::config_headers(headers)?;
45        if let Some((name, _)) = headers
46            .iter()
47            .find(|(name, _)| [http::header::CONNECTION, http::header::UPGRADE].contains(name))
48        {
49            return Err(anyhow::anyhow!("headers: {} is set by the transport", name));
50        }
51        headers.sort_by(|a, b| a.0.as_str().cmp(b.0.as_str()));
52        Ok(Handler {
53            path,
54            host,
55            headers,
56        })
57    }
58}

60#[async_trait]
61impl OutboundStreamHandler for Handler {
62    fn connect_addr(&self) -> OutboundConnect {
63        OutboundConnect::Next
64    }
65
66    async fn handle<'a>(
67        &'a self,
68        sess: &'a Session,
69        _lhs: Option<&mut AnyStream>,
70        stream: Option<AnyStream>,
71    ) -> io::Result<AnyStream> {
72        tracing::trace!("handling outbound stream");
73        let mut stream = stream.ok_or_else(|| io::Error::other("invalid input"))?;
74        let host = self.host.clone().unwrap_or_else(|| sess.destination.host());
75        stream
76            .write_all(&upgrade::upgrade_request(&self.path, &host, &self.headers))
77            .await?;
78        stream.flush().await?;
79
80        let (head, rest) = upgrade::read_head(&mut stream).await?;
81        let resp = upgrade::parse_response(&head)?;
82        let failed =
83            |why: String| io::Error::other(format!("connect httpupgrade {} failed: {}", host, why));
84        if resp.status != 101 {
85            return Err(failed(format!("server answered {}", resp.status)));
86        }
87        if !upgrade::upgrades_to_websocket(&resp.headers) {
88            return Err(failed("not upgraded".to_string()));
89        }
90        // Whatever the server sent after its head is the stream's already.
91        Ok(Box::new(Prefixed::new(rest, stream)))
92    }
93}

95#[cfg(test)]
96mod tests {
97    use super::*;
98
99    #[test]
100    fn test_config() {
101        let none = HashMap::new();
102        assert!(Handler::new(None, "/".to_string(), &none).is_ok());
103        assert!(Handler::new(None, "up".to_string(), &none).is_err());
104        let headers = HashMap::from([("Host".to_string(), "a.example".to_string())]);
105        let handler = Handler::new(None, "/".to_string(), &headers).unwrap();
106        assert_eq!(handler.host.as_deref(), Some("a.example"));
107        assert!(handler.headers.is_empty());
108        assert!(Handler::new(Some("b.example".to_string()), "/".to_string(), &headers).is_err());
109        let headers = HashMap::from([("Upgrade".to_string(), "h2c".to_string())]);
110        assert!(Handler::new(None, "/".to_string(), &headers).is_err());
111        let headers = HashMap::from([("X-Bad".to_string(), "a\r\nb".to_string())]);
112        assert!(Handler::new(None, "/".to_string(), &headers).is_err());
113    }
