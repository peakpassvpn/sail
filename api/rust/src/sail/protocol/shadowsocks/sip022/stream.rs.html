1//! The Shadowsocks 2022 TCP stream.
2//!
3//! A request is: salt, identity headers when the server has users, the
4//! sealed fixed-length header (type, timestamp, length of the next chunk),
5//! the sealed variable-length header (address, padding, initial payload),
6//! then length/payload chunk pairs. A response is: salt, the sealed
7//! fixed-length header (type, timestamp, request salt, length of the first
8//! payload chunk), that chunk, then length/payload chunk pairs. The request
9//! and response header exchange is done by [`connect`] and [`accept`]; the
10//! stream then carries chunks both ways, and writes the response header
11//! itself on the server's first write.
12

31/// Ciphertext requested from the inner stream per read, so one syscall can
32/// carry several chunks.
33const READ_AHEAD: usize = 64 * 1024;
34/// Plaintext sealed per write call.
35const MAX_WRITE: usize = 64 * 1024;
36/// SIP022 raises the chunk payload limit to the full u16 range.
37const MAX_CHUNK: usize = 0xffff;
38/// Type, timestamp and length of the request's fixed-length header.
39const REQUEST_FIXED_LEN: usize = 1 + 8 + 2;
40
41enum ReadState {
42    /// Client only: the response salt.
43    Salt,
44    /// Client only: the response's fixed-length header.
45    ResponseHeader,
46    Length,
47    Data(usize),
48}
49
50enum WriteState {
51    /// Server only: the response header goes out with the first payload.
52    ResponseHeader,
53    Ready,
54    /// Plaintext bytes to report once `write_buf` is flushed.
55    Pending(usize),
56}
57
58pub struct Ss2022Stream<T> {
59    inner: T,
60    method: Method,
61    /// The key the response direction's subkey comes from: the user's.
62    psk: Vec<u8>,
63    /// The client checks the response echoes it, the server echoes it.
64    request_salt: Vec<u8>,
65    enc: Option<ChunkAead>,
66    dec: Option<ChunkAead>,
67    read_buf: BytesMut,
68    plain: BytesMut,
69    write_buf: Vec<u8>,
70    write_pos: usize,
71    read_state: ReadState,
72    write_state: WriteState,
73}

75fn early_eof() -> io::Error {
76    io::Error::new(io::ErrorKind::UnexpectedEof, "early eof")
77}
78
79fn bad(msg: &str) -> io::Error {
80    io::Error::new(io::ErrorKind::InvalidData, msg.to_string())
81}

83/// Appends one sealed chunk, its length then its payload.
84fn seal_chunk(out: &mut Vec<u8>, enc: &mut ChunkAead, data: &[u8]) -> io::Result<()> {
85    let start = out.len();
86    out.extend_from_slice(&(data.len() as u16).to_be_bytes());
87    enc.seal(out, start)?;
88    let start = out.len();
89    out.extend_from_slice(data);
90    enc.seal(out, start)
91}

93fn random_salt(len: usize) -> Vec<u8> {
94    let mut salt = vec![0u8; len];
95    rand::thread_rng().fill_bytes(&mut salt);
96    salt
97}

99/// Opens a client stream: writes the request header to `inner`, with as
100/// much of `payload` as fits in it and the rest as chunks.
101///
102/// `psks` is the key chain, the server's first and the user's last; every
103/// key but the last makes an identity header for the next.
104pub async fn connect<T>(
105    mut inner: T,
106    method: Method,
107    psks: &[Vec<u8>],
108    destination: &SocksAddr,
109    payload: &[u8],
110) -> io::Result<Ss2022Stream<T>>
111where
112    T: AsyncWrite + Unpin,
113{
114    let user_psk = psks.last().ok_or_else(|| io::Error::other("missing PSK"))?;
115    let salt = random_salt(method.key_len());
116    let mut out = Vec::with_capacity(1024 + payload.len());
117    out.extend_from_slice(&salt);
118    for pair in psks.windows(2) {
119        let aes = AesBlock::new(&identity_subkey(&pair[0], &salt)).map_err(|_| crypto_err())?;
120        let mut eih = psk_hash(&pair[1]);
121        aes.encrypt(&mut eih)?;
122        out.extend_from_slice(&eih);
123    }
124    let mut enc = ChunkAead::new(method, &session_subkey(user_psk, &salt))?;
125
126    let padding = padding_len(payload.len());
127    let room = MAX_CHUNK - destination.size() - 2 - padding;
128    let (first, rest) = payload.split_at(min(payload.len(), room));
129    let variable_len = destination.size() + 2 + padding + first.len();
130
131    let start = out.len();
132    out.push(HEADER_TYPE_CLIENT);
133    out.extend_from_slice(&now_secs().to_be_bytes());
134    out.extend_from_slice(&(variable_len as u16).to_be_bytes());
135    enc.seal(&mut out, start)?;
136
137    let start = out.len();
138    let mut addr = BytesMut::new();
139    destination.write_buf(&mut addr, SocksAddrWireType::PortLast);
140    out.extend_from_slice(&addr);
141    out.extend_from_slice(&(padding as u16).to_be_bytes());
142    out.resize(out.len() + padding, 0);
143    out.extend_from_slice(first);
144    enc.seal(&mut out, start)?;
145
146    for data in rest.chunks(MAX_CHUNK) {
147        seal_chunk(&mut out, &mut enc, data)?;
148    }
149    inner.write_all(&out).await?;
150
151    Ok(Ss2022Stream {
152        inner,
153        method,
154        psk: user_psk.clone(),
155        request_salt: salt,
156        enc: Some(enc),
157        dec: None,
158        read_buf: BytesMut::new(),
159        plain: BytesMut::new(),
160        write_buf: Vec::new(),
161        write_pos: 0,
162        read_state: ReadState::Salt,
163        write_state: WriteState::Ready,
164    })
165}

179    pub stream: Ss2022Stream<T>,
180    pub destination: SocksAddr,
181    pub user: Option<Arc<str>>,

184/// Reads and checks a request header from `inner`.
185pub async fn accept<T>(mut inner: T, config: &ServerConfig) -> io::Result<Accepted<T>>
186where
187    T: AsyncRead + Unpin,
188{
189    let method = config.method;
190    let mut salt = vec![0u8; method.key_len()];
191    inner.read_exact(&mut salt).await?;
192
193    let (psk, user) = match &config.users {
194        None => (config.psk.clone(), None),
195        Some(users) => {
196            let mut eih = [0u8; EIH_LEN];
197            inner.read_exact(&mut eih).await?;
198            let aes =
199                AesBlock::new(&identity_subkey(&config.psk, &salt)).map_err(|_| crypto_err())?;
200            aes.decrypt(&mut eih)?;
201            let (_, user) = users.find(&eih).ok_or_else(|| bad("unknown user"))?;
202            (user.psk.clone(), user.name.clone())
203        }
204    };
205
206    let mut dec = ChunkAead::new(method, &session_subkey(&psk, &salt))?;
207    let mut fixed = [0u8; REQUEST_FIXED_LEN + TAG_LEN];
208    inner.read_exact(&mut fixed).await?;
209    dec.open(&mut fixed)?;
210    if fixed[0] != HEADER_TYPE_CLIENT {
211        return Err(bad("bad header type"));
212    }
213    let mut ts = [0u8; 8];
214    ts.copy_from_slice(&fixed[1..9]);
215    check_timestamp(u64::from_be_bytes(ts))?;
216    // Recorded only once the header authenticates, so that garbage cannot
217    // fill the pool.
218    config.salts.check_and_insert(&salt)?;
219    let variable_len = u16::from_be_bytes([fixed[9], fixed[10]]) as usize;
220
221    let mut variable = vec![0u8; variable_len + TAG_LEN];
222    inner.read_exact(&mut variable).await?;
223    let n = dec.open(&mut variable)?;
224    variable.truncate(n);
225    let destination = SocksAddr::try_from((&variable[..], SocksAddrWireType::PortLast))?;
226    let mut rest = &variable[destination.size()..];
227    if rest.len() < 2 {
228        return Err(bad("short request header"));
229    }
230    let padding = u16::from_be_bytes([rest[0], rest[1]]) as usize;
231    rest = &rest[2..];
232    if padding > MAX_PADDING || padding > rest.len() {
233        return Err(bad("bad padding"));
234    }
235    let payload = &rest[padding..];
236    if padding == 0 && payload.is_empty() {
237        return Err(bad("request without padding or payload"));
238    }
239
240    let stream = Ss2022Stream {
241        inner,
242        method,
243        psk,
244        request_salt: salt,
245        enc: None,
246        dec: Some(dec),
247        read_buf: BytesMut::new(),
248        plain: BytesMut::from(payload),
249        write_buf: Vec::new(),
250        write_pos: 0,
251        read_state: ReadState::Length,
252        write_state: WriteState::ResponseHeader,
253    };
254    Ok(Accepted {
255        stream,
256        destination,
257        user,
258    })
259}

261impl<T> Ss2022Stream<T>
262where
263    T: AsyncRead + Unpin,
264{
265    // Reads until `read_buf` holds at least `need` bytes.
266    fn poll_fill(&mut self, cx: &mut Context, need: usize) -> Poll<io::Result<()>> {
267        while self.read_buf.len() < need {
268            self.read_buf
269                .reserve((need - self.read_buf.len()).max(READ_AHEAD));
270            let mut buf = ReadBuf::uninit(self.read_buf.spare_capacity_mut());
271            match Pin::new(&mut self.inner).poll_read(cx, &mut buf) {
272                Poll::Ready(Ok(())) => (),
273                Poll::Ready(Err(e)) => return Poll::Ready(Err(e)),
274                Poll::Pending => {
275                    // Don't keep read-ahead memory around while idle.
276                    if self.read_buf.is_empty() {
277                        self.read_buf = BytesMut::new();
278                    }
279                    return Poll::Pending;
280                }
281            }
282            let n = buf.filled().len();
283            if n == 0 {
284                return Poll::Ready(Err(early_eof()));
285            }
286            // SAFETY: the inner reader initialized `n` bytes of spare capacity.
287            unsafe { self.read_buf.set_len(self.read_buf.len() + n) };
288        }
289        Poll::Ready(Ok(()))
290    }
291}

293impl<T> AsyncRead for Ss2022Stream<T>
294where
295    T: AsyncRead + Unpin,
296{
297    fn poll_read(
298        mut self: Pin<&mut Self>,
299        cx: &mut Context,
300        buf: &mut ReadBuf,
301    ) -> Poll<io::Result<()>> {
302        let me = &mut *self;
303        let start = buf.filled().len();
304        loop {
305            if !me.plain.is_empty() {
306                let n = min(buf.remaining(), me.plain.len());
307                buf.put_slice(&me.plain[..n]);
308                me.plain.advance(n);
309                if me.plain.is_empty() {
310                    me.plain = BytesMut::new();
311                }
312                if buf.remaining() == 0 {
313                    return Poll::Ready(Ok(()));
314                }
315            }
316
317            let key_len = me.method.key_len();
318            let need = match me.read_state {
319                ReadState::Salt => key_len,
320                ReadState::ResponseHeader => 1 + 8 + key_len + 2 + TAG_LEN,
321                ReadState::Length => 2 + TAG_LEN,
322                ReadState::Data(n) => n + TAG_LEN,
323            };
324            if me.read_buf.len() < need {
325                if buf.filled().len() > start {
326                    return Poll::Ready(Ok(()));
327                }
328                if let Err(e) = ready!(me.poll_fill(cx, need)) {
329                    // A clean end is one between chunks, or before the
330                    // server said anything.
331                    if e.kind() == io::ErrorKind::UnexpectedEof
332                        && me.read_buf.is_empty()
333                        && matches!(me.read_state, ReadState::Length | ReadState::Salt)
334                    {
335                        return Poll::Ready(Ok(()));
336                    }
337                    return Poll::Ready(Err(e));
338                }
339            }
340
341            let mut chunk = me.read_buf.split_to(need);
342            match me.read_state {
343                ReadState::Salt => {
344                    let key = session_subkey(&me.psk, &chunk);
345                    me.dec = Some(ChunkAead::new(me.method, &key)?);
346                    me.read_state = ReadState::ResponseHeader;
347                }
348                ReadState::ResponseHeader => {
349                    let dec = me.dec.as_mut().ok_or_else(crypto_err)?;
350                    dec.open(&mut chunk)?;
351                    if chunk[0] != HEADER_TYPE_SERVER {
352                        return Poll::Ready(Err(bad("bad header type")));
353                    }
354                    let mut ts = [0u8; 8];
355                    ts.copy_from_slice(&chunk[1..9]);
356                    check_timestamp(u64::from_be_bytes(ts))?;
357                    if chunk[9..9 + key_len] != me.request_salt[..] {
358                        return Poll::Ready(Err(bad("bad request salt")));
359                    }
360                    let len = u16::from_be_bytes([chunk[9 + key_len], chunk[10 + key_len]]);
361                    me.request_salt = Vec::new();
362                    me.read_state = if len == 0 {
363                        ReadState::Length
364                    } else {
365                        ReadState::Data(len as usize)
366                    };
367                }
368                ReadState::Length => {
369                    let dec = me.dec.as_mut().ok_or_else(crypto_err)?;
370                    dec.open(&mut chunk)?;
371                    let len = u16::from_be_bytes([chunk[0], chunk[1]]) as usize;
372                    me.read_state = ReadState::Data(len);
373                }
374                ReadState::Data(n) => {
375                    let dec = me.dec.as_mut().ok_or_else(crypto_err)?;
376                    dec.open(&mut chunk)?;
377                    chunk.truncate(n);
378                    me.plain = chunk;
379                    me.read_state = ReadState::Length;
380                }
381            }
382        }
383    }
384}

386impl<T> Ss2022Stream<T> {
387    /// Seals up to [`MAX_WRITE`] bytes of `buf` into `write_buf`, after the
388    /// response header if this is the server's first write. Returns how
389    /// much of `buf` it took.
390    fn seal(&mut self, buf: &[u8]) -> io::Result<usize> {
391        let total = min(buf.len(), MAX_WRITE);
392        let mut data = &buf[..total];
393        self.write_buf.clear();
394        self.write_pos = 0;
395        self.write_buf
396            .reserve(total + 512 + total.div_ceil(MAX_CHUNK) * (2 + 2 * TAG_LEN));
397        if matches!(self.write_state, WriteState::ResponseHeader) {
398            let salt = random_salt(self.method.key_len());
399            let mut enc = ChunkAead::new(self.method, &session_subkey(&self.psk, &salt))?;
400            let out = &mut self.write_buf;
401            out.extend_from_slice(&salt);
402            let (first, rest) = data.split_at(min(data.len(), MAX_CHUNK));
403            let start = out.len();
404            out.push(HEADER_TYPE_SERVER);
405            out.extend_from_slice(&now_secs().to_be_bytes());
406            out.extend_from_slice(&self.request_salt);
407            out.extend_from_slice(&(first.len() as u16).to_be_bytes());
408            enc.seal(out, start)?;
409            let start = out.len();
410            out.extend_from_slice(first);
411            enc.seal(out, start)?;
412            data = rest;
413            self.request_salt = Vec::new();
414            self.enc = Some(enc);
415        }
416        let enc = self.enc.as_mut().ok_or_else(crypto_err)?;
417        for chunk in data.chunks(MAX_CHUNK) {
418            seal_chunk(&mut self.write_buf, enc, chunk)?;
419        }
420        Ok(total)
421    }
422}

424impl<T> AsyncWrite for Ss2022Stream<T>
425where
426    T: AsyncWrite + Unpin,
427{
428    fn poll_write(
429        mut self: Pin<&mut Self>,
430        cx: &mut Context,
431        buf: &[u8],
432    ) -> Poll<io::Result<usize>> {
433        if buf.is_empty() {
434            return Poll::Ready(Ok(0));
435        }
436        let me = &mut *self;
437        loop {
438            match me.write_state {
439                WriteState::ResponseHeader | WriteState::Ready => {
440                    let n = me.seal(buf)?;
441                    me.write_state = WriteState::Pending(n);
442                }
443                // As with the legacy stream, the caller is expected to
444                // retry with the same buffer after Pending.
445                WriteState::Pending(consumed) => {
446                    while me.write_pos < me.write_buf.len() {
447                        let n =
448                            ready!(Pin::new(&mut me.inner)
449                                .poll_write(cx, &me.write_buf[me.write_pos..]))?;
450                        if n == 0 {
451                            return Poll::Ready(Err(early_eof()));
452                        }
453                        me.write_pos += n;
454                    }
455                    me.write_state = WriteState::Ready;
456                    return Poll::Ready(Ok(consumed));
457                }
458            }
459        }
460    }
461
462    fn poll_flush(mut self: Pin<&mut Self>, cx: &mut Context) -> Poll<io::Result<()>> {
463        // Drop the write buffer once idle so idle connections don't keep it.
464        if matches!(self.write_state, WriteState::Ready) && self.write_buf.capacity() > 0 {
465            self.write_buf = Vec::new();
466            self.write_pos = 0;
467        }
468        Pin::new(&mut self.inner).poll_flush(cx)
469    }
470
471    fn poll_shutdown(mut self: Pin<&mut Self>, cx: &mut Context) -> Poll<io::Result<()>> {
472        Pin::new(&mut self.inner).poll_shutdown(cx)
473    }
474}

499        method: Method,
500        psks: Vec<Vec<u8>>,
501        config: ServerConfig,
502        want_user: Option<&str>,
503    ) {
504        rt().block_on(async move {
505            let (c, s) = duplex(1 << 20);
506            let dest = SocksAddr::try_from(("example.com", 443)).unwrap();
507            let big: Vec<u8> = (0..200_000u32).map(|i| i as u8).collect();
508            let client = connect(c, method, &psks, &dest, b"hello").await.unwrap();
509            let accepted = accept(s, &config).await.unwrap();
510            assert_eq!(accepted.destination, dest);
511            assert_eq!(accepted.user.as_deref(), want_user);
512            let (mut client, mut server) = (client, accepted.stream);
513
514            let mut got = [0u8; 5];
515            server.read_exact(&mut got).await.unwrap();
516            assert_eq!(&got, b"hello");
517
518            let big2 = big.clone();
519            let w = tokio::spawn(async move {
520                server.write_all(&big2).await.unwrap();
521                server.flush().await.unwrap();
522                server
523            });
524            let mut back = vec![0u8; big.len()];
525            client.read_exact(&mut back).await.unwrap();
526            assert_eq!(back, big);

639            wire.extend_from_slice(&(now_secs() - 31).to_be_bytes());
640            wire.extend_from_slice(&10u16.to_be_bytes());
641            enc.seal(&mut wire, start).unwrap();
642            let err = accept(&wire[..], &config).await.err().unwrap();
643            assert!(err.to_string().contains("timestamp"), "{}", err);
644            // The salt of a rejected request is not recorded.
645            assert!(config.salts.check_and_insert(&salt).is_ok());
646        });
647    }
