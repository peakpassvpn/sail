24impl InboundDatagramHandler for Handler {
25    async fn handle<'a>(&'a self, socket: AnyInboundDatagram) -> io::Result<AnyInboundTransport> {
26        tracing::trace!("handling inbound datagram");
27        let dgram = ShadowedDatagram::new(&self.cipher, &self.password)?;
28        Ok(InboundTransport::Datagram(
29            Box::new(Datagram { dgram, socket }),
30            None,
31        ))
32    }

35pub struct Datagram {
36    dgram: ShadowedDatagram,
37    socket: AnyInboundDatagram,
38}
39
40impl InboundDatagram for Datagram {
41    fn split(
42        self: Box<Self>,
43    ) -> (
44        Box<dyn InboundDatagramRecvHalf>,
45        Box<dyn InboundDatagramSendHalf>,
46    ) {
47        let dgram = Arc::new(self.dgram);
48        let (rh, sh) = self.socket.split();
49        (
50            Box::new(DatagramRecvHalf(dgram.clone(), rh)),
51            Box::new(DatagramSendHalf(dgram, sh)),
52        )
53    }
54
55    fn into_std(self: Box<Self>) -> io::Result<std::net::UdpSocket> {
56        Err(io::Error::other("shadowsocks datagram"))
57    }
58}

60pub struct DatagramRecvHalf(Arc<ShadowedDatagram>, Box<dyn InboundDatagramRecvHalf>);
61
62#[async_trait]
63impl InboundDatagramRecvHalf for DatagramRecvHalf {
64    async fn recv_from(
65        &mut self,
66        buf: &mut [u8],
67    ) -> ProxyResult<(usize, DatagramSource, SocksAddr)> {
68        let mut recv_buf = BytesMut::new();
69        recv_buf.resize(buf.len(), 0);
70        let (n, src_addr, _) = self.1.recv_from(&mut recv_buf).await?;
71        recv_buf.resize(n, 0);
72        let plaintext = self
73            .0
74            .decrypt(recv_buf)
75            .map_err(|e| ProxyError::DatagramWarn(anyhow!("Decrypt payload failed: {}", e)))?;
76        let dst_addr = SocksAddr::try_from((&plaintext[..], SocksAddrWireType::PortLast))
77            .map_err(|e| ProxyError::DatagramWarn(anyhow!("Parse target address failed: {}", e)))?;
78        let header_size = dst_addr.size();
79        let payload_size = plaintext.len() - header_size;
80        assert!(buf.len() >= payload_size);
81        buf[..payload_size].copy_from_slice(&plaintext[header_size..header_size + payload_size]);
82        Ok((payload_size, src_addr, dst_addr))
83    }
84}

88#[async_trait]
89impl InboundDatagramSendHalf for DatagramSendHalf {
90    async fn send_to(
91        &mut self,
92        buf: &[u8],
93        src_addr: &SocksAddr,
94        dst_addr: &SocketAddr,
95    ) -> io::Result<usize> {
96        let mut send_buf = BytesMut::new();
97        src_addr.write_buf(&mut send_buf, SocksAddrWireType::PortLast);
98        send_buf.put_slice(buf);
99        let ciphertext = self.0.encrypt(send_buf).map_err(|_| shadow::crypto_err())?;
100        self.1.send_to(&ciphertext[..], src_addr, dst_addr).await
101    }
102
103    async fn close(&mut self) -> io::Result<()> {
104        self.1.close().await
105    }
