12/// `Packet` commands for `payload`, each small enough for a datagram of
13/// `max_size` bytes. Only the first fragment carries the address.
14pub fn fragment(
15    assoc_id: u16,
16    pkt_id: u16,
17    addr: &SocksAddr,
18    payload: &[u8],
19    max_size: usize,
20) -> io::Result<Vec<Bytes>> {
21    let mut header = PacketHeader {
22        assoc_id,
23        pkt_id,
24        frag_total: 1,
25        frag_id: 0,
26        addr: Some(addr.clone()),
27    };
28    // Sized by the first fragment's header, which is the largest.
29    let room = max_size
30        .checked_sub(header.len())
31        .filter(|n| *n > 0)
32        .ok_or_else(|| {
33            io::Error::other(format!(
34                "tuic: datagrams of {} bytes cannot carry a packet header",
35                max_size
36            ))
37        })?;
38    if payload.len() <= room {
39        return Ok(vec![encode_packet(&header, payload)?]);
40    }
41    let total = payload.len().div_ceil(room);
42    header.frag_total = u8::try_from(total).map_err(|_| {
43        io::Error::other(format!(
44            "tuic: a packet of {} bytes needs more than 255 fragments",
45            payload.len()
46        ))
47    })?;
48    payload
49        .chunks(room)
50        .enumerate()
51        .map(|(i, chunk)| {
52            header.frag_id = i as u8;
53            if i == 1 {
54                header.addr = None;
55            }
56            encode_packet(&header, chunk)
57        })
58        .collect()
59}

61/// A packet some of whose fragments have come.
62struct Partial {
63    fragments: Vec<Option<Bytes>>,
64    received: usize,
65    size: usize,
66    addr: Option<SocksAddr>,
67    started: Instant,
68}
69
70/// Fragments of one association's packets waiting for the rest.
71///
72/// Bounded: at most `max_pending` packets wait at once, the oldest giving
73/// way to a new one, and each waits at most `timeout`. UDP loses packets
74/// anyway, so dropping one is always allowed.
75pub struct Reassembler {
76    pending: HashMap<u16, Partial>,
77    max_pending: usize,
78    timeout: Duration,
79}
80
81impl Reassembler {
82    pub fn new(max_pending: usize, timeout: Duration) -> Self {
83        Self {
84            pending: HashMap::new(),
85            max_pending: max_pending.max(1),
86            timeout,
87        }
88    }
89
90    /// Takes one fragment, returning the packet it completes, if any.
91    pub fn feed(
92        &mut self,
93        header: PacketHeader,
94        data: Bytes,
95        now: Instant,
96    ) -> Option<(SocksAddr, Bytes)> {
97        if header.frag_total <= 1 {
98            return header.addr.map(|addr| (addr, data));
99        }
100        let total = header.frag_total as usize;
101        let id = header.frag_id as usize;
102        if id >= total {
103            return None;
104        }
105        self.pending
106            .retain(|_, p| now.saturating_duration_since(p.started) < self.timeout);
107        if let Some(p) = self.pending.get(&header.pkt_id) {
108            // A packet id reused for another packet: the old one is lost.
109            if p.fragments.len() != total {
110                self.pending.remove(&header.pkt_id);
111            }
112        }
113        if !self.pending.contains_key(&header.pkt_id) {
114            if self.pending.len() >= self.max_pending {
115                let oldest = self
116                    .pending
117                    .iter()
118                    .min_by_key(|(_, p)| p.started)
119                    .map(|(k, _)| *k);
120                if let Some(oldest) = oldest {
121                    self.pending.remove(&oldest);
122                }
123            }
124            self.pending.insert(
125                header.pkt_id,
126                Partial {
127                    fragments: vec![None; total],
128                    received: 0,
129                    size: 0,
130                    addr: None,
131                    started: now,
132                },
133            );
134        }
135        let p = self.pending.get_mut(&header.pkt_id)?;
136        if p.fragments[id].is_some() {
137            return None;
138        }
139        // A reassembled packet is still one UDP packet.
140        if p.size + data.len() > u16::MAX as usize {
141            self.pending.remove(&header.pkt_id);
142            return None;
143        }
144        if id == 0 {
145            p.addr = header.addr;
146        }
147        p.size += data.len();
148        p.fragments[id] = Some(data);
149        p.received += 1;
150        if p.received < total {
151            return None;
152        }
153        let p = self.pending.remove(&header.pkt_id)?;
154        let addr = p.addr?;
155        let mut packet = BytesMut::with_capacity(p.size);
156        for fragment in p.fragments.into_iter().flatten() {
157            packet.extend_from_slice(&fragment);
158        }
159        Some((addr, packet.freeze()))
160    }
161
162    #[cfg(test)]
163    fn pending(&self) -> usize {
164        self.pending.len()
165    }
166}

173    fn decode(wire: Bytes) -> (PacketHeader, Bytes) {
174        match decode_datagram(wire).unwrap() {
175            Datagram::Packet(h, p) => (h, p),
176            Datagram::Heartbeat => panic!("heartbeat"),
177        }
178    }

180    fn addr() -> SocksAddr {
181        SocksAddr::Domain("example.com".into(), 443)
182    }

184    #[test]
185    fn a_small_packet_is_one_fragment() {
186        let wire = fragment(1, 2, &addr(), b"hello", 1200).unwrap();
187        assert_eq!(wire.len(), 1);
188        let (h, p) = decode(wire[0].clone());
189        assert_eq!((h.frag_total, h.frag_id), (1, 0));
190        let mut r = Reassembler::new(4, Duration::from_secs(10));
191        assert_eq!(
192            r.feed(h, p, Instant::now()),
193            Some((addr(), Bytes::from("hello")))
194        );
195    }
196
197    #[test]
198    fn fragments_fit_and_reassemble_in_any_order() {
199        let payload: Vec<u8> = (0..5000u32).map(|i| i as u8).collect();
200        let wire = fragment(3, 9, &addr(), &payload, 1200).unwrap();
201        assert!(wire.len() > 4);
202        assert!(wire.iter().all(|w| w.len() <= 1200));
203        let mut frags: Vec<_> = wire.into_iter().map(decode).collect();
204        assert!(frags[0].0.addr.is_some());
205        assert!(frags[1..].iter().all(|(h, _)| h.addr.is_none()));
206        frags.reverse();
207        let last = frags.pop().unwrap();
208        let mut r = Reassembler::new(4, Duration::from_secs(10));
209        let now = Instant::now();
210        for (h, p) in frags {
211            assert_eq!(r.feed(h, p, now), None);
212        }
213        let (a, p) = r.feed(last.0, last.1, now).unwrap();
214        assert_eq!(a, addr());
215        assert_eq!(&p[..], &payload[..]);
216        assert_eq!(r.pending(), 0);
217    }

219    #[test]
220    fn duplicates_are_ignored() {
221        let payload = vec![1u8; 3000];
222        let frags: Vec<_> = fragment(1, 1, &addr(), &payload, 1200)
223            .unwrap()
224            .into_iter()
225            .map(decode)
226            .collect();
227        let mut r = Reassembler::new(4, Duration::from_secs(10));
228        let now = Instant::now();
229        assert_eq!(r.feed(frags[0].0.clone(), frags[0].1.clone(), now), None);
230        assert_eq!(r.feed(frags[0].0.clone(), frags[0].1.clone(), now), None);
231        let mut done = None;
232        for (h, p) in frags.into_iter().skip(1) {
233            done = r.feed(h, p, now);
234        }
235        assert_eq!(done.unwrap().1.len(), 3000);
236    }

238    #[test]
239    fn pending_packets_are_bounded_and_expire() {
240        let mut r = Reassembler::new(2, Duration::from_secs(10));
241        let now = Instant::now();
242        let first = |pkt_id| {
243            decode(
244                fragment(1, pkt_id, &addr(), &[0u8; 3000], 1200)
245                    .unwrap()
246                    .remove(0),
247            )
248        };
249        for pkt_id in 0..5 {
250            let (h, p) = first(pkt_id);
251            r.feed(h, p, now);
252        }
253        assert_eq!(r.pending(), 2);
254        let (h, p) = first(9);
255        r.feed(h, p, now + Duration::from_secs(11));
256        assert_eq!(r.pending(), 1);
257    }

259    #[test]
260    fn a_lost_first_fragment_loses_the_packet() {
261        let frags: Vec<_> = fragment(1, 1, &addr(), &[0u8; 3000], 1200)
262            .unwrap()
263            .into_iter()
264            .map(decode)
265            .collect();
266        let mut r = Reassembler::new(4, Duration::from_secs(10));
267        let now = Instant::now();
268        // Another sender's fragment 0 without address never reaches here
269        // (decoding rejects it), so feed only the rest: nothing completes.
270        for (h, p) in frags.into_iter().skip(1) {
271            assert_eq!(r.feed(h, p, now), None);
272        }
273    }
