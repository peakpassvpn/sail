19/// Which way UDP packets travel.
20#[derive(Deserialize, Debug, Clone, Copy, Default, PartialEq, Eq)]

22pub enum UdpRelayMode {
23    /// QUIC datagrams, fragmented to fit.
24    #[default]
25    Native,
26    /// One unidirectional stream per packet.

43/// The transport parameters of a TUIC connection.
44pub fn transport_config(
45    congestion: CongestionControl,
46    tuning: &crate::runtime::options::Quic,
47    side: Side,
48) -> quinn::TransportConfig {
49    let mut config = crate::transport::quic::transport_config(tuning, side, congestion.factory());
50    // In `quic` relay mode every UDP packet is a stream of its own.
51    config.max_concurrent_uni_streams(quinn::VarInt::from_u32(tuning.max_concurrent_streams));
52    // TUIC keeps a connection alive with heartbeats while it relays, and
53    // lets it go idle otherwise.
54    config.keep_alive_interval(None);
55    config
56}

58pub fn parse_uuid(kind: &str, tag: &str, field: &str, uuid: &str) -> Result<[u8; 16]> {
59    uuid::Uuid::parse_str(uuid)
60        .map(|u| *u.as_bytes())
61        .map_err(|e| anyhow!("[{}] {}: {}: invalid UUID: {}", tag, kind, field, e))
62}

64/// The `Authenticate` token: TLS keying material exported with the UUID
65/// as label and the password as context.
66pub fn token(
67    conn: &quinn::Connection,
68    uuid: &[u8; 16],
69    password: &[u8],
70) -> io::Result<[u8; TOKEN_LEN]> {
71    let mut token = [0u8; TOKEN_LEN];
72    conn.export_keying_material(&mut token, uuid, password)
73        .map_err(|_| io::Error::other("tuic: export keying material failed"))?;
74    Ok(token)
75}

77/// Compares tokens in time independent of where they differ.
78#[cfg_attr(not(feature = "inbound-tuic"), allow(dead_code))]
79pub fn tokens_equal(a: &[u8], b: &[u8]) -> bool {
80    a.len() == b.len() && a.iter().zip(b).fold(0u8, |acc, (x, y)| acc | (x ^ y)) == 0
81}

83/// Relays running on a connection. Heartbeats are sent only while there
84/// are any.
85#[derive(Clone, Default)]
86pub struct Activity(Arc<AtomicUsize>);
87
88impl Activity {
89    pub fn start(&self) -> ActiveGuard {
90        self.0.fetch_add(1, Ordering::Relaxed);
91        ActiveGuard(self.0.clone())
92    }
93
94    pub fn is_active(&self) -> bool {
95        self.0.load(Ordering::Relaxed) > 0
96    }
97}
98
99/// One relay, counted while it lives.
100pub struct ActiveGuard(Arc<AtomicUsize>);
101
102impl Drop for ActiveGuard {
103    fn drop(&mut self) {
104        self.0.fetch_sub(1, Ordering::Relaxed);
105    }
106}

108/// Sends `Heartbeat` every `interval` while `activity` says something is
109/// relayed, until the connection closes.
110pub async fn heartbeat(conn: quinn::Connection, activity: Activity, interval: Duration) {
111    let mut ticker = tokio::time::interval(interval);
112    ticker.tick().await;
113    loop {
114        tokio::select! {
115            _ = ticker.tick() => {}
116            _ = conn.closed() => return,
117        }
118        if activity.is_active() {
119            if let Err(e) = conn.send_datagram(encode_heartbeat()) {
120                trace!("tuic heartbeat failed: {}", e);
121                return;
122            }
123        }
124    }
125}

127/// Sends one UDP packet on `conn` the way `mode` says.
128pub async fn send_packet(
129    conn: &quinn::Connection,
130    mode: UdpRelayMode,
131    assoc_id: u16,
132    pkt_id: u16,
133    addr: &SocksAddr,
134    payload: &[u8],
135) -> io::Result<()> {
136    match mode {
137        UdpRelayMode::Native => {
138            let max = conn
139                .max_datagram_size()
140                .ok_or_else(|| io::Error::other("tuic: the peer does not take QUIC datagrams"))?;
141            for datagram in fragment(assoc_id, pkt_id, addr, payload, max)? {
142                conn.send_datagram(datagram).map_err(io::Error::other)?;
143            }
144            Ok(())
145        }
146        UdpRelayMode::Quic => {
147            let header = PacketHeader {
148                assoc_id,
149                pkt_id,
150                frag_total: 1,
151                frag_id: 0,
152                addr: Some(addr.clone()),
153            };
154            let wire = encode_packet(&header, payload)?;
155            let mut send = conn.open_uni().await.map_err(io::Error::other)?;
156            send.write_all(&wire).await?;
157            send.finish().map_err(io::Error::other)?;
158            Ok(())
159        }
160    }
161}

167#[cfg(test)]
168mod tests {
169    use super::*;
170
171    #[derive(Deserialize)]
172    #[serde(deny_unknown_fields)]
173    struct Modes {
174        congestion_control: CongestionControl,
175        udp_relay_mode: UdpRelayMode,
176    }
177
178    #[test]
179    fn modes_are_named_as_in_sing_box() {
180        let parse = |cc: &str, mode: &str| {
181            serde_json::from_str::<Modes>(&format!(
182                r#"{{"congestion_control": "{}", "udp_relay_mode": "{}"}}"#,
183                cc, mode
184            ))
185        };
186        let m = parse("new_reno", "quic").unwrap();
187        assert_eq!(m.congestion_control, CongestionControl::NewReno);
188        assert_eq!(m.udp_relay_mode, UdpRelayMode::Quic);
189        assert!(parse("bbr", "native").is_ok());
190        assert!(parse("cubic", "native").is_ok());
191        assert!(parse("reno", "native").is_err());
192        assert!(parse("bbr", "tcp").is_err());
193    }

195    #[test]
196    fn tokens_compare() {
197        assert!(tokens_equal(b"abc", b"abc"));
198        assert!(!tokens_equal(b"abc", b"abd"));
199        assert!(!tokens_equal(b"abc", b"ab"));
200    }

202    #[test]
203    fn activity_counts_guards() {
204        let a = Activity::default();
205        assert!(!a.is_active());

210        drop(h);
