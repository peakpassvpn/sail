22#[derive(Deserialize)]
23#[serde(deny_unknown_fields)]
24struct SocksInboundOptions {
25    /// Clients must authenticate as one of these; anyone may connect when
26    /// there are none.
27    #[serde(default)]
28    users: Vec<SocksUser>,
29}
30
31#[derive(Deserialize)]
32#[serde(deny_unknown_fields)]
33pub(crate) struct SocksUser {
34    pub username: String,
35    pub password: String,
36}
37
38/// The users by username, with their passwords.
39///
40/// A username may appear only once: which of two passwords would name the
41/// user is otherwise ambiguous. RFC 1929 carries each in at most 255 bytes,
42/// so a longer one could never authenticate.
43pub(crate) fn users_by_name(tag: &str, users: Vec<SocksUser>) -> Result<HashMap<String, String>> {
44    let mut map = HashMap::with_capacity(users.len());
45    for user in users {
46        if user.username.is_empty() || user.username.len() > 255 {
47            return Err(anyhow!(
48                "[{}] inbound: users: username \"{}\" must be 1 to 255 bytes",
49                tag,
50                user.username
51            ));
52        }
53        if user.password.is_empty() || user.password.len() > 255 {
54            return Err(anyhow!(
55                "[{}] inbound: users: {}: password must be 1 to 255 bytes",
56                tag,
57                user.username
58            ));
59        }
60        if map.contains_key(&user.username) {
61            return Err(anyhow!(
62                "[{}] inbound: users: username \"{}\" appears more than once",
63                tag,
64                user.username
65            ));
66        }
67        map.insert(user.username, user.password);
68    }
69    Ok(map)
70}
71
72fn build(ctx: &InboundContext<'_>) -> Result<AnyInboundHandler> {
73    let options: SocksInboundOptions = ctx.options()?;
74    let users = users_by_name(ctx.tag, options.users)?;
75    let stream = Arc::new(StreamHandler::new(users, Arc::new(Associations::default())));
76    // No datagram handler: UDP ASSOCIATE binds a relay socket per
77    // association, as RFC 1928 and sing-box do, and nothing is served on
78    // the listen port's UDP.
79    Ok(Arc::new(Handler::new(
80        ctx.tag.to_owned(),
81        Some(stream),
82        None,
83    )))
84}
