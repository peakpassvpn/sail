2use std::io;

4use async_trait::async_trait;
5use futures::future::TryFutureExt;

7use crate::{adapter::*, session::*};

9pub struct Handler {
10    pub address: String,
11    pub port: u16,
12    pub username: String,
13    pub password: String,
14}
15
16#[async_trait]
17impl OutboundStreamHandler for Handler {
18    fn connect_addr(&self) -> OutboundConnect {
19        OutboundConnect::Proxy(Network::Tcp, self.address.clone(), self.port)
20    }
21
22    async fn handle<'a>(
23        &'a self,
24        sess: &'a Session,
25        _lhs: Option<&mut AnyStream>,
26        stream: Option<AnyStream>,
27    ) -> io::Result<AnyStream> {
28        tracing::trace!("handling outbound stream");
29        let mut stream = stream.ok_or_else(|| io::Error::other("invalid input"))?;
30        let auth = match (&self.username, &self.password) {
31            (auth_username, _) if auth_username.is_empty() => None,
32            (auth_username, auth_password) => Some(Auth {
33                username: auth_username.to_owned(),
34                password: auth_password.to_owned(),
35            }),
36        };
37        match &sess.destination {
38            SocksAddr::Ip(a) => {
39                let _ = async_socks5::connect(&mut stream, a.to_owned(), auth)
40                    .map_err(io::Error::other)
41                    .await?;
42            }
43            SocksAddr::Domain(domain, port) => {
44                let _ = async_socks5::connect(&mut stream, (domain.to_owned(), *port), auth)
45                    .map_err(io::Error::other)
46                    .await?;
47            }
48        }
49        Ok(stream)
50    }
51}

53#[cfg(test)]
54mod tests {
55    use super::*;
56    use tokio::net::TcpListener;
57
58    #[tokio::test]
59    async fn test_socks5_outbound_handler() {
60        let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
61        let addr = listener.local_addr().unwrap();
62
63        let handler = Handler {
64            address: "127.0.0.1".to_string(),
65            port: addr.port(),
66            username: "".to_string(),
67            password: "".to_string(),
68        };
69
70        let sess = Session {
71            destination: SocksAddr::Domain("google.com".to_string(), 80),
72            ..Default::default()
73        };
74
75        // Mock a SOCKS5 server in a separate task.
76        //
77        // It has to consume everything the client sends, not merely enough to
78        // know what to reply: a socket dropped with unread bytes still in it is
79        // reset rather than closed, and the client loses the reply it was in
80        // the middle of being given. This used to read the greeting as two
81        // bytes, one short of the shortest there is, and every read after it
82        // was off by that byte -- so the request went unread, the close became
83        // a reset, and whether the client got the reply out first was a race it
84        // won often enough on Unix to look like a passing test.
85        tokio::spawn(async move {
86            let (mut socket, _) = listener.accept().await.unwrap();
87            use tokio::io::{AsyncReadExt, AsyncWriteExt};
88
89            // Greeting: VER, NMETHODS, and then that many method bytes.
90            let mut greeting = [0u8; 2];
91            socket.read_exact(&mut greeting).await.unwrap();
92            let mut methods = vec![0u8; greeting[1] as usize];
93            socket.read_exact(&mut methods).await.unwrap();
94            socket.write_all(&[0x05, 0x00]).await.unwrap();
95
96            // Request: VER, CMD, RSV, ATYP, and then the address and port.
97            let mut request = [0u8; 4];
98            socket.read_exact(&mut request).await.unwrap();
99            let address_len = match request[3] {
100                0x01 => 4,
101                0x04 => 16,
102                0x03 => {
103                    let mut len = [0u8; 1];
104                    socket.read_exact(&mut len).await.unwrap();
105                    len[0] as usize
106                }
107                other => panic!("the client asked for address type {}", other),
108            };
109            let mut address_and_port = vec![0u8; address_len + 2];
110            socket.read_exact(&mut address_and_port).await.unwrap();
111
112            // Reply
113            socket
114                .write_all(&[0x05, 0x00, 0x00, 0x01, 0, 0, 0, 0, 0, 0])
115                .await
116                .unwrap();
117        });
118
119        let client_stream = tokio::net::TcpStream::connect(addr).await.unwrap();
120        let result = handler
121            .handle(&sess, None, Some(Box::new(client_stream)))
122            .await;
123        assert!(
124            result.is_ok(),
125            "the handshake failed: {}",
126            result.err().unwrap()
127        );
128    }
129}
