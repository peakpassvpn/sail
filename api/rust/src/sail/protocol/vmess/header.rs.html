1//! The VMess request and response headers, AEAD form only.
2//!
3//! A request is an auth ID -- a timestamp, four random bytes and their
4//! CRC32, AES-encrypted under a key from the user's -- then the header's
5//! length and the header, each sealed with AES-128-GCM under keys derived
6//! from the user's key, the auth ID and a connection nonce. The header
7//! carries the body keys, the options, the security and the command and
8//! destination. The response is its own sealed length and header, under
9//! keys derived from the body keys.
10

46/// The longest request header there can be: the fixed part, the longest
47/// address, the most padding and the checksum.
48const MAX_HEADER_LEN: usize = 38 + 1 + 255 + 2 + 15 + 4;

50const TAG_LEN: usize = 16;

52/// The user's key: MD5 of the UUID and a constant.
53pub fn cmd_key(uuid: &[u8; 16]) -> [u8; 16] {
54    let mut md5 = Md5::new();
55    md5.update(uuid);
56    md5.update(b"c48619fe-8f02-49e0-b9e9-edf763e17e21");
57    md5.finalize().into()
58}

60fn now() -> u64 {
61    SystemTime::now()
62        .duration_since(UNIX_EPOCH)
63        .map(|d| d.as_secs())
64        .unwrap_or(0)
65}

67fn crypto_error() -> io::Error {
68    io::Error::new(io::ErrorKind::InvalidData, "vmess: bad header")
69}
70
71fn gcm(key: &[u8]) -> io::Result<AeadCtx> {
72    AeadCtx::new_default_tag(&Algorithm::aes_128_gcm(), key).map_err(io::Error::other)
73}
74
75/// Seals `data` with AES-128-GCM, appending the tag.
76fn seal(key: &[u8], nonce: &[u8], data: &mut Vec<u8>, aad: &[u8]) -> io::Result<()> {
77    let mut tag = [0u8; TAG_LEN];
78    gcm(key)?
79        .seal_in_place_mut(nonce, data, &mut tag, aad)
80        .map_err(io::Error::other)?;
81    data.extend_from_slice(&tag);
82    Ok(())
83}
84
85/// Opens `data`, ciphertext and tag, in place; the plaintext is what is
86/// left.
87fn open(key: &[u8], nonce: &[u8], data: &mut Vec<u8>, aad: &[u8]) -> io::Result<()> {
88    if data.len() < TAG_LEN {
89        return Err(crypto_error());
90    }
91    let tag = data.split_off(data.len() - TAG_LEN);
92    gcm(key)?
93        .open_in_place_mut(nonce, data, &tag, aad)
94        .map_err(|_| crypto_error())
95}

97fn auth_id_cipher(cmd_key: &[u8; 16]) -> Aes128 {
98    let key = vmess_kdf_1_one_shot(cmd_key, KDF_SALT_CONST_AUTH_ID_ENCRYPTION_KEY);
99    Aes128::new((&key[..16]).into())
100}
101
102fn auth_id(cmd_key: &[u8; 16], time: u64) -> [u8; 16] {
103    let mut id = [0u8; 16];
104    id[..8].copy_from_slice(&time.to_be_bytes());
105    rand::thread_rng().fill_bytes(&mut id[8..12]);
106    let crc = crc32fast::hash(&id[..12]);
107    id[12..].copy_from_slice(&crc.to_be_bytes());
108    auth_id_cipher(cmd_key).encrypt_block((&mut id).into());
109    id
110}

112/// FNV-1a, 32 bits: the checksum at the end of a request header.
113fn fnv1a(data: &[u8]) -> u32 {
114    data.iter().fold(0x811c9dc5u32, |h, b| {
115        (h ^ *b as u32).wrapping_mul(0x01000193)
116    })
117}

119/// A request header, the part that is sealed.
120#[derive(Debug, Clone)]
121pub struct RequestHeader {
122    pub body_iv: [u8; 16],
123    pub body_key: [u8; 16],
124    /// Echoed by the server in its response.
125    pub response_auth: u8,
126    pub option: u8,
127    pub security: u8,
128    pub command: u8,
129    /// None for Mux.
130    pub address: Option<SocksAddr>,
131}
132
133impl RequestHeader {
134    /// A new request with fresh body keys.
135    pub fn new(option: u8, security: u8, command: u8, address: Option<SocksAddr>) -> Self {
136        let mut rng = rand::thread_rng();
137        let mut header = RequestHeader {
138            body_iv: [0; 16],
139            body_key: [0; 16],
140            response_auth: rng.gen(),
141            option,
142            security,
143            command,
144            address,
145        };
146        rng.fill_bytes(&mut header.body_iv);
147        rng.fill_bytes(&mut header.body_key);
148        header
149    }
150
151    fn encode(&self) -> Vec<u8> {
152        let mut rng = rand::thread_rng();
153        let padding: u8 = rng.gen_range(0..16);
154        let mut buf = BytesMut::with_capacity(64);
155        buf.put_u8(VERSION);
156        buf.put_slice(&self.body_iv);
157        buf.put_slice(&self.body_key);
158        buf.put_u8(self.response_auth);
159        buf.put_u8(self.option);
160        buf.put_u8(padding << 4 | self.security);
161        buf.put_u8(0);
162        buf.put_u8(self.command);
163        if let Some(address) = &self.address {
164            write_addr_port(&mut buf, address);
165        }
166        let mut pad = [0u8; 15];
167        rng.fill_bytes(&mut pad);
168        buf.put_slice(&pad[..padding as usize]);
169        let checksum = fnv1a(&buf);
170        buf.put_u32(checksum);
171        buf.to_vec()
172    }
173
174    fn decode(buf: &[u8]) -> io::Result<Self> {
175        let invalid =
176            |what: &str| io::Error::new(io::ErrorKind::InvalidData, format!("vmess: {}", what));
177        if buf.len() < 38 + 4 {
178            return Err(invalid("header too short"));
179        }
180        let (body, checksum) = buf.split_at(buf.len() - 4);
181        if fnv1a(body).to_be_bytes() != checksum {
182            return Err(invalid("bad header checksum"));
183        }
184        if body[0] != VERSION {
185            return Err(invalid("unknown version"));
186        }
187        let mut header = RequestHeader {
188            body_iv: body[1..17].try_into().map_err(|_| invalid("short"))?,
189            body_key: body[17..33].try_into().map_err(|_| invalid("short"))?,
190            response_auth: body[33],
191            option: body[34],
192            security: body[35] & 0x0f,
193            command: body[37],
194            address: None,
195        };
196        let padding = (body[35] >> 4) as usize;
197        let rest = &body[38..];
198        let used = match header.command {
199            COMMAND_TCP | COMMAND_UDP => {
200                let (address, used) = parse_addr_port(rest)?;
201                header.address = Some(address);
202                used
203            }
204            COMMAND_MUX => 0,
205            _ => return Err(invalid("unknown command")),
206        };
207        if rest.len() != used + padding {
208            return Err(invalid("bad header length"));
209        }
210        Ok(header)
211    }
212
213    /// The request as it goes on the wire, sealed for the user `cmd_key`.
214    pub fn seal(&self, cmd_key: &[u8; 16]) -> io::Result<Vec<u8>> {
215        let auth_id = auth_id(cmd_key, now());
216        let mut nonce = [0u8; 8];
217        rand::thread_rng().fill_bytes(&mut nonce);
218        let header = self.encode();
219
220        let mut length = (header.len() as u16).to_be_bytes().to_vec();
221        let key = vmess_kdf_3_one_shot(
222            cmd_key,
223            KDF_SALT_CONST_VMESS_HEADER_PAYLOAD_LENGTH_AEAD_KEY,
224            &auth_id,
225            &nonce,
226        );
227        let iv = vmess_kdf_3_one_shot(
228            cmd_key,
229            KDF_SALT_CONST_VMESS_HEADER_PAYLOAD_LENGTH_AEAD_IV,
230            &auth_id,
231            &nonce,
232        );
233        seal(&key[..16], &iv[..12], &mut length, &auth_id)?;
234
235        let mut payload = header;
236        let key = vmess_kdf_3_one_shot(
237            cmd_key,
238            KDF_SALT_CONST_VMESS_HEADER_PAYLOAD_AEAD_KEY,
239            &auth_id,
240            &nonce,
241        );
242        let iv = vmess_kdf_3_one_shot(
243            cmd_key,
244            KDF_SALT_CONST_VMESS_HEADER_PAYLOAD_AEAD_IV,
245            &auth_id,
246            &nonce,
247        );
248        seal(&key[..16], &iv[..12], &mut payload, &auth_id)?;
249
250        let mut out = Vec::with_capacity(16 + length.len() + 8 + payload.len());
251        out.extend_from_slice(&auth_id);
252        out.extend_from_slice(&length);
253        out.extend_from_slice(&nonce);
254        out.extend_from_slice(&payload);
255        Ok(out)
256    }
257
258    /// The key and IV of the response body: the first 16 bytes of the
259    /// SHA-256 of the request's.
260    pub fn response_keys(&self) -> ([u8; 16], [u8; 16]) {
261        use sha2::Sha256;
262        let key = Sha256::digest(self.body_key);
263        let iv = Sha256::digest(self.body_iv);
264        (
265            key[..16].try_into().expect("SHA-256 is 32 bytes"),
266            iv[..16].try_into().expect("SHA-256 is 32 bytes"),
267        )
268    }
269
270    /// The server's response header, sealed.
271    pub fn seal_response(&self) -> io::Result<Vec<u8>> {
272        let (key, iv) = self.response_keys();
273        let mut length = 4u16.to_be_bytes().to_vec();
274        let len_key = vmess_kdf_1_one_shot(&key, KDF_SALT_CONST_AEAD_RESP_HEADER_LEN_KEY);
275        let len_iv = vmess_kdf_1_one_shot(&iv, KDF_SALT_CONST_AEAD_RESP_HEADER_LEN_IV);
276        seal(&len_key[..16], &len_iv[..12], &mut length, &[])?;
277        // The echoed byte, the option, and no command.
278        let mut header = vec![self.response_auth, self.option, 0, 0];
279        let key = vmess_kdf_1_one_shot(&key, KDF_SALT_CONST_AEAD_RESP_HEADER_PAYLOAD_KEY);
280        let iv = vmess_kdf_1_one_shot(&iv, KDF_SALT_CONST_AEAD_RESP_HEADER_PAYLOAD_IV);
281        seal(&key[..16], &iv[..12], &mut header, &[])?;
282        length.extend_from_slice(&header);
283        Ok(length)
284    }
285}

287/// Reads the server's response for a request: `pending` holds what has
288/// been read so far. Returns how many bytes the response took, or None if
289/// more are needed.
290pub fn open_response(request: &RequestHeader, pending: &[u8]) -> io::Result<Option<usize>> {
291    const LEN_PART: usize = 2 + TAG_LEN;
292    if pending.len() < LEN_PART {
293        return Ok(None);
294    }
295    let (key, iv) = request.response_keys();
296    let len_key = vmess_kdf_1_one_shot(&key, KDF_SALT_CONST_AEAD_RESP_HEADER_LEN_KEY);
297    let len_iv = vmess_kdf_1_one_shot(&iv, KDF_SALT_CONST_AEAD_RESP_HEADER_LEN_IV);
298    let mut length = pending[..LEN_PART].to_vec();
299    open(&len_key[..16], &len_iv[..12], &mut length, &[])?;
300    let len = u16::from_be_bytes([length[0], length[1]]) as usize;
301    let total = LEN_PART + len + TAG_LEN;
302    if pending.len() < total {
303        return Ok(None);
304    }
305    let key = vmess_kdf_1_one_shot(&key, KDF_SALT_CONST_AEAD_RESP_HEADER_PAYLOAD_KEY);
306    let iv = vmess_kdf_1_one_shot(&iv, KDF_SALT_CONST_AEAD_RESP_HEADER_PAYLOAD_IV);
307    let mut header = pending[LEN_PART..total].to_vec();
308    open(&key[..16], &iv[..12], &mut header, &[])?;
309    if header.first() != Some(&request.response_auth) {
310        return Err(io::Error::new(
311            io::ErrorKind::InvalidData,
312            "vmess: response is not for this request",
313        ));
314    }
315    Ok(Some(total))
316}

318/// Auth IDs seen recently. An ID is good for the two minutes either side
319/// of its timestamp, so each is remembered for at least four: IDs go into
320/// a generation that is at most four minutes old, and are dropped with the
321/// one before it. A generation that fills up is rotated early, which
322/// bounds the memory at the cost of remembering less under a flood.
323pub struct ReplayFilter {
324    inner: Mutex<Generations>,
325}
326
327struct Generations {
328    current: HashSet<[u8; 16]>,
329    previous: HashSet<[u8; 16]>,
330    since: Instant,
331}
332
333const REPLAY_WINDOW: Duration = Duration::from_secs(2 * MAX_TIME_DIFFERENCE);
334const REPLAY_MAX_ENTRIES: usize = 1 << 20;
335
336impl Default for ReplayFilter {
337    fn default() -> Self {
338        ReplayFilter {
339            inner: Mutex::new(Generations {
340                current: HashSet::new(),
341                previous: HashSet::new(),
342                since: Instant::now(),
343            }),
344        }
345    }
346}
347
348impl ReplayFilter {
349    /// Whether `id` is new; it is remembered either way.
350    pub fn check(&self, id: &[u8; 16]) -> bool {
351        let mut g = self.inner.lock();
352        let age = g.since.elapsed();
353        if age >= 2 * REPLAY_WINDOW {
354            g.current.clear();
355            g.previous.clear();
356            g.since = Instant::now();
357        } else if age >= REPLAY_WINDOW || g.current.len() >= REPLAY_MAX_ENTRIES {
358            g.previous = std::mem::take(&mut g.current);
359            g.since = Instant::now();
360        }
361        if g.previous.contains(id) {
362            return false;
363        }
364        g.current.insert(*id)
365    }
366}

386/// The server's side of authentication: its users and the auth IDs they
387/// used.
388pub struct Authenticator<T> {
389    users: Vec<User<T>>,
390    replay: ReplayFilter,
391}
392
393/// Why a request was refused.
394#[derive(Debug, PartialEq, Eq)]
395pub enum AuthError {
396    UnknownUser,
397    BadTime,
398    Replayed,
399}
400
401impl<T> Authenticator<T> {
402    pub fn new(users: Vec<User<T>>) -> Self {
403        Authenticator {
404            users,
405            replay: ReplayFilter::default(),
406        }
407    }
408
409    /// The user an auth ID is from, if it is theirs, current and new.
410    pub fn authenticate(&self, auth_id: &[u8; 16], now: u64) -> Result<&User<T>, AuthError> {
411        for user in &self.users {
412            let mut id = *auth_id;
413            user.cipher.decrypt_block((&mut id).into());
414            let crc = u32::from_be_bytes([id[12], id[13], id[14], id[15]]);
415            if crc32fast::hash(&id[..12]) != crc {
416                continue;
417            }
418            let time = u64::from_be_bytes(id[..8].try_into().expect("8 bytes"));
419            if time.abs_diff(now) > MAX_TIME_DIFFERENCE {
420                return Err(AuthError::BadTime);
421            }
422            if !self.replay.check(auth_id) {
423                return Err(AuthError::Replayed);
424            }
425            return Ok(user);
426        }
427        Err(AuthError::UnknownUser)
428    }
429
430    /// Reads and opens a request: the user it is from and its header.
431    pub async fn read_request<R: AsyncRead + Unpin>(
432        &self,
433        r: &mut R,
434    ) -> io::Result<(&User<T>, RequestHeader)> {
435        let mut auth_id = [0u8; 16];
436        r.read_exact(&mut auth_id).await?;
437        let user = self.authenticate(&auth_id, now()).map_err(|e| {
438            io::Error::new(io::ErrorKind::PermissionDenied, format!("vmess: {:?}", e))
439        })?;
440        let mut length = vec![0u8; 2 + TAG_LEN];
441        r.read_exact(&mut length).await?;
442        let mut nonce = [0u8; 8];
443        r.read_exact(&mut nonce).await?;
444        let key = vmess_kdf_3_one_shot(
445            &user.cmd_key,
446            KDF_SALT_CONST_VMESS_HEADER_PAYLOAD_LENGTH_AEAD_KEY,
447            &auth_id,
448            &nonce,
449        );
450        let iv = vmess_kdf_3_one_shot(
451            &user.cmd_key,
452            KDF_SALT_CONST_VMESS_HEADER_PAYLOAD_LENGTH_AEAD_IV,
453            &auth_id,
454            &nonce,
455        );
456        open(&key[..16], &iv[..12], &mut length, &auth_id)?;
457        let len = u16::from_be_bytes([length[0], length[1]]) as usize;
458        if len > MAX_HEADER_LEN {
459            return Err(crypto_error());
460        }
461        let mut header = vec![0u8; len + TAG_LEN];
462        r.read_exact(&mut header).await?;
463        let key = vmess_kdf_3_one_shot(
464            &user.cmd_key,
465            KDF_SALT_CONST_VMESS_HEADER_PAYLOAD_AEAD_KEY,
466            &auth_id,
467            &nonce,
468        );
469        let iv = vmess_kdf_3_one_shot(
470            &user.cmd_key,
471            KDF_SALT_CONST_VMESS_HEADER_PAYLOAD_AEAD_IV,
472            &auth_id,
473            &nonce,
474        );
475        open(&key[..16], &iv[..12], &mut header, &auth_id)?;
476        Ok((user, RequestHeader::decode(&header)?))
477    }
478}
