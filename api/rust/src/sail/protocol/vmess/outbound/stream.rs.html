11/// What every request of an outbound shares.
12pub struct Client {
13    pub cmd_key: [u8; 16],
14    pub security: ClientSecurity,
15    pub global_padding: bool,
16}
17
18impl Client {
19    /// Sends a request for `command` and returns the body that follows.
20    pub async fn open(
21        &self,
22        mut stream: AnyStream,
23        command: u8,
24        address: Option<SocksAddr>,
25    ) -> io::Result<VmessStream<AnyStream>> {
26        let (security, option) = self.security.request(command, self.global_padding);
27        let request = RequestHeader::new(option, security, command, address);
28        stream.write_all(&request.seal(&self.cmd_key)?).await?;
29        VmessStream::client(stream, &request, command == COMMAND_UDP)
30    }
31}

33pub struct Handler {
34    pub address: String,
35    pub port: u16,
36    pub client: std::sync::Arc<Client>,
37}
38
39#[async_trait]
40impl OutboundStreamHandler for Handler {
41    fn connect_addr(&self) -> OutboundConnect {
42        OutboundConnect::Proxy(Network::Tcp, self.address.clone(), self.port)
43    }
44
45    async fn handle<'a>(
46        &'a self,
47        sess: &'a Session,
48        _lhs: Option<&mut AnyStream>,
49        stream: Option<AnyStream>,
50    ) -> io::Result<AnyStream> {
51        tracing::trace!("handling outbound stream");
52        let stream = stream.ok_or_else(|| io::Error::other("invalid input"))?;
53        let stream = self
54            .client
55            .open(stream, COMMAND_TCP, Some(sess.destination.clone()))
56            .await?;
57        Ok(Box::new(stream))
58    }
