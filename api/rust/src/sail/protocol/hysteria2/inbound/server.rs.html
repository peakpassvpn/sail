33/// Streams and sessions accepted and not yet taken.
34const ACCEPT_QUEUE: usize = 1024;
35/// How long a stream may wait for room in the accept queue.
36const ACCEPT_QUEUE_TIMEOUT: Duration = Duration::from_secs(5);
37/// UDP sessions one connection may have at once.
38const MAX_UDP_SESSIONS: usize = 1024;
39/// Packets queued for a UDP session before more are dropped.
40const UDP_SESSION_QUEUE: usize = 256;
41/// A UDP session without packets from the client for this long is closed.
42/// The client may go on using its ID; that starts a new session.
43const UDP_SESSION_IDLE: Duration = Duration::from_secs(300);

90#[async_trait]
91impl InboundDatagramHandler for DatagramHandler {
92    async fn handle<'a>(&'a self, socket: AnyInboundDatagram) -> io::Result<AnyInboundTransport> {
93        let socket = socket.into_std()?;
94        let local_addr = socket.local_addr()?;
95        let socket = quic::wrap_socket(socket, self.obfs.as_ref())?;
96        let endpoint = endpoint_on(socket, Some(self.server_config.clone()))?;
97        let (tx, rx) = mpsc::channel(ACCEPT_QUEUE);
98        let server = self.server.clone();
99        let server_config = self.server_config.clone();
100        tokio::spawn(async move {
101            loop {
102                let accept = std::pin::pin!(endpoint.accept());
103                let closed = std::pin::pin!(tx.closed());
104                let incoming = match futures::future::select(accept, closed).await {
105                    Either::Left((incoming, _)) => incoming,
106                    // Nobody takes what we accept any more.
107                    Either::Right(_) => None,
108                };
109                let Some(incoming) = incoming else {
110                    break;
111                };
112                let conn = Conn {
113                    server: server.clone(),
114                    tx: tx.clone(),
115                    local_addr,
116                };
117                let config = server_config.clone();
118                tokio::spawn(async move {
119                    let remote = incoming.remote_address();
120                    if let Err(e) = conn.serve(incoming, config).await {
121                        debug!("hysteria2 connection from {}: {}", remote, e);
122                    }
123                });
124            }
125            endpoint.close(0u32.into(), b"");
126        });
127        Ok(InboundTransport::Incoming(Box::new(Incoming(rx))))
128    }
129}

137impl Conn {
138    async fn serve(self, incoming: quinn::Incoming, config: quinn::ServerConfig) -> io::Result<()> {
139        // A controller of its own, which the authentication will set.
140        let congestion = CongestionHandle::default();
141        let mut config = config;
142        config.transport_config(Arc::new(quic::transport_config(
143            &self.server.tuning,
144            Side::Server,
145            &congestion,
146        )));
147        let conn = incoming
148            .accept_with(Arc::new(config))
149            .map_err(io::Error::other)?
150            .await
151            .map_err(io::Error::other)?;
152        trace!(
153            "hysteria2 accepted connection from {}",
154            conn.remote_address()
155        );
156        let _control = quic::open_control_stream(&conn).await?;
157        let drain = tokio::spawn(quic::drain_uni_streams(conn.clone()));
158
159        let state = Arc::new(ConnState {
160            server: self.server,
161            tx: self.tx,
162            local_addr: self.local_addr,
163            conn: conn.clone(),
164            congestion,
165            user: OnceLock::new(),
166            udp_started: AtomicBool::new(false),
167            sessions: Arc::new(UdpSessions::default()),
168        });
169        let result = loop {
170            match conn.accept_bi().await {
171                Ok((send, recv)) => {
172                    let state = state.clone();
173                    tokio::spawn(async move {
174                        if let Err(e) = state.handle_stream(send, recv).await {
175                            debug!("hysteria2 stream: {}", e);
176                        }
177                    });
178                }
179                Err(quinn::ConnectionError::ApplicationClosed(_))
180                | Err(quinn::ConnectionError::LocallyClosed)
181                | Err(quinn::ConnectionError::TimedOut) => break Ok(()),
182                Err(e) => break Err(io::Error::other(e)),
183            }
184        };
185        drain.abort();
186        result
187    }
188}

190struct ConnState {
191    server: Arc<Server>,
192    tx: mpsc::Sender<AnyBaseInboundTransport>,
193    local_addr: SocketAddr,
194    conn: quinn::Connection,
195    congestion: CongestionHandle,
196    /// Set once the connection authenticated: the user, by name.
197    user: OnceLock<Option<Arc<str>>>,
198    udp_started: AtomicBool,

202impl ConnState {
203    fn session(&self, network: Network) -> Session {
204        Session {
205            network,
206            source: self.conn.remote_address(),
207            local_addr: self.local_addr,
208            user: self.user.get().cloned().flatten(),
209            ..Default::default()
210        }
211    }
212
213    async fn handle_stream(
214        self: Arc<Self>,
215        mut send: quinn::SendStream,
216        mut recv: quinn::RecvStream,
217    ) -> io::Result<()> {
218        let handshake = self.server.handshake_timeout;
219        let ty = timeout(handshake, proto::read_varint(&mut recv))
220            .await
221            .map_err(|_| io::Error::new(io::ErrorKind::TimedOut, "stream handshake"))??;
222        if ty == proto::FRAME_TYPE_TCP_REQUEST {
223            if self.user.get().is_none() {
224                // Not a proxy connection; to a web server, a frame it does
225                // not know on a request stream is an error.
226                let _ = recv.stop(0u32.into());
227                let _ = send.reset(0u32.into());
228                return Err(io::Error::other("TCP request before authentication"));
229            }
230            let destination = timeout(handshake, proto::read_tcp_request(&mut recv))
231                .await
232                .map_err(|_| io::Error::new(io::ErrorKind::TimedOut, "TCP request"))??;
233            // Answered before the outbound connects: the reference client
234            // waits for the answer before it sends anything, and a failed
235            // connection is a closed stream all the same.
236            send.write_all(&proto::tcp_response(true, ""))
237                .await
238                .map_err(io::Error::other)?;
239            let mut sess = self.session(Network::Tcp);
240            sess.destination = destination;
241            sess.stream_id = Some(StreamId::U64(send.id().index()));
242            let stream = Box::new(QuicStream::new(send, recv));
243            return match timeout(
244                ACCEPT_QUEUE_TIMEOUT,
245                self.tx.send(BaseInboundTransport::Stream(stream, sess)),
246            )
247            .await
248            {
249                Ok(Ok(())) => Ok(()),
250                Ok(Err(_)) => Ok(()),
251                Err(_) => Err(io::Error::other("accept queue full")),
252            };
253        }
254
255        let fields = timeout(handshake, h3::read_headers(&mut recv, Some(ty)))
256            .await
257            .map_err(|_| io::Error::new(io::ErrorKind::TimedOut, "request headers"))??;
258        if is_auth_request(&fields) {
259            let password = h3::field(&fields, proto::HEADER_AUTH).unwrap_or_default();
260            if let Some(user) = self.server.users.get(password) {
261                return self.authenticated(user.clone(), &fields, send).await;
262            }
263        }
264        self.server.masquerade.serve(fields, send, recv).await
265    }
266
267    /// Answers a good authentication, and makes the connection a proxy
268    /// connection.
269    async fn authenticated(
270        self: &Arc<Self>,
271        user: Option<Arc<str>>,
272        fields: &[Field],
273        mut send: quinn::SendStream,
274    ) -> io::Result<()> {
275        let server = &self.server;
276        let _ = self.user.set(user);
277        // As the reference server: we send at what the client says it can
278        // receive, capped by what we may send at; not knowing, BBR.
279        let client_rx: u64 = h3::field(fields, proto::HEADER_CC_RX)
280            .and_then(|v| v.parse().ok())
281            .unwrap_or(0);
282        let tx = if server.ignore_client_bandwidth {
283            0
284        } else if server.send_bps > 0 && client_rx > server.send_bps {
285            server.send_bps
286        } else {
287            client_rx
288        };
289        if tx > 0 {
290            debug!("hysteria2 sends with brutal at {} bytes/s", tx);
291            self.congestion.set_brutal(tx);
292        } else {
293            self.congestion.set_bbr();
294        }
295        let rx = if server.ignore_client_bandwidth {
296            "auto".to_string()
297        } else {
298            server.recv_bps.to_string()
299        };
300        let padding = proto::padding(proto::AUTH_RESPONSE_PADDING);
301        let status = proto::STATUS_AUTH_OK.to_string();
302        h3::write_headers(
303            &mut send,
304            &[
305                (":status", &status),
306                (proto::HEADER_UDP, "true"),
307                (proto::HEADER_CC_RX, &rx),
308                (proto::HEADER_PADDING, &padding),
309            ],
310        )
311        .await?;
312        let _ = send.finish();
313        if !self.udp_started.swap(true, Ordering::Relaxed) {
314            tokio::spawn(self.clone().serve_datagrams());
315        }
316        Ok(())
317    }

319    /// Hands the datagrams of the connection to their sessions, starting a
320    /// session for an ID it has not seen or has closed.
321    async fn serve_datagrams(self: Arc<Self>) {
322        while let Ok(datagram) = self.conn.read_datagram().await {
323            let msg = match UdpMessage::decode(&datagram) {
324                Ok(msg) => msg,
325                Err(e) => {
326                    debug!("hysteria2: invalid UDP message: {}", e);
327                    continue;
328                }
329            };
330            let known = {
331                let sessions = self.sessions.lock();
332                if !sessions.contains_key(&msg.session_id) && sessions.len() >= MAX_UDP_SESSIONS {
333                    continue;
334                }
335                sessions.contains_key(&msg.session_id)
336            };
337            // Not under the lock: a session dropped at once takes it too.
338            if !known {
339                let Some(entry) = self.new_udp_session(msg.session_id) else {
340                    continue;
341                };
342                self.sessions.lock().insert(msg.session_id, entry);
343            }
344            let mut sessions = self.sessions.lock();
345            let Some(entry) = sessions.get_mut(&msg.session_id) else {
346                continue;
347            };
348            let Some(packet) = entry.defragger.feed(&msg) else {
349                continue;
350            };
351            let destination = match proto::parse_addr(msg.addr) {
352                Ok(destination) => destination,
353                Err(e) => {
354                    debug!("hysteria2: invalid UDP destination: {}", e);
355                    continue;
356                }
357            };
358            // A session that does not keep up loses packets, as UDP would;
359            // one that is gone is forgotten, and the next packet starts
360            // another.
361            if let Err(mpsc::error::TrySendError::Closed(_)) =
362                entry.tx.try_send((packet, destination))
363            {
364                sessions.remove(&msg.session_id);
365            }
366        }
367        // The connection is gone: so are its sessions, whose halves see
368        // the end at once instead of idling out.
369        self.sessions.lock().clear();
370    }

372    /// Starts the UDP session `id`, handing it on.
373    fn new_udp_session(&self, id: u32) -> Option<UdpSessionEntry> {
374        let (tx, rx) = mpsc::channel(UDP_SESSION_QUEUE);
375        let serial = self.sessions.next_serial.fetch_add(1, Ordering::Relaxed);
376        let mut sess = self.session(Network::Udp);
377        // Sessions from one client address are told apart by this.
378        sess.stream_id = Some(StreamId::Uuid(uuid::Uuid::new_v4()));
379        let datagram = Datagram {
380            rx,
381            source: DatagramSource::new(sess.source, sess.stream_id),
382            guard: SessionGuard {
383                sessions: self.sessions.clone(),
384                id,
385                serial,
386            },
387            sender: UdpSender {
388                conn: self.conn.clone(),
389                id,
390                next_packet: AtomicU32::new(0),
391            },
392        };
393        // Taken or not, never waited on: this is the datagram loop.
394        self.tx
395            .try_send(BaseInboundTransport::Datagram(
396                Box::new(datagram),
397                Some(sess),
398            ))
399            .ok()?;
400        Some(UdpSessionEntry {
401            tx,
402            defragger: Defragger::default(),
403            serial,
404        })
405    }

408fn is_auth_request(fields: &[Field]) -> bool {
409    h3::field(fields, ":method") == Some("POST")
410        && h3::field(fields, ":authority") == Some(proto::AUTH_HOST)
411        && h3::field(fields, ":path") == Some(proto::AUTH_PATH)
412}

429impl UdpSessions {
430    fn lock(&self) -> std::sync::MutexGuard<'_, HashMap<u32, UdpSessionEntry>> {
431        self.map.lock().unwrap_or_else(|e| e.into_inner())

435/// Closes the session when its receiving half goes: packets for the ID
436/// then start a new one.
437struct SessionGuard {
438    sessions: Arc<UdpSessions>,
439    id: u32,
440    serial: u64,
441}
442
443impl Drop for SessionGuard {
444    fn drop(&mut self) {
445        let mut sessions = self.sessions.lock();
446        if sessions
447            .get(&self.id)
448            .is_some_and(|e| e.serial == self.serial)
449        {
450            sessions.remove(&self.id);
451        }
452    }
453}

455struct UdpSender {
456    conn: quinn::Connection,
457    id: u32,
458    next_packet: AtomicU32,
459}
460
461impl UdpSender {
462    fn send(&self, payload: &[u8], from: &SocksAddr) -> io::Result<()> {
463        let max = self
464            .conn
465            .max_datagram_size()
466            .ok_or_else(|| io::Error::other("hysteria2: the client takes no datagrams"))?;
467        let packet_id = self.next_packet.fetch_add(1, Ordering::Relaxed) as u16;
468        let addr = proto::format_addr(from);
469        for fragment in proto::fragment(self.id, packet_id, &addr, payload, max)? {
470            self.conn
471                .send_datagram(Bytes::from(fragment))
472                .map_err(io::Error::other)?;
473        }
474        Ok(())
475    }
476}

508struct DatagramRecvHalf {
509    rx: mpsc::Receiver<Packet>,
510    source: DatagramSource,
511    _guard: SessionGuard,
512}
513
514#[async_trait]
515impl InboundDatagramRecvHalf for DatagramRecvHalf {
516    async fn recv_from(
517        &mut self,
518        buf: &mut [u8],
519    ) -> ProxyResult<(usize, DatagramSource, SocksAddr)> {
520        let (packet, destination) = match timeout(UDP_SESSION_IDLE, self.rx.recv()).await {
521            Ok(Some(p)) => p,
522            Ok(None) => return Err(ProxyError::DatagramFatal(anyhow!("session closed"))),
523            Err(_) => return Err(ProxyError::DatagramFatal(anyhow!("session idle"))),
524        };
525        if packet.len() > buf.len() {
526            return Err(ProxyError::DatagramWarn(anyhow!(
527                "UDP packet of {} bytes, buffer of {}",
528                packet.len(),
529                buf.len()
530            )));
531        }
532        buf[..packet.len()].copy_from_slice(&packet);
533        Ok((packet.len(), self.source.clone(), destination))
534    }
535}

680    #[tokio::test]
681    async fn a_proxy_masquerade_passes_strangers_to_the_site_behind() {
682        use tokio::io::{AsyncReadExt, AsyncWriteExt};
683        let site = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
684        let site_addr = site.local_addr().unwrap();
685        let (seen_tx, seen_rx) = tokio::sync::oneshot::channel();
686        tokio::spawn(async move {
687            let (mut stream, _) = site.accept().await.unwrap();
688            let mut request = Vec::new();
689            let mut buf = [0u8; 1024];
690            while !request.ends_with(b"\r\n\r\n") {
691                let n = stream.read(&mut buf).await.unwrap();
692                request.extend_from_slice(&buf[..n]);
693            }
694            stream
695                .write_all(b"HTTP/1.0 200 OK\r\nX-Up: 1\r\nConnection: close\r\n\r\nupstream")
696                .await
697                .unwrap();
698            let _ = seen_tx.send(String::from_utf8(request).unwrap());
699        });
700        let masquerade =
701            Masquerade::new(MasqueradeOptions::Url(format!("http://{}/base", site_addr))).unwrap();
702        let f = serve(masquerade).await;
703        let conn = f.connect().await;
704        let (headers, body) = request(&conn, &GET).await;
705        assert_eq!(h3::field(&headers, ":status"), Some("200"));
706        assert_eq!(h3::field(&headers, "x-up"), Some("1"));
707        assert_eq!(h3::field(&headers, "connection"), None);
708        assert_eq!(body, b"upstream");
709        let seen = seen_rx.await.unwrap();
710        assert!(seen.starts_with("GET /base/ HTTP/1.0\r\n"), "{}", seen);
711        // The host asked for, as rewrite_host is off.
712        assert!(seen.contains("host: example.com\r\n"), "{}", seen);
713    }

715    #[tokio::test]
716    async fn an_authenticated_connection_proxies_tcp_as_the_user() {
717        let mut f = serve(Masquerade::NotFound).await;
718        let conn = f.connect().await;
719        let (headers, _) = request(&conn, &auth("pw")).await;
720        assert_eq!(h3::field(&headers, ":status"), Some("233"));
721        assert_eq!(h3::field(&headers, "hysteria-udp"), Some("true"));
722        assert_eq!(h3::field(&headers, "hysteria-cc-rx"), Some("0"));
723
724        let (mut send, mut recv) = conn.open_bi().await.unwrap();
725        let destination = SocksAddr::try_from(("example.com", 443)).unwrap();
726        send.write_all(&proto::tcp_request(&destination, b"early"))
727            .await
728            .unwrap();
729        proto::read_tcp_response(&mut recv).await.unwrap();
730        let Some(BaseInboundTransport::Stream(mut stream, sess)) = f.incoming.next().await else {
731            panic!("no stream");
732        };
733        assert_eq!(sess.destination, destination);
734        assert_eq!(sess.user.as_deref(), Some("alice"));
735        let mut buf = [0u8; 5];
736        tokio::io::AsyncReadExt::read_exact(&mut stream, &mut buf)
737            .await
738            .unwrap();
739        assert_eq!(&buf, b"early");
740
741        // A UDP message starts a session.
742        let mut msg = bytes::BytesMut::new();
743        UdpMessage {
744            session_id: 7,
745            packet_id: 0,
746            fragment_id: 0,
747            fragment_count: 1,
748            addr: "1.2.3.4:53",
749            payload: b"query",
750        }
751        .encode(&mut msg);
752        conn.send_datagram(msg.freeze()).unwrap();
753        let Some(BaseInboundTransport::Datagram(datagram, Some(sess))) = f.incoming.next().await
754        else {
755            panic!("no datagram");
756        };
757        assert_eq!(sess.user.as_deref(), Some("alice"));
758        let (mut r, _s) = datagram.split();
759        let mut buf = [0u8; 64];
760        let (n, _, destination) = r.recv_from(&mut buf).await.ok().unwrap();
761        assert_eq!(&buf[..n], b"query");
762        assert_eq!(destination.to_string(), "1.2.3.4:53");
763    }
