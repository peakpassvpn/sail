12pub struct Handler {
13    pub address: String,
14    pub port: u16,
15    pub password: String,
16}
17
18#[async_trait]
19impl OutboundDatagramHandler for Handler {
20    fn connect_addr(&self) -> OutboundConnect {
21        OutboundConnect::Proxy(Network::Tcp, self.address.clone(), self.port)
22    }
23
24    fn transport_type(&self) -> DatagramTransportType {
25        DatagramTransportType::Reliable
26    }
27
28    async fn handle<'a>(
29        &'a self,
30        sess: &'a Session,
31        transport: Option<AnyOutboundTransport>,
32    ) -> io::Result<AnyOutboundDatagram> {
33        tracing::trace!("handling outbound datagram");
34        let stream = if let Some(OutboundTransport::Stream(stream)) = transport {
35            stream
36        } else {
37            return Err(io::Error::other("invalid input"));
38        };
39        let mut buf = BytesMut::new();
40        let password = Sha224::digest(self.password.as_bytes());
41        let password = hex::encode(&password[..]);
42        buf.put_slice(password.as_bytes());
43        buf.put_slice(b"\r\n");
44        buf.put_u8(0x03); // udp
45        sess.destination
46            .write_buf(&mut buf, SocksAddrWireType::PortLast);
47        buf.put_slice(b"\r\n");
48
49        let destination = match &sess.destination {
50            SocksAddr::Domain(domain, port) => Some(SocksAddr::Domain(domain.to_owned(), *port)),
51            _ => None,
52        };
53
54        Ok(Box::new(Datagram {
55            stream,
56            destination,
57            head: Some(buf),
58        }))
59    }
60}

62pub struct Datagram<S> {
63    stream: S,
64    destination: Option<SocksAddr>,
65    head: Option<BytesMut>,
66}
67
68impl<S> OutboundDatagram for Datagram<S>
69where
70    S: 'static + AsyncRead + AsyncWrite + Unpin + Send + Sync,
71{
72    fn split(
73        self: Box<Self>,
74    ) -> (
75        Box<dyn OutboundDatagramRecvHalf>,
76        Box<dyn OutboundDatagramSendHalf>,
77    ) {
78        let (r, w) = tokio::io::split(self.stream);
79        (
80            Box::new(DatagramRecvHalf(r, self.destination)),
81            Box::new(DatagramSendHalf(w, self.head)),
82        )
83    }
84}

88#[async_trait]
89impl<T> OutboundDatagramRecvHalf for DatagramRecvHalf<T>
90where
91    T: AsyncRead + AsyncWrite + Send + Sync,
92{
93    async fn recv_from(&mut self, buf: &mut [u8]) -> io::Result<(usize, SocksAddr)> {
94        let addr = SocksAddr::read_from(&mut self.0, SocksAddrWireType::PortLast).await?;
95        let mut buf2 = [0; 4];
96        self.0.read_exact(&mut buf2).await?;
97        let payload_len = u16::from_be_bytes([buf2[0], buf2[1]]) as usize;
98        // TODO Check CLRF?
99        if buf.len() < payload_len {
100            return Err(io::Error::new(io::ErrorKind::Interrupted, "Small buffer"));
101        }
102        self.0.read_exact(&mut buf[..payload_len]).await?;
103        // If the initial destination is of domain type, we return that
104        // domain address instead of the real source address. That also
105        // means we assume all received packets are comming from a same
106        // address.
107        if let Some(dest) = &self.1 {
108            trace!(
109                "trojan outbound received UDP {} bytes from {}",
110                payload_len,
111                dest
112            );
113            Ok((payload_len, dest.clone()))
114        } else {
115            trace!(
116                "trojan outbound received UDP {} bytes from {}",
117                payload_len,
118                &addr
119            );
120            Ok((payload_len, addr))
121        }
122    }
123}

127#[async_trait]
128impl<T> OutboundDatagramSendHalf for DatagramSendHalf<T>
129where
130    T: AsyncRead + AsyncWrite + Send + Sync,
131{
132    async fn send_to(&mut self, buf: &[u8], target: &SocksAddr) -> io::Result<usize> {
133        trace!("trojan outbound send UDP {} bytes to {}", buf.len(), target);
134        let mut data = BytesMut::new();
135        target.write_buf(&mut data, SocksAddrWireType::PortLast);
136        data.put_u16(buf.len() as u16);
137        data.put_slice(b"\r\n");
138        data.put_slice(buf);
139
140        // Writes the header along with the first payload.
141        if self.1.is_some() {
142            if let Some(mut head) = self.1.take() {
143                head.extend_from_slice(&data);
144                return self.0.write_all(&head).map_ok(|_| buf.len()).await;
145            }
146        }
147
148        self.0.write_all(&data).map_ok(|_| buf.len()).await
149    }
150
151    async fn close(&mut self) -> io::Result<()> {
152        self.0.shutdown().await
153    }
