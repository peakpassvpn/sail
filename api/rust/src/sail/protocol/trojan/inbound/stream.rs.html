53#[async_trait]
54impl<T> InboundDatagramRecvHalf for DatagramRecvHalf<T>
55where
56    T: AsyncRead + Send + Sync + Unpin,
57{
58    async fn recv_from(
59        &mut self,
60        buf: &mut [u8],
61    ) -> ProxyResult<(usize, DatagramSource, SocksAddr)> {
62        let dst_addr = SocksAddr::read_from(&mut self.0, SocksAddrWireType::PortLast)
63            .map_err(|e| ProxyError::DatagramFatal(e.into()))
64            .await?;
65        let mut buf2 = [0; 4];
66        self.0
67            .read_exact(&mut buf2)
68            .map_err(|e| ProxyError::DatagramFatal(e.into()))
69            .await?;
70        let payload_len = u16::from_be_bytes([buf2[0], buf2[1]]) as usize;
71        if buf.len() < payload_len {
72            return Err(ProxyError::DatagramFatal(anyhow!("Small buffer")));
73        }
74        // TODO Check CRLF?
75        self.0
76            .read_exact(&mut buf[..payload_len])
77            .map_err(|e| ProxyError::DatagramFatal(e.into()))
78            .await?;
79        trace!(
80            "trojan inbound received UDP {} bytes for {}",
81            payload_len,
82            &dst_addr
83        );
84        Ok((payload_len, self.1.clone(), dst_addr))
85    }
86}

98        src_addr: &SocksAddr,
99        _dst_addr: &SocketAddr,
100    ) -> io::Result<usize> {
101        trace!(
102            "trojan inbound send UDP {} bytes for {}",
103            buf.len(),
104            &src_addr
105        );
106        let mut data = BytesMut::new();
107        src_addr.write_buf(&mut data, SocksAddrWireType::PortLast);
108        data.put_u16(buf.len() as u16);
109        data.put_slice(b"\r\n");
110        data.put_slice(buf);
111        self.0.write_all(&data).map_ok(|_| buf.len()).await
112    }
113
114    async fn close(&mut self) -> io::Result<()> {
115        self.0.shutdown().await
116    }

119/// The first bytes of a Trojan request: the password's SHA-224 in lowercase
120/// hex, then CRLF.
121const KEY_LEN: usize = 56;
122const AUTH_LEN: usize = KEY_LEN + 2;
123
124/// Whether `buf`, the first bytes of a connection, can still be the start of
125/// a Trojan request. Checked as bytes arrive, so that a connection that is
126/// plainly something else goes to the fallback without waiting for more.
127fn could_be_auth(buf: &[u8]) -> bool {
128    buf.iter().enumerate().all(|(i, &b)| match i {
129        i if i < KEY_LEN => b.is_ascii_digit() || (b'a'..=b'f').contains(&b),
130        KEY_LEN => b == b'\r',
131        _ => b == b'\n',
132    })
133}

135pub struct Handler {
136    /// The users by the key their password makes, with their names.
137    keys: HashMap<Vec<u8>, Option<std::sync::Arc<str>>>,
138    /// Where what fails to authenticate goes; closed without one.
139    fallback: Option<Fallback>,
140}
141
142impl Handler {
143    /// Takes the users as their passwords and names.
144    pub fn new(users: Vec<(String, Option<String>)>, fallback: Option<Fallback>) -> Self {
145        let mut keys = HashMap::new();
146        for (pass, name) in users {
147            let key = Sha224::digest(pass.as_bytes());
148            let key = hex::encode(&key[..]);
149            keys.insert(key.as_bytes().to_vec(), name.map(Into::into));
150        }
151        Handler { keys, fallback }
152    }
153}
154
155#[async_trait]
156impl InboundStreamHandler for Handler {
157    async fn handle<'a>(
158        &'a self,
159        mut sess: Session,
160        mut stream: AnyStream,
161    ) -> std::io::Result<AnyInboundTransport> {
162        tracing::trace!("handling inbound stream");
163        // The key and its CRLF, and no more: what is read here is what the
164        // fallback is given if the key is not a user's.
165        let mut auth = [0u8; AUTH_LEN];
166        let mut read = 0;
167        let reading = async {
168            while read < AUTH_LEN {
169                let n = stream.read(&mut auth[read..]).await?;
170                if n == 0 {
171                    return Ok(false);
172                }
173                read += n;
174                if !could_be_auth(&auth[..read]) {
175                    return Ok(false);
176                }
177            }
178            Ok::<_, io::Error>(true)
179        };
180        let complete = match self.fallback {
181            // A peer that sends part of a key and waits is not a client.
182            Some(_) => timeout(HEADER_TIMEOUT, reading)
183                .await
184                .unwrap_or(Ok(false))?,
185            None => reading.await?,
186        };
187        let user = match complete {
188            true => self.keys.get(&auth[..KEY_LEN]).ok_or("unknown password"),
189            false => Err("not a Trojan request"),
190        };
191        let user = match user {
192            Ok(user) => user,
193            Err(why) => {
194                return Err(match &self.fallback {
195                    Some(fallback) => fallback.relay(&sess, stream, auth[..read].to_vec(), why),
196                    None => io::Error::new(io::ErrorKind::PermissionDenied, why),
197                })
198            }
199        };
200        sess.user = user.clone();
201        let cmd = stream.read_u8().await?;
202        let dst_addr = SocksAddr::read_from(&mut stream, SocksAddrWireType::PortLast).await?;
203        sess.destination = dst_addr;
204        let mut crlf = [0u8; 2];
205        stream.read_exact(&mut crlf).await?;
206        if crlf != *b"\r\n" {
207            return Err(io::Error::other("invalid request"));
208        }
209        match cmd {
210            // tcp
211            0x01 => Ok(InboundTransport::Stream(stream, sess)),
212            // udp
213            0x03 => {
214                sess.network = Network::Udp;
215                Ok(InboundTransport::Datagram(
216                    Box::new(Datagram::new(
217                        stream,
218                        DatagramSource::new(sess.source, sess.stream_id),
219                    )),
220                    Some(sess),
221                ))
222            }
223            _ => Err(io::Error::other("invalid command")),
224        }
225    }
226}

228#[cfg(test)]
229mod tests {
230    use super::*;
231
232    #[test]
233    fn test_could_be_auth() {
234        let key = hex::encode(Sha224::digest(b"password"));
235        let mut auth = key.clone().into_bytes();
236        auth.extend_from_slice(b"\r\n");
237        assert!(could_be_auth(&auth));
238        assert!(could_be_auth(&auth[..10]));
239        assert!(could_be_auth(b""));
240        assert!(!could_be_auth(b"GET / HTTP/1.1\r\n"));
241        assert!(!could_be_auth(key.to_uppercase().as_bytes()));
242        let mut bad_crlf = key.into_bytes();
243        bad_crlf.extend_from_slice(b"\n\r");
244        assert!(!could_be_auth(&bad_crlf));
245    }
