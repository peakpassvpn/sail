36    fn evict_oldest(&mut self) -> bool {
37        while let Some(oldest) = self.insertion_order.pop_front() {

40            }

95#[derive(Debug)]
96struct SharedIngress {
97    generation: RwLock<NetworkGeneration>,
98    shard_count: u16,
99    hasher: RandomState,
100    ledger: Arc<ResourceLedger>,
101    directory: Vec<Mutex<DirectoryStripe>>,
102    max_directory_entries_per_stripe: usize,
103    queues: Vec<Mutex<QueueState>>,
104    local_caches: Vec<Arc<Mutex<LocalRouteCache>>>,
105    cache_pressure_active: AtomicBool,
106    counters: ConcurrentCounters,
107}

109impl SharedIngress {
110    fn hashed_owner(&self, key: FlowKey) -> io::Result<(ShardId, u64)> {
111        let hash = self.hasher.hash_one(key);
112        let owner = ShardId::new(
113            u16::try_from(hash % u64::from(self.shard_count))
114                .map_err(|_| io::Error::other("shard hash exceeded u16"))?,
115        );
116        Ok((owner, hash))
117    }
118
119    /// The shard that packets of `endpoint` go to under the current
120    /// generation. Ownership is the stable hash alone; the directory only
121    /// caches it.
122    fn endpoint_owner(&self, endpoint: IpEndpoint) -> io::Result<ShardId> {
123        let generation = *self
124            .generation
125            .read()
126            .map_err(|_| io::Error::other("network generation lock poisoned"))?;
127        self.hashed_owner(FlowKey {
128            endpoint,
129            generation,
130        })
131        .map(|(owner, _)| owner)
132    }
133
134    fn owner(&self, key: FlowKey) -> io::Result<(ShardId, u64)> {
135        let (owner, hash) = self.hashed_owner(key)?;
136        if self.cache_pressure_active.load(Ordering::Acquire) {
137            return Ok((owner, hash));
138        }
139        let stripe_index = usize::try_from(hash % DIRECTORY_STRIPES as u64)
140            .map_err(|_| io::Error::other("directory stripe exceeded usize"))?;
141        let mut stripe = self.directory[stripe_index]
142            .lock()
143            .map_err(|_| io::Error::other("flow directory lock poisoned"))?;
144        if self.cache_pressure_active.load(Ordering::Acquire) {
145            return Ok((owner, hash));
146        }
147        if let Some(entry) = stripe.entries.get(&key) {
148            return Ok((entry.owner, hash));
149        }
150        // Closed flows never remove their entries, so the directory is a
151        // bounded FIFO cache like the `ShardRouter` model; without the bound it
152        // grew until metadata pressure cleared every routing cache.
153        if stripe.entries.len() >= self.max_directory_entries_per_stripe && stripe.evict_oldest() {
154            self.counters
155                .directory_entries
156                .fetch_sub(1, Ordering::Relaxed);
157        }
158        let mut lease = self
159            .ledger
160            .try_acquire(ResourceKind::MetadataBytes, DIRECTORY_ENTRY_CHARGE);
161        if lease.is_err() && stripe.evict_oldest() {
162            self.counters
163                .directory_entries
164                .fetch_sub(1, Ordering::Relaxed);
165            lease = self
166                .ledger
167                .try_acquire(ResourceKind::MetadataBytes, DIRECTORY_ENTRY_CHARGE);
168        }
169        if let Ok(lease) = lease {
170            stripe.entries.insert(
171                key,
172                DirectoryEntry {
173                    owner,
174                    _lease: lease,
175                },
176            );
177            stripe.insertion_order.push_back(key);
178            self.counters
179                .directory_entries
180                .fetch_add(1, Ordering::Relaxed);
181        } else {
182            atomic_add_counter(&self.counters.directory_admission_failures, 1);
183        }
184        Ok((owner, hash))
185    }
186
187    fn enqueue(
188        &self,
189        target: ShardId,
190        hash: u64,
191        packet: Packet,
192        class: PacketClass,
193    ) -> io::Result<()> {
194        let bytes = packet.payload().len();
195        let packet_kind = match class {
196            PacketClass::Control => ResourceKind::ControlPacketBytes,
197            PacketClass::Data => ResourceKind::PacketBytes,
198        };
199        let Ok(packet_lease) = self.ledger.try_acquire(packet_kind, bytes) else {
200            atomic_add_counter(&self.counters.dropped_packets, 1);
201            return Ok(());
202        };
203        let Ok(metadata_lease) = self
204            .ledger
205            .try_acquire(ResourceKind::MetadataBytes, FORWARDED_ENTRY_CHARGE)
206        else {
207            atomic_add_counter(&self.counters.dropped_packets, 1);
208            return Ok(());
209        };
210        let forwarded = ForwardedPacket {
211            packet,
212            _packet_lease: packet_lease,
213            _metadata_lease: metadata_lease,
214        };
215        let mut queue = self.queues[target.index()]
216            .lock()
217            .map_err(|_| io::Error::other("shard queue lock poisoned"))?;
218        let work_class = match class {
219            PacketClass::Control => WorkClass::Control,
220            PacketClass::Data => WorkClass::Data {
221                flow: FlowId::new(hash),
222                weight: 1,
223            },
224        };
225        match queue.scheduler.enqueue(work_class, bytes, forwarded) {
226            Ok(()) => {
227                atomic_add_counter(&self.counters.enqueued_packets, 1);
228                atomic_add_counter(&self.counters.enqueued_bytes, bytes);
229                if class == PacketClass::Control {
230                    atomic_add_counter(&self.counters.forwarded_control_packets, 1);
231                }
232                let wake = queue.waker.take();
233                drop(queue);
234                if let Some(waker) = wake {
235                    atomic_add_counter(&self.counters.forwarded_wakeups, 1);
236                    waker.wake();
237                }
238            }
239            Err(_) => {
240                atomic_add_counter(&self.counters.dropped_packets, 1);
241            }
242        }
243        Ok(())
244    }
245
246    fn drain(&self, owner: ShardId, out: &mut PacketBatch) -> io::Result<usize> {
247        let capacity = out.limit().saturating_sub(out.len());
248        if capacity == 0 {
249            return Ok(0);
250        }
251        let mut queue = self.queues[owner.index()]
252            .lock()
253            .map_err(|_| io::Error::other("shard queue lock poisoned"))?;
254        let round = queue
255            .scheduler
256            .run_round_limited(capacity, usize::MAX, |_, forwarded| {
257                out.push(forwarded.packet)
258                    .expect("limited drain must fit the destination batch");
259            });
260        atomic_add_counter(&self.counters.drained_packets, round.packets);
261        atomic_add_counter(&self.counters.drained_bytes, round.bytes);
262        Ok(round.packets)
263    }
264
265    fn poll_forwarded(&self, owner: ShardId, waker: &Waker) -> io::Result<bool> {
266        let mut queue = self.queues[owner.index()]
267            .lock()
268            .map_err(|_| io::Error::other("shard queue lock poisoned"))?;
269        if queue.scheduler.snapshot().queued_packets > 0 {
270            return Ok(true);
271        }
272        if queue
273            .waker
274            .as_ref()
275            .is_none_or(|registered| !registered.will_wake(waker))
276        {
277            queue.waker = Some(waker.clone());
278            atomic_add_counter(&self.counters.forwarded_waker_registrations, 1);
279        }
280        Ok(false)
281    }
282
283    fn clear_waker(&self, owner: ShardId) {
284        if let Ok(mut queue) = self.queues[owner.index()].lock() {
285            queue.waker = None;
286        }
287    }

289    fn observe_pressure(&self) -> io::Result<()> {
290        let pressure = self.ledger.snapshot().pressure;
291        if matches!(pressure, PressureLevel::Critical | PressureLevel::Exhausted) {
292            if self
293                .cache_pressure_active
294                .compare_exchange(false, true, Ordering::AcqRel, Ordering::Acquire)
295                .is_ok()
296            {
297                self.reclaim_routing_caches()?;
298            }
299        } else if pressure == PressureLevel::Normal {
300            self.cache_pressure_active.store(false, Ordering::Release);
301        }
302        Ok(())
303    }
304
305    fn reclaim_routing_caches(&self) -> io::Result<()> {
306        let mut directory_entries = 0_usize;
307        for stripe in &self.directory {
308            let mut stripe = stripe
309                .lock()
310                .map_err(|_| io::Error::other("flow directory lock poisoned"))?;
311            directory_entries = directory_entries.saturating_add(stripe.entries.len());
312            stripe.entries.clear();
313            stripe.insertion_order.clear();
314        }
315        let mut local_cache_entries = 0_usize;
316        for cache in &self.local_caches {
317            let mut cache = cache
318                .lock()
319                .map_err(|_| io::Error::other("local route cache lock poisoned"))?;
320            local_cache_entries = local_cache_entries.saturating_add(cache.entries.len());
321            cache.entries.clear();
322            cache.insertion_order.clear();
323        }
324        self.counters.directory_entries.store(0, Ordering::Release);
325        atomic_add_counter(&self.counters.pressure_cache_reclaims, 1);
326        atomic_add_counter(
327            &self.counters.pressure_reclaimed_directory_entries,
328            directory_entries,
329        );
330        atomic_add_counter(
331            &self.counters.pressure_reclaimed_local_cache_entries,
332            local_cache_entries,
333        );
334        Ok(())
335    }
336
337    fn reset_network(&self, generation: NetworkGeneration) -> io::Result<()> {
338        let mut current_generation = self
339            .generation
340            .write()
341            .map_err(|_| io::Error::other("network generation lock poisoned"))?;
342        for stripe in &self.directory {
343            let mut stripe = stripe
344                .lock()
345                .map_err(|_| io::Error::other("flow directory lock poisoned"))?;
346            stripe.entries.clear();
347            stripe.insertion_order.clear();
348        }
349        for queue in &self.queues {
350            queue
351                .lock()
352                .map_err(|_| io::Error::other("shard queue lock poisoned"))?
353                .scheduler
354                .clear();
355        }
356        for cache in &self.local_caches {
357            let mut cache = cache
358                .lock()
359                .map_err(|_| io::Error::other("local route cache lock poisoned"))?;
360            cache.entries.clear();
361            cache.insertion_order.clear();
362        }
363        self.counters.directory_entries.store(0, Ordering::Relaxed);
364        *current_generation = generation;
365        Ok(())
366    }

368    fn stats(&self) -> io::Result<ShardRouterStats> {
369        let mut queued_packets = 0_usize;
370        let mut queued_bytes = 0_usize;
371        for queue in &self.queues {
372            let queue = queue
373                .lock()
374                .map_err(|_| io::Error::other("shard queue lock poisoned"))?;
375            let snapshot = queue.scheduler.snapshot();
376            queued_packets = queued_packets.saturating_add(snapshot.queued_packets);
377            queued_bytes = queued_bytes.saturating_add(snapshot.queued_bytes);
378        }
379        let mut local_cache_entries = 0_usize;
380        for cache in &self.local_caches {
381            local_cache_entries = local_cache_entries.saturating_add(
382                cache
383                    .lock()
384                    .map_err(|_| io::Error::other("local route cache lock poisoned"))?
385                    .entries
386                    .len(),
387            );
388        }
389        Ok(ShardRouterStats {
390            directory_entries: self.counters.directory_entries.load(Ordering::Relaxed),
391            local_cache_entries,
392            queued_packets,
393            queued_bytes,
394            pressure_cache_reclaims: to_u64(
395                self.counters
396                    .pressure_cache_reclaims
397                    .load(Ordering::Relaxed),
398            ),
399            pressure_reclaimed_directory_entries: to_u64(
400                self.counters
401                    .pressure_reclaimed_directory_entries
402                    .load(Ordering::Relaxed),
403            ),
404            pressure_reclaimed_local_cache_entries: to_u64(
405                self.counters
406                    .pressure_reclaimed_local_cache_entries
407                    .load(Ordering::Relaxed),
408            ),
409            directory_admission_failures: to_u64(
410                self.counters
411                    .directory_admission_failures
412                    .load(Ordering::Relaxed),
413            ),
414            local_cache_hits: to_u64(self.counters.local_cache_hits.load(Ordering::Relaxed)),
415            local_cache_misses: to_u64(self.counters.local_cache_misses.load(Ordering::Relaxed)),
416            local_cache_admission_failures: to_u64(

421            local_routes: to_u64(self.counters.local_routes.load(Ordering::Relaxed)),
422            forwarded_routes: to_u64(self.counters.forwarded_routes.load(Ordering::Relaxed)),
423            forwarded_control_packets: to_u64(

464impl ShardedPacketIoControl {
465    /// Clears flow ownership and queued cross-shard packets, then installs the
466    /// generation used to classify future input.
467    ///
468    /// # Errors
469    ///
470    /// Returns an error when a directory, queue, or generation lock is
471    /// poisoned.
472    pub fn reset_network(&self, generation: NetworkGeneration) -> io::Result<()> {
473        self.shared.reset_network(generation)
474    }
475
476    /// Returns current queue depth and cumulative routing counters.
477    ///
478    /// # Errors
479    ///
480    /// Returns an error when a shard queue lock is poisoned.
481    pub fn stats(&self) -> io::Result<ShardRouterStats> {
482        self.shared.stats()
483    }
484
485    /// The shard that owns the flow of `endpoint`, as packets arriving from
486    /// the network are routed. Opening a flow on it lets its replies return.
487    ///
488    /// # Errors
489    ///
490    /// Returns an error for a poisoned generation lock.
491    pub fn owner(&self, endpoint: IpEndpoint) -> io::Result<ShardId> {
492        self.shared.endpoint_owner(endpoint)
493    }
494
495    /// Predicts the owner selected by the group's stable hash without creating
496    /// directory state. Platform RSS setup and tests may use this as an
497    /// affinity hint; correctness still relies on bounded forwarding.
498    ///
499    /// # Errors
500    ///
501    /// Returns an error for malformed input or a poisoned generation lock.
502    pub fn preferred_owner(&self, packet: &[u8]) -> io::Result<ShardId> {
503        let generation = self
504            .shared
505            .generation
506            .read()
507            .map_err(|_| io::Error::other("network generation lock poisoned"))?;
508        let key = classify_packet(packet, *generation)
509            .map_err(|error| io::Error::new(io::ErrorKind::InvalidData, error))?;
510        self.shared.hashed_owner(key).map(|(owner, _)| owner)
511    }
512}

514/// One platform queue wrapped with striped stable-flow routing and bounded
515/// forwarding to the owner shard. A
516/// [`SingleShardRunner`](crate::SingleShardRunner) owns each returned adapter.
517#[derive(Debug)]
518pub struct ShardedPacketIo<I> {
519    inner: I,
520    shard: ShardId,
521    capabilities: PacketCapabilities,
522    shared: Arc<SharedIngress>,
523    local_cache: Arc<Mutex<LocalRouteCache>>,
524}

526#[derive(Debug)]
527pub enum ShardedPacketIoError {
528    InvalidQueueCount,
529    Capabilities(io::Error),
530    Router(ShardRouterError),
531}
532
533impl fmt::Display for ShardedPacketIoError {
534    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
535        match self {
536            Self::InvalidQueueCount => formatter.write_str(
537                "platform queue group must be non-zero, fit u16, and contain compatible queue capabilities",
538            ),
539            Self::Capabilities(error) => write!(formatter, "invalid packet capabilities: {error}"),
540            Self::Router(error) => write!(formatter, "invalid shard router: {error}"),
541        }
542    }
543}

545impl std::error::Error for ShardedPacketIoError {}

547impl<I: PacketIo> ShardedPacketIo<I> {
548    /// Wraps a complete set of platform queues and returns one adapter per
549    /// shard plus their reset/stats control handle.
550    ///
551    /// # Errors
552    ///
553    /// Returns an error for an empty/oversized group, inconsistent platform
554    /// queue capabilities, or an invalid forwarding scheduler.
555    pub fn group(
556        queues: Vec<I>,
557        ledger: Arc<ResourceLedger>,
558        generation: NetworkGeneration,
559        scheduler: SchedulerConfig,
560    ) -> Result<(Vec<Self>, ShardedPacketIoControl), ShardedPacketIoError> {
561        let queue_count = queues.len();
562        let shard_count = u16::try_from(queue_count)
563            .ok()
564            .filter(|count| *count > 0)
565            .ok_or(ShardedPacketIoError::InvalidQueueCount)?;
566        let mut capabilities = Vec::with_capacity(queue_count);
567        for queue in &queues {
568            let capability = queue
569                .capabilities()
570                .validate()
571                .map_err(ShardedPacketIoError::Capabilities)?;
572            if capability.queue_count != 1 && capability.queue_count != queue_count {
573                return Err(ShardedPacketIoError::InvalidQueueCount);
574            }
575            capabilities.push(PacketCapabilities {
576                queue_count: 1,
577                ..capability
578            });
579        }
580        scheduler
581            .validate()
582            .map_err(|_| ShardedPacketIoError::Router(ShardRouterError::InvalidQueueConfig))?;
583        let queues_state = (0..queue_count)
584            .map(|_| {
585                Scheduler::new(scheduler)
586                    .map(|scheduler| {
587                        Mutex::new(QueueState {
588                            scheduler,
589                            waker: None,
590                        })
591                    })
592                    .map_err(|_| ShardedPacketIoError::Router(ShardRouterError::InvalidQueueConfig))
593            })
594            .collect::<Result<Vec<_>, _>>()?;
595        let local_caches = (0..queue_count)
596            .map(|_| {
597                Arc::new(Mutex::new(LocalRouteCache {
598                    entries: HashMap::new(),
599                    insertion_order: VecDeque::new(),
600                    max_entries: scheduler.max_active_flows,
601                }))
602            })
603            .collect::<Vec<_>>();
604        let shared = Arc::new(SharedIngress {
605            generation: RwLock::new(generation),
606            shard_count,
607            hasher: RandomState::new(),
608            ledger,
609            directory: (0..DIRECTORY_STRIPES)
610                .map(|_| Mutex::new(DirectoryStripe::default()))
611                .collect(),
612            max_directory_entries_per_stripe: scheduler
613                .max_active_flows
614                .div_ceil(DIRECTORY_STRIPES)
615                .max(1),
616            queues: queues_state,
617            local_caches: local_caches.clone(),
618            cache_pressure_active: AtomicBool::new(false),
619            counters: ConcurrentCounters::default(),
620        });
621        let adapters = (0..shard_count)
622            .zip(queues.into_iter().zip(capabilities).zip(local_caches))
623            .map(|(index, ((inner, capabilities), local_cache))| Self {
624                inner,
625                shard: ShardId::new(index),
626                capabilities,
627                shared: Arc::clone(&shared),
628                local_cache,
629            })
630            .collect();
631        Ok((
632            adapters,
633            ShardedPacketIoControl {
634                shared: Arc::clone(&shared),
635            },
636        ))
637    }

649    fn route_received(
650        &mut self,
651        mut received: PacketBatch,
652        out: &mut PacketBatch,
653    ) -> io::Result<()> {
654        self.shared.observe_pressure()?;
655        let generation = self
656            .shared
657            .generation
658            .read()
659            .map_err(|_| io::Error::other("network generation lock poisoned"))?;
660        while let Some(packet) = received.pop_front() {
661            let Ok((key, class)) = classify_packet_with_class(packet.payload(), *generation) else {
662                out.push(packet)
663                    .expect("platform batch must fit the destination batch");
664                continue;
665            };
666            let cached = if self.shared.cache_pressure_active.load(Ordering::Acquire) {
667                None
668            } else {
669                self.local_cache
670                    .lock()
671                    .map_err(|_| io::Error::other("local route cache lock poisoned"))?
672                    .entries
673                    .get(&key)
674                    .map(|entry| (entry.owner, entry.hash))
675            };
676            let (target, hash) = if let Some(route) = cached {
677                atomic_add_counter(&self.shared.counters.local_cache_hits, 1);
678                route
679            } else {
680                atomic_add_counter(&self.shared.counters.local_cache_misses, 1);
681                let route = self.shared.owner(key)?;
682                let mut cache = self
683                    .local_cache
684                    .lock()
685                    .map_err(|_| io::Error::other("local route cache lock poisoned"))?;
686                if !self.shared.cache_pressure_active.load(Ordering::Acquire) {
687                    if cache.entries.len() >= cache.max_entries {
688                        while let Some(oldest) = cache.insertion_order.pop_front() {
689                            if cache.entries.remove(&oldest).is_some() {
690                                break;
691                            }
692                        }
693                    }
694                    match self
695                        .shared
696                        .ledger
697                        .try_acquire(ResourceKind::MetadataBytes, LOCAL_ROUTE_ENTRY_CHARGE)
698                    {
699                        Ok(lease) => {
700                            cache.entries.insert(
701                                key,
702                                LocalRouteEntry {
703                                    owner: route.0,
704                                    hash: route.1,
705                                    _lease: lease,
706                                },
707                            );
708                            cache.insertion_order.push_back(key);
709                        }
710                        Err(_) => {
711                            atomic_add_counter(
712                                &self.shared.counters.local_cache_admission_failures,
713                                1,
714                            );
715                        }
716                    }
717                }
718                route
719            };
720            if target == self.shard {
721                atomic_add_counter(&self.shared.counters.local_routes, 1);
722                out.push(packet)
723                    .expect("platform batch must fit the destination batch");
724            } else {
725                atomic_add_counter(&self.shared.counters.forwarded_routes, 1);
726                self.shared.enqueue(target, hash, packet, class)?;
727            }
728        }
729        Ok(())
730    }
731}
732
733impl<I: PacketIo> PacketIo for ShardedPacketIo<I> {
734    async fn recv(&mut self, out: &mut PacketBatch) -> io::Result<usize> {
735        let forwarded = self.shared.drain(self.shard, out)?;
736        if forwarded > 0 {
737            return Ok(forwarded);
738        }
739        let mut received = PacketBatch::with_limit(self.capabilities.max_batch);
740        let mut platform_recv = Box::pin(self.inner.recv(&mut received));
741        let ingress = Arc::clone(&self.shared);
742        let owner = self.shard;
743        let _registration = WakerRegistration {
744            shared: Arc::clone(&ingress),
745            shard: owner,
746        };
747        let platform_result = poll_fn(|context| {
748            match ingress.poll_forwarded(owner, context.waker()) {
749                Ok(true) => return Poll::Ready(None),
750                Ok(false) => {}
751                Err(error) => return Poll::Ready(Some(Err(error))),
752            }
753            match platform_recv.as_mut().poll(context) {
754                Poll::Ready(result) => Poll::Ready(Some(result)),
755                Poll::Pending => Poll::Pending,
756            }
757        })
758        .await;
759        drop(platform_recv);
760        let Some(platform_result) = platform_result else {
761            let forwarded = self.shared.drain(self.shard, out)?;
762            return if forwarded == 0 {
763                Err(io::Error::from(io::ErrorKind::WouldBlock))
764            } else {
765                Ok(forwarded)
766            };
767        };
768        let reported = platform_result?;
769        if reported == 0 || reported != received.len() || reported > self.capabilities.max_batch {
770            return Err(io::Error::new(
771                io::ErrorKind::InvalidData,
772                "platform recv count does not match returned batch",
773            ));
774        }
775        self.route_received(received, out)?;
776        if out.is_empty() {
777            return Err(io::Error::from(io::ErrorKind::WouldBlock));
778        }
779        Ok(out.len())
780    }
781
782    async fn send(&mut self, packets: &PacketBatch) -> io::Result<usize> {
783        self.inner.send(packets).await
784    }
785
786    fn capabilities(&self) -> PacketCapabilities {
787        self.capabilities
788    }
789
790    fn owns_flow(&self, endpoint: IpEndpoint) -> bool {
791        self.shared
792            .endpoint_owner(endpoint)
793            .is_ok_and(|owner| owner == self.shard)
794    }
795}

797#[cfg(test)]
798mod tests {
799    use super::*;
800    use crate::{BudgetProfile, IpEndpoint, ResourceBudget, TransportProtocol};
801    use std::net::{Ipv4Addr, SocketAddr};
802
803    #[test]
804    fn directory_replaces_an_old_entry_in_the_same_stripe_under_its_own_pressure() {
805        let base = BudgetProfile::Mobile.budget();
806        let budget = ResourceBudget {
807            metadata_bytes: DIRECTORY_ENTRY_CHARGE,
808            ..base
809        };
810        let ledger = ResourceLedger::new(budget).unwrap();
811        let shared = SharedIngress {
812            generation: RwLock::new(NetworkGeneration::new(1)),
813            shard_count: 1,
814            hasher: RandomState::new(),
815            ledger: Arc::clone(&ledger),
816            directory: (0..DIRECTORY_STRIPES)
817                .map(|_| Mutex::new(DirectoryStripe::default()))
818                .collect(),
819            // Isolate the budget-driven eviction path from the count bound.
820            max_directory_entries_per_stripe: usize::MAX,
821            queues: Vec::new(),
822            local_caches: Vec::new(),
823            cache_pressure_active: AtomicBool::new(false),
824            counters: ConcurrentCounters::default(),
825        };
826        let mut first_by_stripe = HashMap::new();
827        let mut collision = None;
828        for port in 1_000..=u16::MAX {
829            let key = FlowKey {
830                endpoint: IpEndpoint {
831                    source: SocketAddr::from((Ipv4Addr::new(10, 0, 0, 1), port)),
832                    destination: SocketAddr::from((Ipv4Addr::new(1, 1, 1, 1), 443)),
833                    protocol: TransportProtocol::Tcp,
834                },
835                generation: NetworkGeneration::new(1),
836            };
837            let stripe =
838                usize::try_from(shared.hasher.hash_one(key) % DIRECTORY_STRIPES as u64).unwrap();
839            if let Some(first) = first_by_stripe.insert(stripe, key) {
840                collision = Some((stripe, first, key));
841                break;
842            }
843        }
844        let (stripe_index, first, second) = collision.expect("65 keys must collide in 64 stripes");
845
846        assert_eq!(shared.owner(first).unwrap().0, ShardId::new(0));
847        assert_eq!(shared.owner(second).unwrap().0, ShardId::new(0));
848        let stripe = shared.directory[stripe_index].lock().unwrap();
849        assert!(!stripe.entries.contains_key(&first));
850        assert!(stripe.entries.contains_key(&second));
851        assert_eq!(stripe.entries.len(), 1);
852        assert_eq!(stripe.insertion_order.len(), 1);
853        assert_eq!(shared.counters.directory_entries.load(Ordering::Relaxed), 1);
854        assert_eq!(
855            shared
856                .counters
857                .directory_admission_failures
858                .load(Ordering::Relaxed),
859            0
860        );
861        assert_eq!(
862            ledger.snapshot().used[ResourceKind::MetadataBytes as usize],
863            DIRECTORY_ENTRY_CHARGE
864        );
865    }
