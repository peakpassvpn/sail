1//! Browser ClientHello fingerprints: BoringSSL set up to send the same
2//! ClientHello a browser sends, so the outer TLS layer looks like the browser.
3//!
4//! Each profile names the browser version and capture it matches; its test
5//! compares against that capture in `tests/fixtures/tls/`.
6
7use std::io::{self, Read, Write};
8
9use anyhow::{anyhow, Result};
10use btls::ssl::{
11    CertificateCompressionAlgorithm, CertificateCompressor, ExtensionType, KeyShare,
12    SslConnectorBuilder, SslOptions, SslRef, SslSignatureAlgorithm, SslVersion,
13};
14
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub enum Fingerprint {
17    /// Chrome 154 on macOS, and on Android, which sends the same
18    /// ClientHello; Edge sends it too.
19    Chrome,
20    /// Firefox 156 on macOS.
21    Firefox,
22    /// Safari 26.3 on macOS, the same ClientHello as the system's
23    /// URLSession, and as URLSession on iOS 26.4: iOS and macOS share one
24    /// network stack.
25    Safari,
26    /// An Android app on OkHttp 4.12 over the platform's Conscrypt, as on
27    /// Android 17.
28    Android,
29}
30
31impl Fingerprint {
32    /// The fingerprint a config names: `chrome` (or `edge`, the same),
33    /// `firefox`, `safari` (or `ios`, the same), or `android` (OkHttp).
34    pub fn from_name(name: &str) -> Result<Self> {
35        match name {
36            "chrome" | "edge" => Ok(Self::Chrome),
37            "firefox" => Ok(Self::Firefox),
38            "safari" | "ios" => Ok(Self::Safari),
39            "android" => Ok(Self::Android),
40            _ => Err(anyhow!(
41                "unsupported fingerprint \"{}\", supported: chrome, edge, firefox, safari, ios, android",
42                name
43            )),
44        }
45    }
46
47    /// ALPN a browser offers when the config sets none.
48    pub(crate) fn default_alpn(self) -> &'static [&'static str] {
49        &["h2", "http/1.1"]
50    }
51
52    /// Settings shared by every connection of a client.
53    pub(crate) fn configure(self, builder: &mut SslConnectorBuilder) -> Result<()> {
54        builder.set_min_proto_version(Some(SslVersion::TLS1_2))?;
55        builder.set_max_proto_version(Some(SslVersion::TLS1_3))?;
56        match self {
57            Self::Chrome => chrome::configure(builder),
58            Self::Firefox => firefox::configure(builder),
59            Self::Safari => safari::configure(builder),
60            Self::Android => android::configure(builder),
61        }
62    }
63
64    /// Settings made per connection. `ech` says whether the connection offers
65    /// real ECH; otherwise it sends ECH GREASE, as the browser does.
66    pub(crate) fn configure_connection(
67        self,
68        ssl: &mut SslRef,
69        alpn: &[String],
70        ech: bool,
71    ) -> Result<()> {
72        match self {
73            Self::Chrome => chrome::configure_connection(ssl, alpn, ech),
74            Self::Firefox => firefox::configure_connection(ssl, ech),
75            Self::Safari => safari::configure_connection(ssl),
76            Self::Android => android::configure_connection(ssl),
77        }
78    }
79}

145    pub(crate) fn configure_connection(ssl: &mut SslRef, alpn: &[String], ech: bool) -> Result<()> {

327fn set_ech_grease_shape(ssl: &mut SslRef, aead_id: u16, payload_len: usize) -> Result<()> {
328    use foreign_types::ForeignTypeRef;
329    // SAFETY: `ssl` is a valid SSL.
330    if unsafe { btls_sys::SSL_set_ech_grease_shape(ssl.as_ptr(), aead_id, payload_len) } != 1 {
331        return Err(anyhow!("set ech grease shape failed"));
332    }
333    Ok(())
334}

404/// Copies `reader` to `output`.
405fn drain(mut reader: impl Read, output: &mut impl Write) -> io::Result<()> {
406    let mut buf = [0u8; 4096];
407    loop {
408        let n = reader.read(&mut buf)?;
409        if n == 0 {
410            return Ok(());
411        }
412        output.write_all(&buf[..n])?;
413    }
414}

416/// Zlib certificate decompression, which Firefox and Safari offer.
417struct Zlib;
418
419impl CertificateCompressor for Zlib {
420    const ALGORITHM: CertificateCompressionAlgorithm = CertificateCompressionAlgorithm::ZLIB;
421    const CAN_COMPRESS: bool = false;
422    const CAN_DECOMPRESS: bool = true;
423
424    fn decompress<W>(&self, input: &[u8], output: &mut W) -> io::Result<()>
425    where
426        W: Write,
427    {
428        let data = miniz_oxide::inflate::decompress_to_vec_zlib(input)
429            .map_err(|e| io::Error::new(io::ErrorKind::InvalidData, format!("{:?}", e)))?;
430        output.write_all(&data)
431    }
432}

434/// Zstandard certificate decompression, which Firefox offers.
435struct Zstd;
436
437impl CertificateCompressor for Zstd {
438    const ALGORITHM: CertificateCompressionAlgorithm = CertificateCompressionAlgorithm::ZSTD;
439    const CAN_COMPRESS: bool = false;
440    const CAN_DECOMPRESS: bool = true;
441
442    fn decompress<W>(&self, input: &[u8], output: &mut W) -> io::Result<()>
443    where
444        W: Write,
445    {
446        let decoder = ruzstd::decoding::StreamingDecoder::new(input)
447            .map_err(|e| io::Error::new(io::ErrorKind::InvalidData, e.to_string()))?;
448        drain(decoder, output)
449    }
450}

452/// Brotli certificate decompression, which Chrome and Firefox offer.
453struct Brotli;
454
455impl CertificateCompressor for Brotli {
456    const ALGORITHM: CertificateCompressionAlgorithm = CertificateCompressionAlgorithm::BROTLI;
457    const CAN_COMPRESS: bool = false;
458    const CAN_DECOMPRESS: bool = true;
459
460    fn decompress<W>(&self, input: &[u8], output: &mut W) -> io::Result<()>
461    where
462        W: Write,
463    {
464        drain(brotli_decompressor::Decompressor::new(input, 4096), output)
