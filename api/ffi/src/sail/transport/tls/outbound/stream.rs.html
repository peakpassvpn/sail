26    pub fn new(
27        server_name: String,
28        alpns: Vec<String>,
29        certificate: Option<String>,
30        insecure: bool,
31        fingerprint: Option<Fingerprint>,
32        ech: bool,
33        ech_disable_dns_lookup: bool,
34        ech_config_list: Option<String>,
35        dns_client: SyncDnsClient,
36    ) -> Result<Self> {
37        if let Some(list) = ech_config_list.as_deref() {

39        }
40        Ok(Handler {

42            client: TlsClient::new(&alpns, certificate.as_deref(), insecure, fingerprint)?,
43            ech: ech.then_some(Ech {
44                fixed_config_list: ech_config_list,
45                disable_dns_lookup: ech_disable_dns_lookup,
46            }),
47            dns_client,
48        })

51    fn resolve_selected_ech_config_list(
52        name: &str,
53        fixed_ech_config_list: Option<&str>,
54        auto_result: Option<anyhow::Result<String>>,
55    ) -> io::Result<Option<String>> {
56        match auto_result {
57            Some(Ok(value)) => {
58                trace!("ech source for {}: https/svcb dns record", name);
59                Ok(Some(value))
60            }
61            Some(Err(err)) => {
62                if let Some(fixed) = fixed_ech_config_list {
63                    trace!(
64                        "auto ech fetch failed for {}, fallback to fixed ech config: {}",
65                        name,
66                        err
67                    );
68                    Ok(Some(fixed.to_string()))
69                } else {
70                    trace!(
71                        "auto ech fetch failed for {}, no fixed ech config available: {}",
72                        name,
73                        err
74                    );
75                    Err(io::Error::other(format!(
76                        "auto ech fetch failed for {}: {}",
77                        name, err
78                    )))
79                }
80            }
81            None => {
82                if fixed_ech_config_list.is_some() {
83                    trace!("ech source for {}: fixed ech config", name);
84                } else {
85                    trace!("ech source for {}: none", name);
86                }
87                Ok(fixed_ech_config_list.map(str::to_string))
88            }
89        }
90    }

92    /// The DNS client's own connections must not look ECH up in DNS.
93    fn should_skip_ech_dns_lookup_for_session(sess: &Session) -> bool {
94        sess.inbound_tag == "dnsclient"
95    }

97    /// The ECHConfigList to offer to `name`, if any.
98    async fn select_ech_config_list(
99        &self,
100        name: &str,
101        sess: &Session,
102    ) -> io::Result<Option<Vec<u8>>> {
103        let Some(ech) = &self.ech else {
104            return Ok(None);
105        };
106        let fixed = ech.fixed_config_list.as_deref();
107        let selected = if ech.disable_dns_lookup {
108            trace!(
109                "ech source for {}: fixed-or-none (dns lookup disabled)",
110                name
111            );
112            fixed.map(str::to_string)
113        } else if Self::should_skip_ech_dns_lookup_for_session(sess) {
114            trace!(
115                "ech source for {}: fixed-or-none (dns lookup skipped)",
116                name
117            );
118            fixed.map(str::to_string)
119        } else {
120            let dns_client = self.dns_client.load_full();
121            let auto_result = dns_client.lookup_ech_config_list(name).await;
122            Self::resolve_selected_ech_config_list(name, fixed, Some(auto_result))?
123        };
124        selected
125            .map(|list| decode_ech_config_list(&list))
126            .transpose()
127    }
128}

130/// An ECHConfigList from base64 or PEM. A lone ECHConfig gets the list's
131/// length prefix.
132fn decode_ech_config_list(ech_config_list: &str) -> io::Result<Vec<u8>> {
133    let base64: String = ech_config_list
134        .lines()
135        .map(str::trim)
136        .filter(|line| !line.starts_with("-----"))
137        .collect();
138    let (list, _) = ensure_ech_config_list_bytes(decode_base64(&base64)?);
139    Ok(list)
140}
141
142fn ensure_ech_config_list_bytes(mut decoded: Vec<u8>) -> (Vec<u8>, bool) {
143    if decoded.len() >= 2 {
144        let declared = u16::from_be_bytes([decoded[0], decoded[1]]) as usize;
145        if declared == decoded.len().saturating_sub(2) {
146            return (decoded, false);
147        }
148    }
149
150    if decoded.len() >= 4 && decoded[0] == 0xfe && decoded[1] == 0x0d {
151        let len = decoded.len();
152        if u16::try_from(len).is_ok() {
153            let mut wrapped = Vec::with_capacity(len + 2);
154            wrapped.extend_from_slice(&(len as u16).to_be_bytes());
155            wrapped.append(&mut decoded);
156            return (wrapped, true);
157        }
158    }
159
160    (decoded, false)
161}

163fn decode_base64(data: &str) -> io::Result<Vec<u8>> {
164    fn value(byte: u8) -> Option<u8> {
165        match byte {
166            b'A'..=b'Z' => Some(byte - b'A'),
167            b'a'..=b'z' => Some(byte - b'a' + 26),
168            b'0'..=b'9' => Some(byte - b'0' + 52),
169            b'+' | b'-' => Some(62),
170            b'/' | b'_' => Some(63),
171            _ => None,
172        }
173    }
174
175    fn decode_chunk(chunk: &[u8; 4], output: &mut Vec<u8>) -> io::Result<()> {
176        if chunk[0] == 64 || chunk[1] == 64 {
177            return Err(io::Error::new(
178                io::ErrorKind::InvalidInput,
179                "invalid base64 padding",
180            ));
181        }
182        output.push((chunk[0] << 2) | (chunk[1] >> 4));
183        match (chunk[2], chunk[3]) {
184            (64, 64) => Ok(()),
185            (64, _) => Err(io::Error::new(
186                io::ErrorKind::InvalidInput,
187                "invalid base64 padding",
188            )),
189            (c2, 64) => {
190                output.push(((chunk[1] & 0x0f) << 4) | (c2 >> 2));
191                Ok(())
192            }
193            (c2, c3) => {
194                output.push(((chunk[1] & 0x0f) << 4) | (c2 >> 2));
195                output.push(((c2 & 0x03) << 6) | c3);
196                Ok(())
197            }
198        }
199    }
200
201    let mut output = Vec::with_capacity(data.len() * 3 / 4);
202    let mut chunk = [0_u8; 4];
203    let mut chunk_len = 0_usize;
204    let mut seen_padding = false;
205
206    for byte in data.bytes() {
207        if byte.is_ascii_whitespace() {
208            continue;
209        }
210        if byte == b'=' {
211            seen_padding = true;
212            chunk[chunk_len] = 64;
213            chunk_len += 1;
214        } else if let Some(decoded) = value(byte) {
215            if seen_padding {
216                return Err(io::Error::new(
217                    io::ErrorKind::InvalidInput,
218                    "invalid base64 padding",
219                ));
220            }
221            chunk[chunk_len] = decoded;
222            chunk_len += 1;
223        } else {
224            return Err(io::Error::new(
225                io::ErrorKind::InvalidInput,
226                "invalid base64 character",
227            ));
228        }
229
230        if chunk_len == 4 {
231            decode_chunk(&chunk, &mut output)?;
232            chunk = [0_u8; 4];
233            chunk_len = 0;
234        }
235    }
236
237    match chunk_len {
238        0 => Ok(output),
239        2 => {
240            output.push((chunk[0] << 2) | (chunk[1] >> 4));
241            Ok(output)
242        }
243        3 => {
244            output.push((chunk[0] << 2) | (chunk[1] >> 4));
245            output.push(((chunk[1] & 0x0f) << 4) | (chunk[2] >> 2));
246            Ok(output)
247        }
248        _ => Err(io::Error::new(
249            io::ErrorKind::InvalidInput,
250            "invalid base64 length",
251        )),
252    }
253}

255#[async_trait]
256impl OutboundStreamHandler for Handler {
257    fn connect_addr(&self) -> OutboundConnect {
258        OutboundConnect::Next
259    }
260
261    async fn handle<'a>(
262        &'a self,
263        sess: &'a Session,
264        _lhs: Option<&mut AnyStream>,
265        stream: Option<AnyStream>,
266    ) -> io::Result<AnyStream> {
267        trace!("handling outbound stream");
268        let stream = stream.ok_or_else(|| io::Error::other("invalid tls input"))?;
269        let name = if !self.server_name.is_empty() {
270            self.server_name.clone()
271        } else {
272            sess.destination.host()
273        };
274        let ech_config_list = self.select_ech_config_list(&name, sess).await?;
275        trace!(
276            "handling TLS {}, ech_enabled={}, ech_config_selected={}",
277            &name,
278            self.ech.is_some(),
279            ech_config_list.is_some()
280        );
281        let tls_stream = self
282            .client
283            .connect(
284                &name,
285                stream,
286                Some(VisionState::of(sess)),
287                ech_config_list.as_deref(),
288            )
289            .await
290            .map_err(|e| {
291                io::Error::new(
292                    io::ErrorKind::InvalidInput,
293                    format!("connect tls failed: {}", e),
294                )
295            })?;
296        // FIXME check negotiated alpn
297        Ok(Box::new(tls_stream))
298    }
299}
