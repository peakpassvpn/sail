27/// The ALPNs of a `tls` block's `alpn`, or `default` when it lists none.
28pub fn alpn_protocols(alpn: Option<&Listable>, default: &[&str]) -> Vec<Vec<u8>> {
29    match alpn.map(|a| a.clone().into_vec()) {
30        Some(list) if !list.is_empty() => list.into_iter().map(String::into_bytes).collect(),
31        _ => default.iter().map(|a| a.as_bytes().to_vec()).collect(),
32    }
33}

35/// A client TLS configuration trusting `certificate` (inline PEM or a
36/// path) instead of the bundled roots, or any server if `insecure`, and
37/// offering `alpns`, if any.
38pub fn client_crypto(
39    certificate: Option<&str>,
40    insecure: bool,
41    alpns: &[Vec<u8>],
42) -> Result<quinn_btls::ClientConfig> {
43    use quinn_btls::QuicSslContext;
44    let mut crypto =
45        quinn_btls::ClientConfig::new().map_err(|e| anyhow!("quic client config: {}", e))?;
46    if insecure {
47        crypto.verify_peer(false);
48    } else {
49        let certs = match certificate {
50            Some(certificate) => load_certificates(certificate)?,
51            None => bundled_root_certs()?.to_vec(),
52        };
53        let store = crypto.ctx_mut().cert_store_mut();
54        for cert in certs {
55            store.add_cert(cert)?;
56        }
57    }
58    if !alpns.is_empty() {
59        crypto
60            .set_alpn(alpns)
61            .map_err(|e| anyhow!("quic alpn: {}", e))?;
62    }
63    Ok(crypto)
64}

66/// What a client dials with, from its `tls` block.
67pub struct ClientTls {
68    /// The name the server is verified by: `server_name`, or the server's
69    /// address.
70    pub server_name: String,
71    pub crypto: quinn_btls::ClientConfig,
72}
73
74impl ClientTls {
75    /// From `tls`, for the server at `server`, offering `default_alpn`
76    /// unless `alpn` is set. Whatever `unsupported` finds is the caller's
77    /// to refuse; this reads none of it.
78    pub fn new(
79        tls: &OutboundTls,
80        server: &str,
81        default_alpn: &[&str],
82        env: &RuntimeEnv,
83    ) -> Result<Self> {
84        let crypto = client_crypto(
85            trusted_certificate(tls, env).as_deref(),
86            tls.insecure,
87            &alpn_protocols(tls.alpn.as_ref(), default_alpn),
88        )?;
89        Ok(Self {
90            server_name: tls.server_name.clone().unwrap_or_else(|| server.to_owned()),
91            crypto,
92        })
93    }
94}

96/// The first of `tls.reality`, `tls.ech` and `tls.utls` that the block
97/// enables: none of them works over QUIC.
98pub fn unsupported(tls: &OutboundTls) -> Option<&'static str> {
99    [
100        ("reality", tls.reality.as_ref().is_some_and(|r| r.enabled)),
101        ("ech", tls.ech.as_ref().is_some_and(|e| e.enabled)),
102        ("utls", tls.utls.as_ref().is_some_and(|u| u.enabled)),
103    ]
104    .into_iter()
105    .find_map(|(field, on)| on.then_some(field))
106}

108/// A server TLS configuration presenting `certificate` with `key`, each
109/// inline PEM or a path, and accepting `alpns`, if any.
110pub fn server_crypto(
111    certificate: &str,
112    key: &str,
113    alpns: &[Vec<u8>],
114) -> Result<quinn_btls::ServerConfig> {
115    use quinn_btls::QuicSslContext;
116    let mut certs = load_certificates(certificate)?.into_iter();
117    let key = load_private_key(key)?;
118
119    let mut crypto =
120        quinn_btls::ServerConfig::new().map_err(|e| anyhow!("quic server config: {}", e))?;
121    let ctx = crypto.ctx_mut();
122    let leaf = certs
123        .next()
124        .ok_or_else(|| anyhow!("no certificate found"))?;
125    ctx.set_certificate(leaf)?;
126    for cert in certs {
127        ctx.add_to_cert_chain(cert)?;
128    }
129    ctx.set_private_key(key)?;
130    ctx.check_private_key()
131        .map_err(|e| anyhow!("private key does not match the certificate: {}", e))?;
132    if !alpns.is_empty() {
133        crypto
134            .set_alpn(alpns)
135            .map_err(|e| anyhow!("quic alpn: {}", e))?;
136    }
137    Ok(crypto)
138}

140/// [`server_crypto`] from an inbound's `tls` block, its errors as the
141/// inbound `tag`'s.
142pub fn inbound_crypto(
143    tag: &str,
144    tls: &InboundTls,
145    env: &RuntimeEnv,
146    alpns: &[Vec<u8>],
147) -> Result<quinn_btls::ServerConfig> {
148    let certificate = tls.certificate(tag, env)?;
149    let key = tls.key(tag, env)?;
150    server_crypto(&certificate, &key, alpns).map_err(|e| anyhow!("[{}] inbound: tls: {}", tag, e))
151}

159/// The congestion controllers quinn has, named as in sing-box.
160#[derive(Deserialize, Debug, Clone, Copy, Default, PartialEq, Eq)]
161#[serde(rename_all = "snake_case")]
162pub enum CongestionControl {
163    /// sing-box's default.
164    #[default]
165    Cubic,
166    NewReno,
167    Bbr,
168}
169
170impl CongestionControl {
171    pub fn factory(self) -> Arc<dyn ControllerFactory + Send + Sync> {
172        match self {
173            Self::Cubic => Arc::new(quinn::congestion::CubicConfig::default()),
174            Self::NewReno => Arc::new(quinn::congestion::NewRenoConfig::default()),
175            Self::Bbr => Arc::new(quinn::congestion::BbrConfig::default()),
176        }
177    }
178}

180/// Which end of connections a transport configuration is for.
181#[derive(Debug, Clone, Copy, PartialEq, Eq)]
182pub enum Side {
183    Client,
184    Server,
185}
186
187/// The transport parameters the `quic` options set for `side`: concurrent
188/// bidirectional streams, the idle timeout and keep-alives, none if the
189/// interval is zero. Congestion is controlled as `congestion` makes it:
190/// [`CongestionControl::factory`], or a protocol's own.
191pub fn transport_config(
192    tuning: &Tuning,
193    side: Side,
194    congestion: Arc<dyn ControllerFactory + Send + Sync>,
195) -> quinn::TransportConfig {
196    let (idle, keep_alive) = match side {
197        Side::Client => (
198            tuning.client_idle_timeout,
199            tuning.client_keep_alive_interval,
200        ),
201        Side::Server => (
202            tuning.server_idle_timeout,
203            tuning.server_keep_alive_interval,
204        ),
205    };
206    let mut config = quinn::TransportConfig::default();
207    config
208        .max_concurrent_bidi_streams(quinn::VarInt::from_u32(tuning.max_concurrent_streams))
209        .max_idle_timeout(quinn::IdleTimeout::try_from(idle).ok())
210        .keep_alive_interval((!keep_alive.is_zero()).then_some(keep_alive))
211        .congestion_controller_factory(congestion);
212    config
213}

215/// A UDP socket for talking to `peer`, bound as `dial` says.
216pub async fn bind(peer: IpAddr, dial: &DialOptions) -> io::Result<std::net::UdpSocket> {
217    let unspecified = match peer {
218        IpAddr::V4(_) => SocketAddr::new(Ipv4Addr::UNSPECIFIED.into(), 0),
219        IpAddr::V6(_) => SocketAddr::new(Ipv6Addr::UNSPECIFIED.into(), 0),
220    };
221    crate::net::new_udp_socket(&unspecified, dial)
222        .await?
223        .into_std()
224}

226/// `socket` as quinn takes it, for wrapping further.
227pub fn wrap_socket(socket: std::net::UdpSocket) -> io::Result<Arc<dyn AsyncUdpSocket>> {
228    socket.set_nonblocking(true)?;
229    quinn::TokioRuntime.wrap_udp_socket(socket)
230}

232/// An endpoint on `socket`, serving with `server` if set.
233pub fn endpoint(
234    socket: std::net::UdpSocket,
235    server: Option<quinn::ServerConfig>,
236) -> io::Result<quinn::Endpoint> {
237    endpoint_on(wrap_socket(socket)?, server)
238}

240/// [`endpoint`] on a socket of quinn's.
241pub fn endpoint_on(
242    socket: Arc<dyn AsyncUdpSocket>,
243    server: Option<quinn::ServerConfig>,
244) -> io::Result<quinn::Endpoint> {
245    quinn::Endpoint::new_with_abstract_socket(
246        quinn_btls::helpers::default_endpoint_config(),
247        server,
248        socket,
249        Arc::new(quinn::TokioRuntime),
250    )
251}

253/// A proxied TCP connection: one bidirectional QUIC stream, and `G`, kept
254/// as long as the stream (TUIC counts its relays by it).
255///
256/// Reads and writes are quinn's own: the peer's FIN reads as the end of
257/// the stream, a reset or a STOP_SENDING fails with `ConnectionReset`, a
258/// lost connection with `NotConnected`. Shutting down finishes our side,
259/// and so does dropping the stream.
260pub struct QuicStream<G = ()> {
261    send: quinn::SendStream,
262    recv: quinn::RecvStream,
263    _guard: G,
264}
265
266impl QuicStream {
267    pub fn new(send: quinn::SendStream, recv: quinn::RecvStream) -> Self {
268        Self::guarded(send, recv, ())
269    }
270}
271
272impl<G> QuicStream<G> {
273    pub fn guarded(send: quinn::SendStream, recv: quinn::RecvStream, guard: G) -> Self {
274        Self {
275            send,
276            recv,
277            _guard: guard,
278        }
279    }
280}

282impl<G: Unpin> AsyncRead for QuicStream<G> {
283    fn poll_read(
284        mut self: Pin<&mut Self>,
285        cx: &mut Context<'_>,
286        buf: &mut ReadBuf<'_>,
287    ) -> Poll<io::Result<()>> {
288        Pin::new(&mut self.recv).poll_read(cx, buf)
289    }
290}
291
292impl<G: Unpin> AsyncWrite for QuicStream<G> {
293    fn poll_write(
294        mut self: Pin<&mut Self>,
295        cx: &mut Context<'_>,
296        buf: &[u8],
297    ) -> Poll<io::Result<usize>> {
298        // quinn's own `poll_write` fails with a `WriteError`.
299        Pin::new(&mut self.send)
300            .poll_write(cx, buf)
301            .map_err(io::Error::from)
302    }
303
304    fn poll_flush(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<io::Result<()>> {
305        Pin::new(&mut self.send).poll_flush(cx)
306    }
307
308    fn poll_shutdown(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<io::Result<()>> {
309        Pin::new(&mut self.send).poll_shutdown(cx)
310    }
311}

338    /// A server whose certificate has only the IP SAN 127.0.0.1, and a
339    /// client trusting that certificate.
340    fn ip_san_pair() -> (quinn::Endpoint, quinn_btls::ClientConfig) {
341        let cert = rcgen::generate_simple_self_signed(vec!["127.0.0.1".into()]).unwrap();
342        let pem = cert.cert.pem();
343        let server = server_crypto(&pem, &cert.key_pair.serialize_pem(), &[]).unwrap();
344        let server = endpoint(
345            std::net::UdpSocket::bind((Ipv4Addr::LOCALHOST, 0)).unwrap(),
346            Some(server_config(server).unwrap()),
347        )
348        .unwrap();
349        (server, client_crypto(Some(&pem), false, &[]).unwrap())
350    }

352    /// Dials `server` as `server_name`: the SNI the server saw, if the
353    /// handshake succeeds.
354    async fn dial(
355        server: &quinn::Endpoint,
356        crypto: quinn_btls::ClientConfig,
357        server_name: &str,
358    ) -> Result<Option<String>> {
359        let client = endpoint(
360            std::net::UdpSocket::bind((Ipv4Addr::LOCALHOST, 0)).unwrap(),
361            None,
362        )
363        .unwrap();
364        let config = quinn::ClientConfig::new(Arc::new(crypto));
365        let accept = async {
366            let conn = server.accept().await.unwrap().await?;
367            let data = conn.handshake_data().unwrap();
368            let data = data.downcast::<quinn_btls::HandshakeData>().unwrap();
369            Ok::<_, quinn::ConnectionError>(data.server_name)
370        };
371        let connect = async {
372            let connecting =
373                client.connect_with(config, server.local_addr().unwrap(), server_name)?;
374            Ok::<_, anyhow::Error>(connecting.await?)
375        };
376        let (sni, conn) = tokio::join!(accept, connect);
377        conn?;
378        Ok(sni?)
379    }
