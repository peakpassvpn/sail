11pub struct Handler {
12    pub address: String,
13    pub port: u16,
14    pub cipher: String,
15    pub password: String,
16}

18#[async_trait]
19impl OutboundDatagramHandler for Handler {
20    fn connect_addr(&self) -> OutboundConnect {
21        OutboundConnect::Proxy(Network::Udp, self.address.clone(), self.port)
22    }
23
24    fn transport_type(&self) -> DatagramTransportType {
25        DatagramTransportType::Unreliable
26    }
27
28    async fn handle<'a>(
29        &'a self,
30        sess: &'a Session,
31        transport: Option<AnyOutboundTransport>,
32    ) -> io::Result<AnyOutboundDatagram> {
33        tracing::trace!("handling outbound datagram");
34        let server_addr = SocksAddr::try_from((&self.address, self.port))?;
35
36        let socket = if let Some(OutboundTransport::Datagram(socket)) = transport {
37            socket
38        } else {
39            // Don't accept stream transport because we can't determine datagram
40            // boundary.
41            return Err(io::Error::other("invalid ss input"));
42        };
43
44        let dgram = ShadowedDatagram::new(&self.cipher, &self.password)?;
45
46        let destination = match &sess.destination {
47            SocksAddr::Domain(domain, port) => Some(SocksAddr::Domain(domain.to_owned(), *port)),
48            _ => None,
49        };
50
51        Ok(Box::new(Datagram {
52            dgram,
53            socket,
54            destination,
55            server_addr,
56        }))
57    }
58}

60pub struct Datagram {
61    pub dgram: ShadowedDatagram,
62    pub socket: Box<dyn OutboundDatagram>,
63    pub destination: Option<SocksAddr>,
64    pub server_addr: SocksAddr,
65}

67impl OutboundDatagram for Datagram {
68    fn split(
69        self: Box<Self>,
70    ) -> (
71        Box<dyn OutboundDatagramRecvHalf>,
72        Box<dyn OutboundDatagramSendHalf>,
73    ) {
74        let dgram = Arc::new(self.dgram);
75        let (r, s) = self.socket.split();
76        (
77            Box::new(DatagramRecvHalf(dgram.clone(), r, self.destination)),
78            Box::new(DatagramSendHalf {
79                dgram,
80                send_half: s,
81                server_addr: self.server_addr,
82            }),
83        )
84    }
85}

93#[async_trait]
94impl OutboundDatagramRecvHalf for DatagramRecvHalf {
95    async fn recv_from(&mut self, buf: &mut [u8]) -> io::Result<(usize, SocksAddr)> {
96        let mut recv_buf = BytesMut::new();
97        recv_buf.resize(buf.len(), 0);
98        let (n, _) = self.1.recv_from(&mut recv_buf).await?;
99        recv_buf.resize(n, 0);
100        let plaintext = self.0.decrypt(recv_buf).map_err(|_| shadow::crypto_err())?;
101        let src_addr = SocksAddr::try_from((&plaintext[..], SocksAddrWireType::PortLast))?;
102        let payload_len = plaintext.len() - src_addr.size();
103        assert!(payload_len <= buf.len());
104        buf[..payload_len]
105            .copy_from_slice(&plaintext[src_addr.size()..src_addr.size() + payload_len]);
106        Ok((payload_len, self.2.clone().unwrap_or(src_addr)))
107    }
108}

110pub struct DatagramSendHalf {
111    dgram: Arc<ShadowedDatagram>,
112    send_half: Box<dyn OutboundDatagramSendHalf>,
113    server_addr: SocksAddr,
114}
115
116#[async_trait]
117impl OutboundDatagramSendHalf for DatagramSendHalf {
118    async fn send_to(&mut self, buf: &[u8], target: &SocksAddr) -> io::Result<usize> {
119        let mut send_buf = BytesMut::new();
120        target.write_buf(&mut send_buf, SocksAddrWireType::PortLast);
121        send_buf.put_slice(buf);
122        let ciphertext = self
123            .dgram
124            .encrypt(send_buf)
125            .map_err(|_| shadow::crypto_err())?;
126        self.send_half
127            .send_to(&ciphertext, &self.server_addr)
128            .map_ok(|_| buf.len())
129            .await
130    }
131
132    async fn close(&mut self) -> io::Result<()> {
133        self.send_half.close().await
134    }
