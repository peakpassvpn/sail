1//! TUIC v5 commands on the wire.
2//!
3//! Every command is `VER TYPE OPT`. What differs from SOCKS is the address:
4//! its type bytes are TUIC's own, and `None` (a lone `0xff`, no port) stands
5//! for the address of a fragment that is not the first of its packet.
6

15pub const VERSION: u8 = 0x05;
16
17pub const CMD_AUTHENTICATE: u8 = 0x00;
18pub const CMD_CONNECT: u8 = 0x01;
19pub const CMD_PACKET: u8 = 0x02;
20pub const CMD_DISSOCIATE: u8 = 0x03;
21pub const CMD_HEARTBEAT: u8 = 0x04;

23const ADDR_NONE: u8 = 0xff;
24const ADDR_DOMAIN: u8 = 0x00;
25const ADDR_IPV4: u8 = 0x01;
26const ADDR_IPV6: u8 = 0x02;

28/// `Packet` without `VER TYPE`: `ASSOC_ID PKT_ID FRAG_TOTAL FRAG_ID SIZE`.
29pub const PACKET_FIXED_LEN: usize = 8;
30
31/// The length of the token `Authenticate` carries, exported from TLS.
32pub const TOKEN_LEN: usize = 32;

34fn invalid(msg: &str) -> io::Error {
35    io::Error::new(io::ErrorKind::InvalidData, format!("tuic: {}", msg))
36}

38/// The encoded length of `addr`, `None` included.
39pub fn address_len(addr: Option<&SocksAddr>) -> usize {
40    match addr {
41        None => 1,
42        Some(SocksAddr::Ip(SocketAddr::V4(_))) => 1 + 4 + 2,
43        Some(SocksAddr::Ip(SocketAddr::V6(_))) => 1 + 16 + 2,
44        Some(SocksAddr::Domain(domain, _)) => 1 + 1 + domain.len() + 2,
45    }
46}
47
48pub fn write_address<B: BufMut>(buf: &mut B, addr: Option<&SocksAddr>) -> io::Result<()> {
49    match addr {
50        None => buf.put_u8(ADDR_NONE),
51        Some(SocksAddr::Ip(SocketAddr::V4(a))) => {
52            buf.put_u8(ADDR_IPV4);
53            buf.put_slice(&a.ip().octets());
54            buf.put_u16(a.port());
55        }
56        Some(SocksAddr::Ip(SocketAddr::V6(a))) => {
57            buf.put_u8(ADDR_IPV6);
58            buf.put_slice(&a.ip().octets());
59            buf.put_u16(a.port());
60        }
61        Some(SocksAddr::Domain(domain, port)) => {
62            let len = u8::try_from(domain.len())
63                .ok()
64                .filter(|n| *n > 0)
65                .ok_or_else(|| invalid("domain length out of range"))?;
66            buf.put_u8(ADDR_DOMAIN);
67            buf.put_u8(len);
68            buf.put_slice(domain.as_bytes());
69            buf.put_u16(*port);
70        }
71    }
72    Ok(())
73}

75fn domain_addr(domain: &[u8], port: u16) -> io::Result<SocksAddr> {
76    let domain = std::str::from_utf8(domain).map_err(|_| invalid("domain is not UTF-8"))?;
77    if domain.is_empty() {
78        return Err(invalid("empty domain"));
79    }
80    SocksAddr::try_from((domain, port))
81}

83fn ipv6_addr(octets: [u8; 16], port: u16) -> SocksAddr {
84    // Go writes an IPv4 address it keeps in 16 bytes as IPv6.
85    let ip = Ipv6Addr::from(octets);
86    match ip.to_ipv4_mapped() {
87        Some(v4) => SocksAddr::from((IpAddr::V4(v4), port)),
88        None => SocksAddr::from((IpAddr::V6(ip), port)),
89    }
90}

92/// Reads an address off the front of `buf`, returning it and how many bytes
93/// it took.
94pub fn decode_address(buf: &[u8]) -> io::Result<(Option<SocksAddr>, usize)> {
95    let short = || invalid("address truncated");
96    let (&kind, rest) = buf.split_first().ok_or_else(short)?;
97    let port_at = |rest: &[u8], n: usize| -> io::Result<u16> {
98        rest.get(n..n + 2)
99            .map(|p| u16::from_be_bytes([p[0], p[1]]))
100            .ok_or_else(short)
101    };
102    match kind {
103        ADDR_NONE => Ok((None, 1)),
104        ADDR_IPV4 => {
105            let ip: [u8; 4] = rest
106                .get(..4)
107                .ok_or_else(short)?
108                .try_into()
109                .map_err(|_| short())?;
110            let port = port_at(rest, 4)?;
111            Ok((Some(SocksAddr::from((Ipv4Addr::from(ip), port))), 1 + 4 + 2))
112        }
113        ADDR_IPV6 => {
114            let ip: [u8; 16] = rest
115                .get(..16)
116                .ok_or_else(short)?
117                .try_into()
118                .map_err(|_| short())?;
119            let port = port_at(rest, 16)?;
120            Ok((Some(ipv6_addr(ip, port)), 1 + 16 + 2))
121        }
122        ADDR_DOMAIN => {
123            let (&len, rest) = rest.split_first().ok_or_else(short)?;
124            let len = len as usize;
125            let domain = rest.get(..len).ok_or_else(short)?;
126            let port = port_at(rest, len)?;
127            Ok((Some(domain_addr(domain, port)?), 1 + 1 + len + 2))
128        }
129        other => Err(invalid(&format!("unknown address type {:#04x}", other))),
130    }
131}

133/// Reads an address from a stream.
134pub async fn read_address<R: AsyncRead + Unpin>(r: &mut R) -> io::Result<Option<SocksAddr>> {
135    match r.read_u8().await? {
136        ADDR_NONE => Ok(None),
137        ADDR_IPV4 => {
138            let mut ip = [0u8; 4];
139            r.read_exact(&mut ip).await?;
140            let port = r.read_u16().await?;
141            Ok(Some(SocksAddr::from((Ipv4Addr::from(ip), port))))
142        }
143        ADDR_IPV6 => {
144            let mut ip = [0u8; 16];
145            r.read_exact(&mut ip).await?;
146            let port = r.read_u16().await?;
147            Ok(Some(ipv6_addr(ip, port)))
148        }
149        ADDR_DOMAIN => {
150            let len = r.read_u8().await? as usize;
151            let mut domain = vec![0u8; len];
152            r.read_exact(&mut domain).await?;
153            let port = r.read_u16().await?;
154            Ok(Some(domain_addr(&domain, port)?))
155        }
156        other => Err(invalid(&format!("unknown address type {:#04x}", other))),
157    }
158}

160/// Reads `VER TYPE`, checking the version, and returns the type.
161pub async fn read_command<R: AsyncRead + Unpin>(r: &mut R) -> io::Result<u8> {
162    let mut head = [0u8; 2];
163    r.read_exact(&mut head).await?;
164    if head[0] != VERSION {
165        return Err(invalid(&format!("unsupported version {:#04x}", head[0])));
166    }
167    Ok(head[1])
168}

170#[cfg_attr(not(feature = "outbound-tuic"), allow(dead_code))]
171pub fn encode_authenticate(uuid: &[u8; 16], token: &[u8; TOKEN_LEN]) -> Bytes {
172    let mut buf = BytesMut::with_capacity(2 + 16 + TOKEN_LEN);
173    buf.put_u8(VERSION);
174    buf.put_u8(CMD_AUTHENTICATE);
175    buf.put_slice(uuid);
176    buf.put_slice(token);
177    buf.freeze()
178}

180/// `Connect` to `addr`, with `payload` after it.
181#[cfg_attr(not(feature = "outbound-tuic"), allow(dead_code))]
182pub fn encode_connect(addr: &SocksAddr, payload: &[u8]) -> io::Result<Bytes> {
183    let mut buf = BytesMut::with_capacity(2 + address_len(Some(addr)) + payload.len());
184    buf.put_u8(VERSION);
185    buf.put_u8(CMD_CONNECT);
186    write_address(&mut buf, Some(addr))?;
187    buf.put_slice(payload);
188    Ok(buf.freeze())
189}

191#[cfg_attr(not(feature = "outbound-tuic"), allow(dead_code))]
192pub fn encode_dissociate(assoc_id: u16) -> Bytes {
193    let mut buf = BytesMut::with_capacity(4);
194    buf.put_u8(VERSION);
195    buf.put_u8(CMD_DISSOCIATE);
196    buf.put_u16(assoc_id);
197    buf.freeze()
198}

200pub fn encode_heartbeat() -> Bytes {

204/// The header of one `Packet`: one fragment of a UDP packet.
205#[derive(Debug, Clone, PartialEq, Eq)]
206pub struct PacketHeader {
207    pub assoc_id: u16,
208    pub pkt_id: u16,
209    pub frag_total: u8,
210    pub frag_id: u8,
211    /// The target from the client, the source from the server; only the
212    /// first fragment has one.
213    pub addr: Option<SocksAddr>,
214}
215
216impl PacketHeader {
217    /// The encoded length of the whole command, `VER TYPE` included,
218    /// without the payload.
219    pub fn len(&self) -> usize {
220        2 + PACKET_FIXED_LEN + address_len(self.addr.as_ref())
221    }
222
223    fn check(&self) -> io::Result<()> {
224        if self.frag_total == 0 || self.frag_id >= self.frag_total {
225            return Err(invalid("fragment id out of range"));
226        }
227        if self.frag_id == 0 && self.addr.is_none() {
228            return Err(invalid("first fragment without an address"));
229        }
230        Ok(())
231    }
232}
233
234/// The whole `Packet` command: header and `payload`.
235pub fn encode_packet(header: &PacketHeader, payload: &[u8]) -> io::Result<Bytes> {
236    let size = u16::try_from(payload.len()).map_err(|_| invalid("packet too large"))?;
237    let mut buf = BytesMut::with_capacity(header.len() + payload.len());
238    buf.put_u8(VERSION);
239    buf.put_u8(CMD_PACKET);
240    buf.put_u16(header.assoc_id);
241    buf.put_u16(header.pkt_id);
242    buf.put_u8(header.frag_total);
243    buf.put_u8(header.frag_id);
244    buf.put_u16(size);
245    write_address(&mut buf, header.addr.as_ref())?;
246    buf.put_slice(payload);
247    Ok(buf.freeze())
248}
249
250/// A `Packet` after its `VER TYPE`, read from a unidirectional stream.
251pub async fn read_packet<R: AsyncRead + Unpin>(r: &mut R) -> io::Result<(PacketHeader, Bytes)> {
252    let mut fixed = [0u8; PACKET_FIXED_LEN];
253    r.read_exact(&mut fixed).await?;
254    let addr = read_address(r).await?;
255    let header = PacketHeader {
256        assoc_id: u16::from_be_bytes([fixed[0], fixed[1]]),
257        pkt_id: u16::from_be_bytes([fixed[2], fixed[3]]),
258        frag_total: fixed[4],
259        frag_id: fixed[5],
260        addr,
261    };
262    header.check()?;
263    let size = u16::from_be_bytes([fixed[6], fixed[7]]) as usize;
264    let mut payload = vec![0u8; size];
265    r.read_exact(&mut payload).await?;
266    Ok((header, Bytes::from(payload)))
267}

269/// What a QUIC datagram carries.
270#[derive(Debug, PartialEq, Eq)]
271pub enum Datagram {
272    Packet(PacketHeader, Bytes),
273    Heartbeat,
274}
275
276pub fn decode_datagram(data: Bytes) -> io::Result<Datagram> {
277    if data.len() < 2 {
278        return Err(invalid("datagram truncated"));
279    }
280    if data[0] != VERSION {
281        return Err(invalid(&format!("unsupported version {:#04x}", data[0])));
282    }
283    match data[1] {
284        CMD_HEARTBEAT => Ok(Datagram::Heartbeat),
285        CMD_PACKET => {
286            let body = &data[2..];
287            if body.len() < PACKET_FIXED_LEN {
288                return Err(invalid("packet truncated"));
289            }
290            let (addr, addr_len) = decode_address(&body[PACKET_FIXED_LEN..])?;
291            let header = PacketHeader {
292                assoc_id: u16::from_be_bytes([body[0], body[1]]),
293                pkt_id: u16::from_be_bytes([body[2], body[3]]),
294                frag_total: body[4],
295                frag_id: body[5],
296                addr,
297            };
298            header.check()?;
299            let size = u16::from_be_bytes([body[6], body[7]]) as usize;
300            let start = 2 + PACKET_FIXED_LEN + addr_len;
301            if data.len() - start != size {
302                return Err(invalid("packet size mismatch"));
303            }
304            Ok(Datagram::Packet(header, data.slice(start..)))
305        }
306        other => Err(invalid(&format!(
307            "unexpected command {:#04x} in a datagram",
308            other
309        ))),
310    }
311}

317    fn roundtrip(addr: Option<SocksAddr>) {
318        let mut buf = BytesMut::new();
319        write_address(&mut buf, addr.as_ref()).unwrap();
320        assert_eq!(buf.len(), address_len(addr.as_ref()));
321        let (decoded, n) = decode_address(&buf).unwrap();
322        assert_eq!(n, buf.len());
323        assert_eq!(decoded, addr);
324        let rt = tokio::runtime::Builder::new_current_thread()
325            .build()
326            .unwrap();
327        let read = rt.block_on(read_address(&mut &buf[..])).unwrap();
328        assert_eq!(read, addr);
329    }

397    #[test]
398    fn packets_roundtrip_through_datagrams_and_streams() {
399        let header = PacketHeader {
400            assoc_id: 0xabcd,
401            pkt_id: 7,
402            frag_total: 1,
403            frag_id: 0,

435    fn malformed_datagrams_are_errors() {
436        let header = PacketHeader {
437            assoc_id: 1,
438            pkt_id: 1,

446        let mut longer = BytesMut::from(&wire[..]);

464        let bad = PacketHeader {
465            frag_total: 0,
466            ..header
467        };
468        assert!(decode_datagram(encode_packet(&bad, b"x").unwrap()).is_err());

470        assert!(decode_datagram(encode_dissociate(1)).is_err());
