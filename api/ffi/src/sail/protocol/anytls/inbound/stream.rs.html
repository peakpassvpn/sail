23/// Streams handshaken and waiting for the listener to take them.
24const INCOMING_QUEUE: usize = 64;
25
26pub struct Handler {
27    /// Users by the SHA-256 of their password, with their names.
28    users: HashMap<[u8; 32], Option<Arc<str>>>,
29    padding: Arc<PaddingScheme>,
30    /// How long a stream has to name its destination.
31    handshake_timeout: Duration,
32    /// Where what fails to authenticate goes; closed without one.
33    fallback: Option<Fallback>,
34}
35
36impl Handler {
37    pub fn new(
38        users: HashMap<[u8; 32], Option<Arc<str>>>,
39        padding: Arc<PaddingScheme>,
40        handshake_timeout: Duration,
41        fallback: Option<Fallback>,
42    ) -> Self {
43        Handler {
44            users,
45            padding,
46            handshake_timeout,
47            fallback,
48        }
49    }
50}

52#[async_trait]
53impl InboundStreamHandler for Handler {
54    async fn handle<'a>(
55        &'a self,
56        mut sess: ProxySession,
57        mut stream: AnyStream,
58    ) -> io::Result<AnyInboundTransport> {
59        tracing::trace!("handling inbound stream");
60        // The password's hash, and no more: what is read here is what the
61        // fallback is given if it is not a user's. Nothing tells a hash from
62        // anything else before all of it is in.
63        let mut hash = [0u8; AUTH_HASH_LEN];
64        let mut read = 0;
65        let reading = async {
66            while read < AUTH_HASH_LEN {
67                let n = stream.read(&mut hash[read..]).await?;
68                if n == 0 {
69                    return Ok(false);
70                }
71                read += n;
72            }
73            Ok::<_, io::Error>(true)
74        };
75        let complete = match self.fallback {
76            // A peer that sends part of a hash and waits is not a client.
77            Some(_) => timeout(HEADER_TIMEOUT, reading)
78                .await
79                .unwrap_or(Ok(false))?,
80            None => reading.await?,
81        };
82        let user = match complete {
83            true => self.users.get(&hash).ok_or("anytls: unknown user"),
84            false => Err("anytls: not an AnyTLS request"),
85        };
86        let user = match user {
87            Ok(user) => user,
88            Err(why) => {
89                return Err(match &self.fallback {
90                    Some(fallback) => fallback.relay(&sess, stream, hash[..read].to_vec(), why),
91                    None => io::Error::new(io::ErrorKind::PermissionDenied, why),
92                })
93            }
94        };
95        // The padding after it is a user's to send, and the handshake's
96        // deadline bounds it.
97        read_auth_padding(&mut stream).await?;
98        sess.user = user.clone();
99        let (tx, rx) = mpsc::channel(INCOMING_QUEUE);
100        let handshake_timeout = self.handshake_timeout;
101        let session = Session::server(
102            stream,
103            self.padding.clone(),
104            Box::new(move |stream| {
105                tokio::spawn(accept(stream, sess.clone(), tx.clone(), handshake_timeout));
106            }),
107        );
108        Ok(InboundTransport::Incoming(Box::new(Incoming {
109            rx,
110            _session: session,
111        })))
112    }
113}

115/// Reads what a new stream asks for, and passes it on to be routed.
116async fn accept(
117    mut stream: Stream,
118    mut sess: ProxySession,
119    tx: mpsc::Sender<AnyBaseInboundTransport>,
120    handshake_timeout: Duration,
121) {
122    let sid = stream.id();
123    // A stream for UDP over TCP is handed on like any other, and served
124    // where every inbound's are; version 1 is refused here, where the
125    // client can be told.
126    let handshake = async {
127        let destination = SocksAddr::read_from(&mut stream, SocksAddrWireType::PortLast).await?;
128        if uot::version(&destination) == Some(1) {
129            return Err(io::Error::other("udp-over-tcp version 1 is not supported"));
130        }
131        Ok::<_, io::Error>(destination)
132    };
133    let destination = match tokio::time::timeout(handshake_timeout, handshake).await {
134        Ok(Ok(v)) => v,
135        Ok(Err(e)) => {
136            debug!("anytls stream {}: {}", sid, e);
137            let _ = stream.report(Some(&e.to_string())).await;
138            return;
139        }
140        Err(_) => {
141            debug!("anytls stream {}: handshake timed out", sid);
142            return;
143        }
144    };
145    if stream.report(None).await.is_err() {
146        return;
147    }
148    // Unique with the source, which is this connection's.
149    sess.stream_id = Some(StreamId::U64(sid as u64));
150    sess.destination = destination;
151    let _ = tx
152        .send(AnyBaseInboundTransport::Stream(Box::new(stream), sess))
153        .await;
154}

156/// The streams of one session, as they are opened.
157struct Incoming {
158    rx: mpsc::Receiver<AnyBaseInboundTransport>,
159    /// Kept for as long as streams may come.
160    _session: Arc<Session>,
161}
162
163impl FuturesStream for Incoming {
164    type Item = AnyBaseInboundTransport;
165
166    fn poll_next(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Self::Item>> {
167        self.rx.poll_recv(cx)
168    }
169}
