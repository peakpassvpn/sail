1//! The padding scheme: how many of a session's first writes are split up or
2//! padded out, and to which sizes.
3//!
4//! A scheme is lines of `key=value`. `stop=N` says padding ends at the N-th
5//! write; `K=...` gives the sizes of write K as comma-separated `min-max`
6//! ranges, each a record of that many bytes, and `c` marks, a check: once the
7//! payload has run out, a write stops at the next check instead of sending
8//! the records after it as padding. Write 0 is the authentication, which
9//! takes the first size of its line as the length of its padding.
10//!
11//! The server sends its scheme to any client whose `padding-md5` differs, so
12//! both ends match `anytls-go` and `sing-anytls` byte for byte in how the
13//! text is read and hashed.
14
15use std::collections::HashMap;
16
17use anyhow::{anyhow, Result};
18use md5::{Digest, Md5};
19use rand::Rng;

21/// The scheme every client starts with and every server defaults to.
22pub const DEFAULT_SCHEME: &str = "stop=8
230=30-30
241=100-400
252=400-500,c,500-1000,c,500-1000,c,500-1000,c,500-1000
263=9-9,500-1000
274=500-1000
285=500-1000
296=500-1000
307=500-1000";
31
32/// One element of a write's sizes.
33#[derive(Debug, Clone, Copy, PartialEq, Eq)]
34pub enum Size {
35    /// A record of this many bytes.
36    Record(usize),
37    /// A check: stop here if nothing is left to send.
38    Check,
39}
40
41/// The largest record a scheme may ask for: what the length of a waste
42/// frame can carry, with its header.
43const MAX_RECORD: u64 = u16::MAX as u64;
44
45#[derive(Debug, Clone, Copy, PartialEq, Eq)]
46enum Range {
47    Between(u64, u64),
48    Check,
49}
50
51/// A parsed padding scheme.
52#[derive(Debug, Clone)]
53pub struct PaddingScheme {
54    /// The text as given, which is what is hashed and what a server sends.
55    raw: Vec<u8>,
56    md5: String,
57    stop: u32,
58    writes: HashMap<u32, Vec<Range>>,
59}
60
61impl PaddingScheme {
62    /// The default scheme.
63    pub fn default_scheme() -> Self {
64        // The default scheme is a constant known to parse.
65        Self::parse(DEFAULT_SCHEME.as_bytes()).expect("the default padding scheme parses")
66    }
67
68    /// Reads a scheme the way the reference implementations do: only a
69    /// missing or unreadable `stop` makes it unusable, and a range that
70    /// does not read is skipped. This is for schemes a server pushes.
71    pub fn parse(raw: &[u8]) -> Option<Self> {
72        let map = string_map(raw);
73        let stop = map.get("stop")?.parse::<i64>().ok()?;
74        let stop = u32::try_from(stop).unwrap_or(0);
75        let mut writes = HashMap::new();
76        for (key, value) in &map {
77            let Ok(index) = key.parse::<u32>() else {
78                continue;
79            };
80            let ranges = value
81                .split(',')
82                .filter_map(|range| parse_range(range).ok().flatten())
83                .collect();
84            writes.insert(index, ranges);
85        }
86        Some(PaddingScheme {
87            raw: raw.to_vec(),
88            md5: hex_md5(raw),
89            stop,
90            writes,
91        })
92    }
93
94    /// Reads a scheme from a configuration, where anything that would be
95    /// skipped is a mistake instead.
96    pub fn parse_strict(raw: &str) -> Result<Self> {
97        if raw.len() > u16::MAX as usize {
98            return Err(anyhow!("longer than a frame can carry"));
99        }
100        let mut stop = None;
101        for line in raw.split('\n') {
102            let (key, value) = line
103                .split_once('=')
104                .ok_or_else(|| anyhow!("line \"{}\" is not key=value", line))?;
105            if key == "stop" {
106                let n = value
107                    .parse::<u32>()
108                    .map_err(|_| anyhow!("stop: \"{}\" is not a number", value))?;
109                stop = Some(n);
110                continue;
111            }
112            key.parse::<u32>()
113                .map_err(|_| anyhow!("key \"{}\" is neither stop nor a write number", key))?;
114            for range in value.split(',') {
115                parse_range(range).map_err(|e| anyhow!("{}: {}", key, e))?;
116            }
117        }
118        if stop.is_none() {
119            return Err(anyhow!("stop is missing"));
120        }
121        Self::parse(raw.as_bytes()).ok_or_else(|| anyhow!("does not parse"))
122    }
123
124    pub fn raw(&self) -> &[u8] {
125        &self.raw
126    }
127
128    /// The lowercase hex MD5 of the text, which a client reports.
129    pub fn md5(&self) -> &str {
130        &self.md5
131    }
132
133    /// The write at which padding ends.
134    pub fn stop(&self) -> u32 {
135        self.stop
136    }
137
138    /// The sizes of write `pkt`, drawn at random from its ranges. Empty for
139    /// a write the scheme says nothing about.
140    pub fn sizes(&self, pkt: u32) -> Vec<Size> {
141        let Some(ranges) = self.writes.get(&pkt) else {
142            return Vec::new();
143        };
144        let mut rng = rand::thread_rng();
145        ranges
146            .iter()
147            .map(|range| match *range {
148                Range::Check => Size::Check,
149                // As the reference draws it: from [min, max).
150                Range::Between(min, max) if min == max => Size::Record(min as usize),
151                Range::Between(min, max) => Size::Record(rng.gen_range(min..max) as usize),
152            })
153            .collect()
154    }
155
156    /// The length of the padding sent with the authentication.
157    pub fn auth_padding(&self) -> usize {
158        match self.sizes(0).first() {
159            Some(Size::Record(n)) => *n,
160            _ => 0,
161        }
162    }
163}
164
165/// One range: `Ok(None)` for one the reference would skip without saying,
166/// an error for one that does not read at all.
167fn parse_range(range: &str) -> Result<Option<Range>> {
168    if range == "c" {
169        return Ok(Some(Range::Check));
170    }
171    let (min, max) = range
172        .split_once('-')
173        .filter(|(_, max)| !max.contains('-'))
174        .ok_or_else(|| anyhow!("\"{}\" is neither min-max nor c", range))?;
175    let min = min
176        .parse::<i64>()
177        .map_err(|_| anyhow!("\"{}\": not a number", range))?;
178    let max = max
179        .parse::<i64>()
180        .map_err(|_| anyhow!("\"{}\": not a number", range))?;
181    let (min, max) = if min > max { (max, min) } else { (min, max) };
182    if min <= 0 || max <= 0 {
183        return Ok(None);
184    }
185    let (min, max) = (min as u64, max as u64);
186    if max > MAX_RECORD {
187        return Err(anyhow!("\"{}\": larger than {}", range, MAX_RECORD));
188    }
189    Ok(Some(Range::Between(min, max)))
190}

192/// `key=value` lines, as `util.StringMapFromBytes` reads them: a line
193/// without `=` is ignored and the value runs to the end of the line.
194pub fn string_map(raw: &[u8]) -> HashMap<String, String> {
195    String::from_utf8_lossy(raw)
196        .split('\n')
197        .filter_map(|line| {
198            line.split_once('=')
199                .map(|(k, v)| (k.to_string(), v.to_string()))
200        })
201        .collect()
202}
203
204fn hex_md5(raw: &[u8]) -> String {
205    Md5::digest(raw)
206        .iter()
207        .map(|b| format!("{:02x}", b))
208        .collect()
209}

211#[cfg(test)]
212mod tests {
213    use super::*;
214
215    #[test]
216    fn default_scheme_reads_and_hashes_as_the_reference() {
217        let scheme = PaddingScheme::default_scheme();
218        assert_eq!(scheme.stop(), 8);
219        // The md5 of the default scheme's text, as sing-anytls computes it.
220        assert_eq!(scheme.md5(), "75cff2ad89aadf5e257059ee571ebe11");
221        assert_eq!(scheme.auth_padding(), 30);
222        assert_eq!(scheme.sizes(3)[0], Size::Record(9));
223        let two = scheme.sizes(2);
224        assert_eq!(two.len(), 9);
225        assert_eq!(two[1], Size::Check);
226        match two[0] {
227            Size::Record(n) => assert!((400..500).contains(&n)),
228            Size::Check => panic!("expected a record"),
229        }
230        assert!(scheme.sizes(8).is_empty());
231    }
232
233    #[test]
234    fn lenient_parse_skips_what_the_reference_skips() {
235        let scheme = PaddingScheme::parse(b"stop=3\n0=5-5\n1=x-9,0-4,7-3,c\nnoise").unwrap();
236        assert_eq!(scheme.stop(), 3);
237        assert_eq!(scheme.auth_padding(), 5);
238        let one = scheme.sizes(1);
239        assert_eq!(one.len(), 2);
240        match one[0] {
241            Size::Record(n) => assert!((3..7).contains(&n)),
242            Size::Check => panic!("expected a record"),
243        }
244        assert_eq!(one[1], Size::Check);
245        assert!(PaddingScheme::parse(b"0=1-2").is_none());
246        assert!(PaddingScheme::parse(b"stop=x").is_none());
247        assert!(PaddingScheme::parse(b"").is_none());
248    }
249
250    #[test]
251    fn strict_parse_rejects_mistakes() {
252        assert!(PaddingScheme::parse_strict(DEFAULT_SCHEME).is_ok());
253        assert!(PaddingScheme::parse_strict("stop=2\n0=1-2\n1=3").is_err());
254        assert!(PaddingScheme::parse_strict("0=1-2").is_err());
255        assert!(PaddingScheme::parse_strict("stop=2\nfoo=1-2").is_err());
256        assert!(PaddingScheme::parse_strict("stop=2\n1=1-70000").is_err());
257        assert!(PaddingScheme::parse_strict("stop=2\n1=1-2,d").is_err());
258        assert!(PaddingScheme::parse_strict("stop=2\n\n1=1-2").is_err());
259    }
260
261    #[test]
262    fn string_map_reads_as_the_reference() {
263        let map = string_map(b"v=2\nclient=sing-box/1.13\npadding-md5=ab=cd\nbad");
264        assert_eq!(map["v"], "2");
265        assert_eq!(map["client"], "sing-box/1.13");
266        assert_eq!(map["padding-md5"], "ab=cd");
267        assert_eq!(map.len(), 3);
268    }
