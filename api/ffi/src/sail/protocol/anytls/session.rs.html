1//! A session: many streams over one authenticated TLS connection.
2//!
3//! A reader task takes frames off the connection and hands stream data to
4//! each stream's queue; a writer task owns the other half and writes what
5//! the streams and the reader queue for it, padding the client's first
6//! writes as the padding scheme says. Every queue is bounded, so a stream
7//! whose reader falls behind stalls its session, as it does in the
8//! reference implementation, rather than growing without limit.
9//!
10//! A stream closes whole: `FIN` tells the peer to close its end without
11//! answering, so a stream that has sent one reads no further either, and
12//! one that has received one writes no further.
13

35/// The padding scheme a client uses, which its server may replace. Shared
36/// by all of the client's sessions.
37pub type PaddingCell = Arc<RwLock<Arc<PaddingScheme>>>;
38
39/// Frames or runs of frames queued for the writer.
40const WRITE_QUEUE: usize = 64;
41/// Data frames queued for a stream's reader.
42const STREAM_QUEUE: usize = 32;
43/// The most a stream writes into one frame. Smaller than a frame could be,
44/// so that the write queue stays small.
45const MAX_WRITE: usize = 16 * 1024;
46/// The most the writer coalesces into one write once padding is over.
47const MAX_COALESCE: usize = 64 * 1024;
48/// Streams open at once on one session. A server refuses more.
49pub const MAX_STREAMS: usize = 1024;
50/// How long a client waits for a `SYNACK` on a reused session before it
51/// takes the session to be stuck, as the reference client does.
52const SYNACK_TIMEOUT: Duration = Duration::from_secs(3);
53
54fn closed_error() -> io::Error {
55    io::Error::new(io::ErrorKind::BrokenPipe, "anytls stream closed")
56}

58/// What a stream and its session share.
59#[derive(Default)]
60struct StreamState {
61    /// The peer sent `FIN`.
62    fin_received: AtomicBool,
63    /// The server failed to open the stream, and said why.
64    error: Mutex<Option<String>>,
65}
66
67struct Slot {
68    data: mpsc::Sender<Bytes>,
69    state: Arc<StreamState>,
70}
71
72/// What the tasks and the session share.
73struct Shared {
74    streams: Mutex<HashMap<u32, Slot>>,
75    cancel: CancellationToken,
76    peer_version: AtomicU8,
77    /// Bumped by every `SYNACK`, so that a pending timeout can tell whether
78    /// one has come since it was armed.
79    synack_generation: AtomicU64,
80}
81
82impl Shared {
83    fn close(&self) {
84        self.cancel.cancel();
85        // Dropping the queues' senders ends every stream's reads.
86        if let Ok(mut streams) = self.streams.lock() {
87            streams.clear();
88        }
89    }
90
91    fn remove(&self, sid: u32) -> Option<Slot> {
92        self.streams.lock().ok()?.remove(&sid)
93    }
94
95    fn sender(&self, sid: u32) -> Option<mpsc::Sender<Bytes>> {
96        self.streams
97            .lock()
98            .ok()?
99            .get(&sid)
100            .map(|slot| slot.data.clone())
101    }
102}

131impl Session {
132    /// A client session over `conn`, which has been authenticated.
133    pub fn client(conn: AnyStream, padding: PaddingCell) -> Arc<Session> {
134        let md5 = padding
135            .read()
136            .map(|p| p.md5().to_string())
137            .unwrap_or_default();
138        let settings = frame::encode(
139            CMD_SETTINGS,
140            0,
141            &frame::settings(&[
142                ("v", &VERSION.to_string()),
143                ("client", CLIENT_NAME),
144                ("padding-md5", &md5),
145            ]),
146        );
147        Self::start(conn, Role::Client { padding }, Some(settings))
148    }
149
150    /// A server session over `conn`, which has been authenticated.
151    pub fn server(
152        conn: AnyStream,
153        padding: Arc<PaddingScheme>,
154        on_stream: OnStream,
155    ) -> Arc<Session> {
156        Self::start(conn, Role::Server { padding, on_stream }, None)
157    }
158
159    fn start(conn: AnyStream, role: Role, settings: Option<BytesMut>) -> Arc<Session> {
160        let shared = Arc::new(Shared {
161            streams: Mutex::new(HashMap::new()),
162            cancel: CancellationToken::new(),
163            peer_version: AtomicU8::new(0),
164            synack_generation: AtomicU64::new(0),
165        });
166        let (tx, rx) = mpsc::channel(WRITE_QUEUE);
167        let (r, w) = tokio::io::split(conn);
168        let padding = match &role {
169            Role::Client { padding } => Some(padding.clone()),
170            Role::Server { .. } => None,
171        };
172        let session = Arc::new(Session {
173            shared: shared.clone(),
174            tx: tx.clone(),
175            next_sid: AtomicU32::new(0),
176            settings: Mutex::new(settings),
177        });
178
179        let writer_shared = shared.clone();
180        tokio::spawn(async move {
181            let cancel = writer_shared.cancel.clone();
182            tokio::select! {
183                result = write_loop(w, rx, padding) => {
184                    if let Err(e) = result {
185                        debug!("anytls session write failed: {}", e);
186                    }
187                }
188                _ = cancel.cancelled() => {}
189            }
190            writer_shared.close();
191        });
192
193        let reader = Reader {
194            shared: shared.clone(),
195            tx,
196            role,
197            session: Arc::downgrade(&session),
198        };
199        tokio::spawn(async move {
200            let cancel = shared.cancel.clone();
201            tokio::select! {
202                result = reader.run(BufReader::new(r)) => {
203                    if let Err(e) = result {
204                        debug!("anytls session read ended: {}", e);
205                    }
206                }
207                _ = cancel.cancelled() => {}
208            }
209            shared.close();
210        });
211        session
212    }
213
214    pub fn is_closed(&self) -> bool {
215        self.shared.cancel.is_cancelled()
216    }
217
218    pub fn close(&self) {
219        self.shared.close();
220    }

222    /// Opens a stream whose first data is `first`: the destination, and
223    /// whatever else is known to follow it.
224    pub async fn open_stream(self: &Arc<Self>, first: &[u8]) -> io::Result<Stream> {
225        if self.is_closed() {
226            return Err(closed_error());
227        }
228        let sid = self
229            .next_sid
230            .fetch_add(1, Ordering::Relaxed)
231            .wrapping_add(1);
232        let (data_tx, data_rx) = mpsc::channel(STREAM_QUEUE);
233        let state = Arc::new(StreamState::default());
234        {
235            let mut streams = self.shared.streams.lock().map_err(|_| closed_error())?;
236            if streams.len() >= MAX_STREAMS {
237                return Err(io::Error::other("anytls session has too many streams"));
238            }
239            streams.insert(
240                sid,
241                Slot {
242                    data: data_tx,
243                    state: state.clone(),
244                },
245            );
246        }
247        let stream = Stream::new(self.clone(), sid, data_rx, state);
248
249        // Settings, SYN and the destination go out as one write, which is
250        // what the padding scheme counts as the first.
251        let mut unit = self
252            .settings
253            .lock()
254            .ok()
255            .and_then(|mut s| s.take())
256            .unwrap_or_default();
257        frame::put(&mut unit, CMD_SYN, sid, &[]);
258        frame::put(&mut unit, CMD_PSH, sid, first);
259
260        if sid >= 2 && self.shared.peer_version.load(Ordering::Relaxed) >= 2 {
261            self.watch_synack();
262        }
263        self.tx.send(unit).await.map_err(|_| closed_error())?;
264        Ok(stream)
265    }
266
267    /// Closes the session unless a `SYNACK` arrives in time: a reused
268    /// session that has gone quiet is taken to be stuck.
269    fn watch_synack(&self) {
270        let generation = self
271            .shared
272            .synack_generation
273            .fetch_add(1, Ordering::Relaxed)
274            .wrapping_add(1);
275        let shared = Arc::downgrade(&self.shared);
276        let cancel = self.shared.cancel.clone();
277        tokio::spawn(async move {
278            tokio::select! {
279                _ = tokio::time::sleep(SYNACK_TIMEOUT) => {}
280                _ = cancel.cancelled() => return,
281            }
282            if let Some(shared) = shared.upgrade() {
283                if shared.synack_generation.load(Ordering::Relaxed) == generation {
284                    debug!("anytls session got no SYNACK in time, closing it");
285                    shared.close();
286                }
287            }
288        });
289    }
290
291    async fn send(&self, unit: BytesMut) -> io::Result<()> {
292        self.tx.send(unit).await.map_err(|_| closed_error())
293    }
294}

296struct Reader {
297    shared: Arc<Shared>,
298    tx: mpsc::Sender<BytesMut>,
299    role: Role,
300    session: Weak<Session>,
301}
302
303impl Reader {
304    async fn send(&self, cmd: u8, sid: u32, data: &[u8]) -> io::Result<()> {
305        self.tx
306            .send(frame::encode(cmd, sid, data))
307            .await
308            .map_err(|_| closed_error())
309    }
310
311    async fn run<R: AsyncRead + Unpin>(&self, mut r: R) -> io::Result<()> {
312        let mut settings_received = false;
313        let mut raw = [0u8; HEADER_LEN];
314        loop {
315            r.read_exact(&mut raw).await?;
316            let header = Header::decode(&raw);
317            let sid = header.sid;
318            let len = header.len as usize;
319            match header.cmd {
320                CMD_PSH => {
321                    let data = read_data(&mut r, len).await?;
322                    if data.is_empty() {
323                        continue;
324                    }
325                    if let Some(sender) = self.shared.sender(sid) {
326                        // A stream that has gone away drops its data.
327                        let _ = sender.send(data).await;
328                    }
329                }
330                CMD_SYN => {
331                    let Role::Server { on_stream, .. } = &self.role else {
332                        continue;
333                    };
334                    if !settings_received {
335                        self.send(CMD_ALERT, 0, b"client did not send its settings")
336                            .await?;
337                        return Ok(());
338                    }
339                    let Some(session) = self.session.upgrade() else {
340                        return Ok(());
341                    };
342                    let (data_tx, data_rx) = mpsc::channel(STREAM_QUEUE);
343                    let state = Arc::new(StreamState::default());
344                    let accepted = {
345                        let mut streams = self.shared.streams.lock().map_err(|_| closed_error())?;
346                        if streams.contains_key(&sid) {
347                            continue;
348                        }
349                        if streams.len() >= MAX_STREAMS {
350                            false
351                        } else {
352                            streams.insert(
353                                sid,
354                                Slot {
355                                    data: data_tx,
356                                    state: state.clone(),
357                                },
358                            );
359                            true
360                        }
361                    };
362                    if !accepted {
363                        warn!("anytls session has too many streams, refusing one");
364                        self.send(CMD_FIN, sid, &[]).await?;
365                        continue;
366                    }
367                    trace!("anytls stream {} opened", sid);
368                    on_stream(Stream::new(session, sid, data_rx, state));
369                }
370                CMD_SYNACK => {
371                    self.shared
372                        .synack_generation
373                        .fetch_add(1, Ordering::Relaxed);
374                    if len > 0 {
375                        let data = read_data(&mut r, len).await?;
376                        if let Some(slot) = self.shared.remove(sid) {
377                            let message = String::from_utf8_lossy(&data).into_owned();
378                            if let Ok(mut error) = slot.state.error.lock() {
379                                *error = Some(message);
380                            }
381                        }
382                    }
383                }
384                CMD_FIN => {
385                    if let Some(slot) = self.shared.remove(sid) {
386                        slot.state.fin_received.store(true, Ordering::Release);
387                    }
388                }
389                CMD_WASTE => {
390                    skip(&mut r, len).await?;
391                }
392                CMD_SETTINGS => {
393                    if len == 0 {
394                        continue;
395                    }
396                    let data = read_data(&mut r, len).await?;
397                    if let Role::Server { padding, .. } = &self.role {
398                        settings_received = true;
399                        let settings = string_map(&data);
400                        if settings.get("padding-md5").map(String::as_str) != Some(padding.md5()) {
401                            self.send(CMD_UPDATE_PADDING_SCHEME, 0, padding.raw())
402                                .await?;
403                        }
404                        if let Some(v) = settings.get("v").and_then(|v| v.parse::<u8>().ok()) {
405                            if v >= 2 {
406                                self.shared.peer_version.store(v, Ordering::Relaxed);
407                                let settings = frame::settings(&[("v", &VERSION.to_string())]);
408                                self.send(CMD_SERVER_SETTINGS, 0, &settings).await?;
409                            }
410                        }
411                    }
412                }
413                CMD_ALERT => {
414                    if len == 0 {
415                        continue;
416                    }
417                    let data = read_data(&mut r, len).await?;
418                    if matches!(self.role, Role::Client { .. }) {
419                        warn!(
420                            "anytls alert from server: {}",
421                            String::from_utf8_lossy(&data)
422                        );
423                    }
424                    return Ok(());
425                }
426                CMD_UPDATE_PADDING_SCHEME => {
427                    if len == 0 {
428                        continue;
429                    }
430                    let data = read_data(&mut r, len).await?;
431                    if let Role::Client { padding } = &self.role {
432                        match PaddingScheme::parse(&data) {
433                            Some(scheme) => {
434                                debug!("anytls padding scheme updated to {}", scheme.md5());
435                                if let Ok(mut current) = padding.write() {
436                                    *current = Arc::new(scheme);
437                                }
438                            }
439                            None => {
440                                warn!("anytls server sent a padding scheme that does not parse")
441                            }
442                        }
443                    }
444                }
445                CMD_HEART_REQUEST => {
446                    self.send(CMD_HEART_RESPONSE, sid, &[]).await?;
447                }
448                CMD_HEART_RESPONSE => {}
449                CMD_SERVER_SETTINGS => {
450                    if len == 0 {
451                        continue;
452                    }
453                    let data = read_data(&mut r, len).await?;
454                    if let Role::Client { .. } = &self.role {
455                        let settings = string_map(&data);
456                        if let Some(v) = settings.get("v").and_then(|v| v.parse::<u8>().ok()) {
457                            self.shared.peer_version.store(v, Ordering::Relaxed);
458                        }
459                    }
460                }
461                // Unknown commands carry no data.
462                _ => {}
463            }
464        }
465    }
466}

468async fn read_data<R: AsyncRead + Unpin>(r: &mut R, len: usize) -> io::Result<Bytes> {
469    let mut data = vec![0u8; len];
470    r.read_exact(&mut data).await?;
471    Ok(Bytes::from(data))
472}
473
474async fn skip<R: AsyncRead + Unpin>(r: &mut R, mut len: usize) -> io::Result<()> {
475    let mut scratch = [0u8; 1024];
476    while len > 0 {
477        let n = len.min(scratch.len());
478        r.read_exact(&mut scratch[..n]).await?;
479        len -= n;
480    }
481    Ok(())
482}

484async fn write_loop(
485    mut w: WriteHalf<AnyStream>,
486    mut rx: mpsc::Receiver<BytesMut>,
487    padding: Option<PaddingCell>,
488) -> io::Result<()> {
489    let mut pkt: u32 = 0;
490    let mut padding = padding;
491    while let Some(mut unit) = rx.recv().await {
492        if let Some(cell) = &padding {
493            pkt = pkt.wrapping_add(1);
494            let scheme = cell
495                .read()
496                .map(|s| s.clone())
497                .map_err(|_| io::Error::other("padding scheme poisoned"))?;
498            if pkt < scheme.stop() {
499                write_padded(&mut w, unit, &scheme.sizes(pkt)).await?;
500                continue;
501            }
502            padding = None;
503        }
504        while unit.len() < MAX_COALESCE {
505            match rx.try_recv() {
506                Ok(more) => unit.extend_from_slice(&more),
507                Err(_) => break,
508            }
509        }
510        w.write_all(&unit).await?;
511        w.flush().await?;
512    }
513    Ok(())
514}

516/// Writes `unit` in the records `sizes` asks for, as `writeConn` in the
517/// reference does: payload first, padded out with waste frames where the
518/// payload runs short.
519async fn write_padded<W: AsyncWrite + Unpin>(
520    w: &mut W,
521    mut unit: BytesMut,
522    sizes: &[Size],
523) -> io::Result<()> {
524    for size in sizes {
525        let record = match *size {
526            Size::Check if unit.is_empty() => break,
527            Size::Check => continue,
528            Size::Record(n) => n,
529        };
530        let remaining = unit.len();
531        if remaining > record {
532            let head = unit.split_to(record);
533            w.write_all(&head).await?;
534        } else if remaining > 0 {
535            let padding = record as isize - remaining as isize - HEADER_LEN as isize;
536            if padding > 0 {
537                unit.extend_from_slice(&frame::waste(padding as usize));
538            }
539            w.write_all(&unit).await?;
540            unit.clear();
541        } else {
542            w.write_all(&frame::waste(record)).await?;
543        }
544        w.flush().await?;
545    }
546    if !unit.is_empty() {
547        w.write_all(&unit).await?;
548        w.flush().await?;
549    }
550    Ok(())
551}

553/// One stream of a session.
554pub struct Stream {
555    session: Arc<Session>,
556    sid: u32,
557    rx: mpsc::Receiver<Bytes>,
558    pending: Bytes,
559    state: Arc<StreamState>,
560    tx: PollSender<BytesMut>,
561    fin_sent: bool,
562    /// Run when the stream is dropped: how a client puts its session back.
563    on_drop: Option<Box<dyn FnOnce() + Send + Sync>>,
564}
565
566impl Stream {
567    fn new(
568        session: Arc<Session>,
569        sid: u32,
570        rx: mpsc::Receiver<Bytes>,
571        state: Arc<StreamState>,
572    ) -> Self {
573        let tx = PollSender::new(session.tx.clone());
574        Stream {
575            session,
576            sid,
577            rx,
578            pending: Bytes::new(),
579            state,
580            tx,
581            fin_sent: false,
582            on_drop: None,
583        }
584    }
585
586    pub fn id(&self) -> u32 {
587        self.sid
588    }
589
590    pub fn set_on_drop(&mut self, on_drop: Box<dyn FnOnce() + Send + Sync>) {
591        self.on_drop = Some(on_drop);
592    }
593
594    /// Tells a version 2 client whether the stream opened: `None` for
595    /// success, or the error.
596    pub async fn report(&self, error: Option<&str>) -> io::Result<()> {
597        if self.session.shared.peer_version.load(Ordering::Relaxed) < 2 {
598            return Ok(());
599        }
600        let data = error.map(str::as_bytes).unwrap_or_default();
601        let data = &data[..data.len().min(MAX_DATA)];
602        self.session
603            .send(frame::encode(CMD_SYNACK, self.sid, data))
604            .await
605    }
606
607    /// Stops taking data for this stream.
608    fn detach(&mut self) {
609        self.fin_sent = true;
610        self.session.shared.remove(self.sid);
611    }
612
613    fn write_closed(&self) -> bool {
614        self.fin_sent || self.state.fin_received.load(Ordering::Acquire) || self.session.is_closed()
615    }
616}

618impl AsyncRead for Stream {
619    fn poll_read(
620        mut self: Pin<&mut Self>,
621        cx: &mut Context<'_>,
622        buf: &mut ReadBuf<'_>,
623    ) -> Poll<io::Result<()>> {
624        loop {
625            if !self.pending.is_empty() {
626                let n = self.pending.len().min(buf.remaining());
627                let chunk = self.pending.split_to(n);
628                buf.put_slice(&chunk);
629                return Poll::Ready(Ok(()));
630            }
631            match ready!(self.rx.poll_recv(cx)) {
632                Some(data) => self.pending = data,
633                None => {
634                    if let Some(error) = self.state.error.lock().ok().and_then(|e| e.clone()) {
635                        return Poll::Ready(Err(io::Error::new(
636                            io::ErrorKind::ConnectionRefused,
637                            format!("anytls server: {}", error),
638                        )));
639                    }
640                    if self.fin_sent || self.state.fin_received.load(Ordering::Acquire) {
641                        return Poll::Ready(Ok(()));
642                    }
643                    return Poll::Ready(Err(io::Error::new(
644                        io::ErrorKind::ConnectionAborted,
645                        "anytls session closed",
646                    )));
647                }
648            }
649        }
650    }
651}

653impl AsyncWrite for Stream {
654    fn poll_write(
655        mut self: Pin<&mut Self>,
656        cx: &mut Context<'_>,
657        buf: &[u8],
658    ) -> Poll<io::Result<usize>> {
659        if self.write_closed() {
660            return Poll::Ready(Err(closed_error()));
661        }
662        if buf.is_empty() {
663            return Poll::Ready(Ok(0));
664        }
665        ready!(self.tx.poll_reserve(cx)).map_err(|_| closed_error())?;
666        let n = buf.len().min(MAX_WRITE);
667        let sid = self.sid;
668        self.tx
669            .send_item(frame::encode(CMD_PSH, sid, &buf[..n]))
670            .map_err(|_| closed_error())?;
671        Poll::Ready(Ok(n))
672    }
673
674    fn poll_flush(self: Pin<&mut Self>, _cx: &mut Context<'_>) -> Poll<io::Result<()>> {
675        Poll::Ready(Ok(()))
676    }
677
678    fn poll_shutdown(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<io::Result<()>> {
679        if self.fin_sent {
680            return Poll::Ready(Ok(()));
681        }
682        if self.write_closed() {
683            self.detach();
684            return Poll::Ready(Ok(()));
685        }
686        if ready!(self.tx.poll_reserve(cx)).is_ok() {
687            let sid = self.sid;
688            let _ = self.tx.send_item(frame::encode(CMD_FIN, sid, &[]));
689        }
690        self.detach();
691        Poll::Ready(Ok(()))
692    }
693}

695impl Drop for Stream {
696    fn drop(&mut self) {
697        if !self.write_closed() {
698            let fin = frame::encode(CMD_FIN, self.sid, &[]);
699            if let Err(mpsc::error::TrySendError::Full(fin)) = self.session.tx.try_send(fin) {
700                if let Ok(runtime) = tokio::runtime::Handle::try_current() {
701                    let tx = self.session.tx.clone();
702                    runtime.spawn(async move {
703                        let _ = tx.send(fin).await;
704                    });
705                }
706            }
707        }
708        self.session.shared.remove(self.sid);
709        if let Some(on_drop) = self.on_drop.take() {
710            on_drop();
711        }
712    }
713}

715/// The length of the password's SHA-256 a client starts with.
716pub const AUTH_HASH_LEN: usize = 32;
717
718/// Reads the rest of the authentication a client starts with, after the
719/// password's SHA-256: the padding's length, and the padding.
720pub async fn read_auth_padding<R: AsyncRead + Unpin>(r: &mut R) -> io::Result<()> {
721    let len = r.read_u16().await? as usize;
722    skip(r, len).await
723}
724
725/// The authentication a client starts with.
726pub fn auth(hash: &[u8; 32], padding: usize) -> Vec<u8> {
727    let padding = padding.min(u16::MAX as usize);
728    let mut buf = Vec::with_capacity(32 + 2 + padding);
729    buf.extend_from_slice(hash);
730    buf.extend_from_slice(&(padding as u16).to_be_bytes());
731    buf.resize(32 + 2 + padding, 0);
732    buf
733}

799    /// A client and a server session over a pipe, the server echoing every

801    #[test]
802    fn sessions_carry_streams_both_ways() {
803        runtime().block_on(async {
804            let (a, b) = tokio::io::duplex(64 * 1024);
805            let server = Session::server(
806                Box::new(b),
807                Arc::new(PaddingScheme::parse(b"stop=1").unwrap()),
808                Box::new(|mut stream: Stream| {
809                    tokio::spawn(async move {
810                        stream.report(None).await.unwrap();
811                        let mut buf = vec![0u8; 4096];
812                        loop {
813                            let n = stream.read(&mut buf).await.unwrap();
814                            if n == 0 {
815                                break;
816                            }
817                            stream.write_all(&buf[..n]).await.unwrap();
818                        }
819                    });
820                }),
821            );
822            let padding = cell();
823            let client = Session::client(Box::new(a), padding.clone());
824            for i in 0..3u8 {

826                let mut echo = [0u8; 1];

837            }
