31/// UDP sessions one connection carries at most.
32const MAX_UDP_SESSIONS: usize = 1024;
33/// Packets queued for a UDP session before more are dropped.
34const UDP_SESSION_QUEUE: usize = 256;

68    /// The live connection, dialling one if there is none.
69    pub async fn connection(&self) -> io::Result<Arc<Connection>> {
70        let mut guard = self.conn.lock().await;
71        if let Some(conn) = guard.as_ref() {
72            if conn.conn.close_reason().is_none() {
73                return Ok(conn.clone());
74            }
75            debug!(
76                "hysteria2 connection closed: {:?}",
77                conn.conn.close_reason()
78            );
79        }
80        *guard = None;
81        let conn = Arc::new(
82            self.connect()
83                .await
84                .map_err(|e| io::Error::other(format!("hysteria2 connect failed: {:#}", e)))?,
85        );
86        *guard = Some(conn.clone());
87        Ok(conn)
88    }
89
90    /// Forgets `conn` if it is the current connection, so that the next
91    /// request dials a new one.
92    pub async fn discard(&self, conn: &Arc<Connection>) {
93        let mut guard = self.conn.lock().await;
94        if guard.as_ref().is_some_and(|c| Arc::ptr_eq(c, conn)) {
95            *guard = None;
96        }
97    }
98
99    /// Opens a proxied TCP stream to `destination`, with `payload` sent
100    /// along with the request.
101    pub async fn open_stream(
102        &self,
103        destination: &SocksAddr,
104        payload: &[u8],
105    ) -> io::Result<QuicStream> {
106        let conn = self.connection().await?;
107        let (mut send, mut recv) = match conn.conn.open_bi().await {
108            Ok(s) => s,
109            Err(e) => {
110                self.discard(&conn).await;
111                return Err(io::Error::other(e));
112            }
113        };
114        send.write_all(&proto::tcp_request(destination, payload))
115            .await
116            .map_err(io::Error::other)?;
117        timeout(
118            self.options.dial.connect_timeout,
119            proto::read_tcp_response(&mut recv),
120        )
121        .await
122        .map_err(|_| io::Error::new(io::ErrorKind::TimedOut, "hysteria2 TCP response"))??;
123        Ok(QuicStream::new(send, recv))
124    }

126    async fn connect(&self) -> Result<Connection> {
127        let o = &self.options;
128        let ips = o
129            .dns_client
130            .load_full()
131            .direct_lookup(&o.server)
132            .await
133            .with_context(|| format!("lookup {}", o.server))?;
134        let mut last_err = anyhow!("could not resolve {} to any address", o.server);
135        for ip in ips {
136            match timeout(o.dial.connect_timeout, self.connect_to(ip)).await {
137                Ok(Ok(conn)) => return Ok(conn),
138                Ok(Err(e)) => last_err = e,
139                Err(_) => last_err = anyhow!("connect {} timed out", ip),
140            }
141        }
142        Err(last_err)
143    }
144
145    async fn new_socket(&self, ip: IpAddr) -> io::Result<Arc<dyn quinn::AsyncUdpSocket>> {
146        let o = &self.options;
147        quic::wrap_socket(bind(ip, &o.dial).await?, o.obfs.as_ref())
148    }

150    async fn connect_to(&self, ip: IpAddr) -> Result<Connection> {
151        let o = &self.options;
152        let socket = self.new_socket(ip).await?;
153        let (socket, remote, hop): (Arc<dyn quinn::AsyncUdpSocket>, _, _) = if o.ports.len() > 1 {
154            let hop = Arc::new(HopSocket::new(ip, o.ports.clone(), socket));
155            let remote = hop.virtual_addr();
156            (hop.clone(), remote, Some(hop))
157        } else {
158            (socket, SocketAddr::new(ip, o.ports[0]), None)
159        };
160        let endpoint = endpoint_on(socket, None)?;
161        let congestion = CongestionHandle::default();
162        let mut config = quinn::ClientConfig::new(o.crypto.clone());
163        config.transport_config(Arc::new(quic::transport_config(
164            &o.tuning,
165            Side::Client,
166            &congestion,
167        )));
168        let conn = endpoint
169            .connect_with(config, remote, &o.server_name)?
170            .await
171            .with_context(|| format!("quic connect {}", remote))?;
172        trace!("hysteria2 connected to {}", remote);
173
174        let mut tasks = Tasks::default();
175        let control = quic::open_control_stream(&conn).await?;
176        tasks.spawn(quic::drain_uni_streams(conn.clone()));
177
178        let auth = self.authenticate(&conn).await?;
179        // The server tells what it can receive, and we send at most that.
180        // Brutal needs a rate of our own too: without `up_mbps`, BBR.
181        let tx = match auth.rx {
182            Some(rx) if rx > 0 && rx <= o.send_bps => rx,
183            _ => o.send_bps,
184        };
185        if auth.rx.is_some() && tx > 0 {
186            debug!("hysteria2 sends with brutal at {} bytes/s", tx);
187            congestion.set_brutal(tx);
188        } else {
189            congestion.set_bbr();
190        }
191
192        let sessions = Arc::new(Sessions::default());
193        if auth.udp {
194            tasks.spawn(receive_datagrams(conn.clone(), sessions.clone()));
195        }
196        if let (Some(hop), Some(interval)) = (hop, o.hop_interval) {
197            tasks.spawn(hop_ports(
198                conn.clone(),
199                hop,
200                interval,
201                ip,
202                o.dial.clone(),
203                o.obfs.clone(),
204            ));
205        }
206        Ok(Connection {
207            conn,
208            _endpoint: endpoint,
209            _control: control,
210            udp: auth.udp,
211            sessions,
212            next_session: AtomicU32::new(0),
213            next_packet: AtomicU32::new(0),
214            _tasks: tasks,
215        })
216    }

218    /// The HTTP/3 request that makes this connection a proxy connection.
219    async fn authenticate(&self, conn: &quinn::Connection) -> Result<Auth> {
220        let o = &self.options;
221        let (mut send, mut recv) = conn.open_bi().await?;
222        let rx = o.recv_bps.to_string();
223        let padding = proto::padding(proto::AUTH_REQUEST_PADDING);
224        h3::write_headers(
225            &mut send,
226            &[
227                (":method", "POST"),
228                (":scheme", "https"),
229                (":authority", proto::AUTH_HOST),
230                (":path", proto::AUTH_PATH),
231                (proto::HEADER_AUTH, &o.password),
232                (proto::HEADER_CC_RX, &rx),
233                (proto::HEADER_PADDING, &padding),
234            ],
235        )
236        .await?;
237        send.finish()?;
238        let fields = h3::read_headers(&mut recv, None).await?;
239        let status = h3::field(&fields, ":status").unwrap_or_default();
240        if status != proto::STATUS_AUTH_OK.to_string() {
241            return Err(anyhow!("authentication failed, status {}", status));
242        }
243        let udp =
244            h3::field(&fields, proto::HEADER_UDP).is_some_and(|v| v.eq_ignore_ascii_case("true"));
245        let rx = match h3::field(&fields, proto::HEADER_CC_RX) {
246            Some("auto") => None,
247            Some(v) => Some(v.parse().unwrap_or(0)),
248            None => Some(0),
249        };
250        Ok(Auth { udp, rx })
251    }
252}

262/// Tasks of a connection, stopped with it.
263#[derive(Default)]
264struct Tasks(Vec<AbortHandle>);
265
266impl Tasks {
267    fn spawn<F: std::future::Future<Output = ()> + Send + 'static>(&mut self, task: F) {
268        let (handle, registration) = AbortHandle::new_pair();
269        tokio::spawn(Abortable::new(task, registration));
270        self.0.push(handle);
271    }
272}
273
274impl Drop for Tasks {
275    fn drop(&mut self) {
276        for handle in &self.0 {
277            handle.abort();
278        }
279    }
280}

282/// A packet for a UDP session, and where it came from.
283pub type Packet = (Vec<u8>, SocksAddr);
284
285struct SessionEntry {
286    tx: mpsc::Sender<Packet>,
287    defragger: Defragger,
288}
289
290#[derive(Default)]
291struct Sessions(Mutex<HashMap<u32, SessionEntry>>);
292
293impl Sessions {
294    fn lock(&self) -> std::sync::MutexGuard<'_, HashMap<u32, SessionEntry>> {
295        self.0.lock().unwrap_or_else(|e| e.into_inner())
296    }
297}

299pub struct Connection {
300    pub conn: quinn::Connection,
301    _endpoint: quinn::Endpoint,
302    /// Our HTTP/3 control stream, open while the connection is.
303    _control: quinn::SendStream,
304    udp: bool,
305    sessions: Arc<Sessions>,
306    next_session: AtomicU32,
307    next_packet: AtomicU32,
308    _tasks: Tasks,
309}
310
311impl Connection {
312    /// A new UDP session: its ID, and where its packets arrive.
313    pub fn open_session(self: &Arc<Self>) -> io::Result<(UdpSession, mpsc::Receiver<Packet>)> {
314        if !self.udp {
315            return Err(io::Error::other("hysteria2: UDP disabled by the server"));
316        }
317        let mut sessions = self.sessions.lock();
318        if sessions.len() >= MAX_UDP_SESSIONS {
319            return Err(io::Error::other("hysteria2: too many UDP sessions"));
320        }
321        let mut id = self.next_session.fetch_add(1, Ordering::Relaxed);
322        while sessions.contains_key(&id) {
323            id = self.next_session.fetch_add(1, Ordering::Relaxed);
324        }
325        let (tx, rx) = mpsc::channel(UDP_SESSION_QUEUE);
326        sessions.insert(
327            id,
328            SessionEntry {
329                tx,
330                defragger: Defragger::default(),
331            },
332        );
333        Ok((
334            UdpSession {
335                conn: self.clone(),
336                id,
337            },
338            rx,
339        ))
340    }
341}

343/// A UDP session of a connection, closed when dropped.
344pub struct UdpSession {
345    conn: Arc<Connection>,
346    id: u32,
347}
348
349impl UdpSession {
350    pub fn send(&self, payload: &[u8], destination: &SocksAddr) -> io::Result<()> {
351        let max = self
352            .conn
353            .conn
354            .max_datagram_size()
355            .ok_or_else(|| io::Error::other("hysteria2: the server takes no datagrams"))?;
356        let packet_id = self.conn.next_packet.fetch_add(1, Ordering::Relaxed) as u16;
357        let addr = proto::format_addr(destination);
358        for fragment in proto::fragment(self.id, packet_id, &addr, payload, max)? {
359            self.conn
360                .conn
361                .send_datagram(Bytes::from(fragment))
362                .map_err(io::Error::other)?;
363        }
364        Ok(())
365    }
366}
367
368impl Drop for UdpSession {
369    fn drop(&mut self) {
370        self.conn.sessions.lock().remove(&self.id);
371    }
372}

374/// Hands the datagrams of `conn` to their sessions.
375async fn receive_datagrams(conn: quinn::Connection, sessions: Arc<Sessions>) {
376    while let Ok(datagram) = conn.read_datagram().await {
377        let msg = match UdpMessage::decode(&datagram) {
378            Ok(msg) => msg,
379            Err(e) => {
380                debug!("hysteria2: invalid UDP message: {}", e);
381                continue;
382            }
383        };
384        let mut sessions = sessions.lock();
385        let Some(entry) = sessions.get_mut(&msg.session_id) else {
386            continue;
387        };
388        let Some(packet) = entry.defragger.feed(&msg) else {
389            continue;
390        };
391        let from = match proto::parse_addr(msg.addr) {
392            Ok(from) => from,
393            Err(e) => {
394                debug!("hysteria2: invalid UDP source: {}", e);
395                continue;
396            }
397        };
398        // A session that does not keep up loses packets, as UDP would.
399        let _ = entry.tx.try_send((packet, from));
400    }
401    // The connection is gone: its sessions see the end at once.
402    sessions.lock().clear();
403}

405/// Moves to another port every `interval`, until the connection closes.
406async fn hop_ports(
407    conn: quinn::Connection,
408    hop: Arc<HopSocket>,
409    interval: Duration,
410    ip: IpAddr,
411    dial: Arc<DialOptions>,
412    obfs: Option<Salamander>,
413) {
414    loop {
415        if timeout(interval, conn.closed()).await.is_ok() {
416            return;
417        }
418        match bind(ip, &dial)
419            .await
420            .and_then(|s| quic::wrap_socket(s, obfs.as_ref()))
421        {
422            Ok(socket) => {
423                hop.hop(socket);
424                trace!("hysteria2: hopped ports");
425            }
426            Err(e) => debug!("hysteria2: port hop: {}", e),
427        }
428    }
