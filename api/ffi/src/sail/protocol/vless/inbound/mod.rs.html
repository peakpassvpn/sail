22pub(crate) fn register(registry: &mut InboundRegistry) {
23    registry.register(
24        "vless",
25        InboundFactory::standalone(build)
26            .with_blocks(Blocks {
27                tls: true,
28                transport: true,
29                multiplex: true,
30                ..Blocks::NONE
31            })
32            .checked_by(check),

36/// Vision reads the TLS records of the connection it runs on, and may
37/// switch to copying the raw TLS stream: a Vision user needs TLS (or
38/// REALITY) directly under VLESS, with no transport between.
39fn check(tag: &str, options: &Options, blocks: &InboundBlocks) -> Result<()> {
40    let vision = options
41        .get("users")
42        .and_then(|u| u.as_array())
43        .into_iter()
44        .flatten()
45        .position(|u| u.get("flow").and_then(|f| f.as_str()) == Some(FLOW_VISION));
46    match vision {
47        Some(i) if !blocks.has_tls() || blocks.transport.is_some() => Err(anyhow!(
48            "[{}] inbound: users[{}].flow: {} needs tls directly under vless, with no transport",
49            tag,
50            i,
51            FLOW_VISION
52        )),
53        _ => Ok(()),
54    }
55}

57#[derive(Deserialize)]
58#[serde(deny_unknown_fields)]
59struct VlessInboundOptions {
60    users: Vec<VlessUser>,
61    /// Where a connection that fails to authenticate is relayed.
62    #[serde(default)]
63    fallback: Option<FallbackServer>,
64    /// The same, by the ALPN the connection's TLS negotiated; the ones it
65    /// does not name go to `fallback`.
66    #[serde(default)]
67    fallback_for_alpn: HashMap<String, FallbackServer>,
68}
69
70#[derive(Deserialize)]
71#[serde(deny_unknown_fields)]
72struct VlessUser {
73    /// Who the user is to routing (`auth_user`), statistics and logs.
74    #[serde(default)]
75    name: Option<String>,
76    uuid: String,
77    /// `""` or `xtls-rprx-vision`.
78    #[serde(default)]
79    flow: String,
80}
81
82fn build(ctx: &InboundContext<'_>) -> Result<AnyInboundHandler> {
83    let options: VlessInboundOptions = ctx.options()?;
84    if options.users.is_empty() {
85        return Err(anyhow!("[{}] inbound: users: cannot be empty", ctx.tag));
86    }
87    let mut users = HashMap::new();
88    for (i, user) in options.users.into_iter().enumerate() {
89        let uuid = *uuid::Uuid::parse_str(&user.uuid)
90            .map_err(|e| anyhow!("[{}] inbound: users[{}].uuid: {}", ctx.tag, i, e))?
91            .as_bytes();
92        let flow = Flow::parse(&user.flow)
93            .map_err(|e| anyhow!("[{}] inbound: users[{}].flow: {}", ctx.tag, i, e))?;
94        let user = User {
95            name: user.name.map(Into::into),
96            flow,
97        };
98        if users.insert(uuid, user).is_some() {
99            return Err(anyhow!(
100                "[{}] inbound: users[{}].uuid: used by another user",
101                ctx.tag,
102                i
103            ));
104        }
105    }
106    let fallback = fallback::Fallback::new(ctx.tag, options.fallback, options.fallback_for_alpn)?;
107    let stream = Arc::new(StreamHandler::new(users, fallback));
108    Ok(Arc::new(Handler::new(
109        ctx.tag.to_owned(),
110        Some(stream),
111        None,
112    )))
113}

116mod check_tests {

119    fn check_json(users: serde_json::Value, blocks: serde_json::Value) -> Result<()> {
120        let options: Options =
121            serde_json::from_value(serde_json::json!({ "users": users })).unwrap();
122        let blocks = InboundBlocks::parse("v", &serde_json::from_value(blocks).unwrap()).unwrap();
123        check("v", &options, &blocks)
124    }
125
126    #[test]
127    fn a_vision_user_needs_tls_directly_under_it() {
128        let users = serde_json::json!([
129            { "uuid": "a", "flow": "" },
130            { "uuid": "b", "flow": FLOW_VISION },
131        ]);
132        let tls = serde_json::json!({ "enabled": true });
133        assert!(check_json(users.clone(), serde_json::json!({ "tls": tls })).is_ok());
134        let err = check_json(
135            users.clone(),
136            serde_json::json!({ "tls": tls, "transport": { "type": "ws" } }),
137        )
138        .unwrap_err();
139        assert!(err.to_string().contains("users[1].flow"), "{}", err);
140        assert!(check_json(users, serde_json::json!({})).is_err());
141    }
