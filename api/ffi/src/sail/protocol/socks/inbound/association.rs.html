1//! What a `UDP ASSOCIATE` keeps: its place among the inbound's
2//! associations, and which client its relay socket takes datagrams from.
3//!
4//! Each association has a UDP relay socket of its own (RFC 1928), bound when
5//! the client asks and closed when its control connection closes, so a
6//! datagram's association is the socket it arrived on: nothing is guessed.
7//! What is left to check is that the datagram came from the client:
8//!
9//! - its IP must be the control connection's, and
10//! - its address the one the client declared, when it declared one; else
11//!   the address of its first datagram, which the association keeps.
12
13use std::net::{IpAddr, SocketAddr};
14use std::sync::atomic::{AtomicUsize, Ordering};
15use std::sync::Arc;
16
17use crate::session::SocksAddr;
18
19/// How many associations, and so relay sockets, an inbound keeps at once.
20/// A client asking for one more is refused.
21pub const MAX_ASSOCIATIONS: usize = 4096;
22
23/// The live associations of an inbound, counted.
24#[derive(Default)]
25pub struct Associations {
26    live: AtomicUsize,
27}
28
29/// An association's place; dropping it frees the place.
30pub struct Slot(Arc<Associations>);
31
32impl Drop for Slot {
33    fn drop(&mut self) {
34        self.0.live.fetch_sub(1, Ordering::AcqRel);
35    }
36}
37
38impl Associations {
39    /// A place for one more association, unless all are taken.
40    pub fn acquire(self: &Arc<Self>) -> Option<Slot> {
41        self.live
42            .fetch_update(Ordering::AcqRel, Ordering::Acquire, |n| {
43                (n < MAX_ASSOCIATIONS).then_some(n + 1)
44            })
45            .ok()
46            .map(|_| Slot(self.clone()))
47    }
48
49    /// How many associations are live.
50    #[cfg(test)]
51    fn len(&self) -> usize {
52        self.live.load(Ordering::Acquire)
53    }
54}

56/// `addr` with an IPv4-mapped IPv6 address as the IPv4 one, as a dual-stack
57/// socket may report either.
58pub fn canonical(addr: SocketAddr) -> SocketAddr {
59    SocketAddr::new(addr.ip().to_canonical(), addr.port())
60}
61
62/// Which source an association's relay socket takes datagrams from.
63#[derive(Debug)]
64pub struct ClientFilter {
65    /// The control connection's IP.
66    peer_ip: IpAddr,
67    /// The client's address: declared, or learned from its first datagram.
68    client: Option<SocketAddr>,
69}
70
71impl ClientFilter {
72    /// The filter of an association asked for from `peer`, declaring
73    /// `declared`.
74    ///
75    /// An unspecified IP with a port is the peer's IP with that port. A
76    /// zero port, a domain, or an IP other than the peer's declares
77    /// nothing: datagrams must come from the peer's IP in any case, and a
78    /// client behind NAT declares the address it has behind it, which the
79    /// server never sees.
80    pub fn new(peer: SocketAddr, declared: &SocksAddr) -> Self {
81        let peer_ip = peer.ip().to_canonical();
82        let client = match declared {
83            SocksAddr::Ip(addr) if addr.port() != 0 => {
84                let ip = addr.ip().to_canonical();
85                if ip.is_unspecified() {
86                    Some(SocketAddr::new(peer_ip, addr.port()))
87                } else if ip == peer_ip {
88                    Some(SocketAddr::new(ip, addr.port()))
89                } else {
90                    None
91                }
92            }
93            _ => None,
94        };
95        ClientFilter { peer_ip, client }
96    }
97
98    /// Whether a datagram from `src` is the client's. The first from the
99    /// peer's IP pins the client's address when it declared none.
100    pub fn accept(&mut self, src: SocketAddr) -> bool {
101        let src = canonical(src);
102        if src.ip() != self.peer_ip {
103            return false;
104        }
105        match self.client {
106            Some(client) => client == src,
107            None => {
108                self.client = Some(src);
109                true
110            }
111        }
112    }
113}

115#[cfg(test)]
116mod tests {
117    use super::*;
118
119    fn addr(s: &str) -> SocketAddr {
120        s.parse().unwrap()
121    }
122
123    #[test]
124    fn a_declared_address_is_matched_exactly() {
125        let mut f = ClientFilter::new(
126            addr("10.0.0.1:4000"),
127            &SocksAddr::from(addr("10.0.0.1:5000")),
128        );
129        assert!(!f.accept(addr("10.0.0.1:5001")));
130        assert!(!f.accept(addr("10.0.0.2:5000")));
131        assert!(f.accept(addr("10.0.0.1:5000")));
132        assert!(f.accept(addr("[::ffff:10.0.0.1]:5000")));
133    }
134
135    #[test]
136    fn an_unspecified_ip_is_the_peers() {
137        let mut f = ClientFilter::new(
138            addr("10.0.0.1:4000"),
139            &SocksAddr::from(addr("0.0.0.0:7000")),
140        );
141        assert!(!f.accept(addr("10.0.0.1:7001")));
142        assert!(f.accept(addr("10.0.0.1:7000")));
143    }
144
145    #[test]
146    fn without_a_declaration_the_first_datagram_pins_the_port() {
147        let mut f = ClientFilter::new(addr("10.0.0.1:4000"), &SocksAddr::any());
148        // Another host neither passes nor pins.
149        assert!(!f.accept(addr("10.0.0.2:6000")));
150        assert!(f.accept(addr("10.0.0.1:6000")));
151        assert!(!f.accept(addr("10.0.0.1:6001")));
152        assert!(f.accept(addr("10.0.0.1:6000")));
153    }
154
155    #[test]
156    fn a_declaration_of_another_ip_is_no_declaration() {
157        // A client behind NAT declares its private address.
158        let mut f = ClientFilter::new(
159            addr("203.0.113.1:4000"),
160            &SocksAddr::from(addr("192.168.1.2:5000")),
161        );
162        assert!(!f.accept(addr("192.168.1.2:5000")));
163        assert!(f.accept(addr("203.0.113.1:61000")));
164        assert!(!f.accept(addr("203.0.113.1:61001")));
165    }
166
167    #[test]
168    fn associations_are_bounded() {
169        let a = Arc::new(Associations::default());
170        let held: Vec<_> = (0..MAX_ASSOCIATIONS)
171            .map(|_| a.acquire().unwrap())
172            .collect();
173        assert!(a.acquire().is_none());
174        drop(held);
175        assert_eq!(a.len(), 0);
176        assert!(a.acquire().is_some());
177    }
178}
