1//! HTTP and SOCKS4a/5 on one port, as sing-box's `mixed` inbound.
2//!
3//! The first byte tells them apart: a SOCKS request starts with its version,
4//! 4 or 5, which no HTTP method does. UDP is SOCKS5's `UDP ASSOCIATE`, served
5//! as the socks inbound serves it: on a relay socket per association, not on
6//! the listen port.
7//!
8//! sing-box's `set_system_proxy` is not taken: sail does not change the
9//! host's proxy settings, and the field is an unknown one.
10

24pub(crate) fn register(registry: &mut InboundRegistry) {
25    registry.register("mixed", InboundFactory::standalone(build));
26}

28#[derive(Deserialize)]
29#[serde(deny_unknown_fields)]
30struct MixedInboundOptions {
31    /// Clients must authenticate as one of these, by SOCKS5
32    /// username/password or HTTP Basic; anyone may connect when there are
33    /// none.
34    #[serde(default)]
35    users: Vec<MixedUser>,
36}
37
38#[derive(Deserialize, Clone)]
39#[serde(deny_unknown_fields)]
40struct MixedUser {
41    username: String,
42    password: String,
43}
44
45fn build(ctx: &InboundContext<'_>) -> Result<AnyInboundHandler> {
46    let options: MixedInboundOptions = ctx.options()?;
47    // Both protocols' rules apply, as either may carry the credentials.
48    let http_users = http::inbound::users_by_name(
49        ctx.tag,
50        options
51            .users
52            .iter()
53            .cloned()
54            .map(|u| http::inbound::HttpUser {
55                username: u.username,
56                password: u.password,
57            })
58            .collect(),
59    )?;
60    let socks_users = socks::inbound::users_by_name(
61        ctx.tag,
62        options
63            .users
64            .into_iter()
65            .map(|u| socks::inbound::SocksUser {
66                username: u.username,
67                password: u.password,
68            })
69            .collect(),
70    )?;
71    let stream = Arc::new(StreamHandler {
72        http: http::inbound::StreamHandler::new(http_users),
73        socks: socks::inbound::StreamHandler::new(
74            socks_users,
75            Arc::new(socks::inbound::Associations::default()),
76        ),
77    });
78    // UDP ASSOCIATE binds a relay socket per association, as the socks
79    // inbound does; nothing is served on the listen port's UDP.
80    Ok(Arc::new(Handler::new(
81        ctx.tag.to_owned(),
82        Some(stream),
83        None,
84    )))
85}
86
87pub struct StreamHandler {
88    http: http::inbound::StreamHandler,
89    socks: socks::inbound::StreamHandler,
90}
91
92#[async_trait]
93impl InboundStreamHandler for StreamHandler {
94    async fn handle<'a>(
95        &'a self,
96        sess: Session,
97        mut stream: AnyStream,
98    ) -> std::io::Result<AnyInboundTransport> {
99        tracing::trace!("handling inbound stream");
100        let first = stream.read_u8().await?;
101        match first {
102            0x04 | 0x05 => self.socks.handle_version(sess, stream, first).await,
103            _ => {
104                self.http
105                    .handle_with_prefix(sess, stream, vec![first])
106                    .await
107            }
108        }
109    }
110}
