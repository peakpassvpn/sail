1//! The URL tests `urltest`, `load-balance` and `fallback` check their
2//! members with: every member at once, through it, every `interval`, while
3//! the group is in use.
4
5use std::sync::{Arc, Mutex, RwLock, Weak};
6use std::time::Duration;
7
8use futures::future::{abortable, AbortHandle, BoxFuture};
9use futures::FutureExt;

19/// How long one test may take, by default, before its member counts as
20/// failed.
21pub const DEFAULT_TIMEOUT: Duration = Duration::from_secs(5);
22
23/// A failed connection asks for the members to be tested again, but not
24/// more often than this.
25const MIN_RETEST: Duration = Duration::from_secs(2);
26
27pub const DEFAULT_URL: &str = "https://www.gstatic.com/generate_204";
28pub const DEFAULT_INTERVAL: Duration = Duration::from_secs(3 * 60);
29
30/// Called with the latencies after every round of tests.
31pub type OnTested = Box<dyn Fn(&[Option<Duration>]) + Send + Sync>;
32
33pub struct Checker {
34    tag: String,
35    members: Vec<AnyOutboundHandler>,
36    probe: HttpProbe,
37    dns_client: SyncDnsClient,
38    interval: Duration,
39    /// How long one test may take before its member counts as failed.
40    timeout: Duration,
41    /// Tests pause once the group has not been used for this long, and
42    /// resume, at once, when it is used again.
43    idle: Option<Duration>,
44    latencies: MemberLatencies,
45    /// Whether a round of tests has been done: before, every member is
46    /// taken to be up.
47    tested: std::sync::atomic::AtomicBool,
48    last_used: Mutex<Instant>,
49    wake: Notify,
50    on_tested: OnTested,
51    /// The test loop, until there is a runtime to spawn it on.
52    task: Mutex<Option<BoxFuture<'static, ()>>>,
53}
54
55impl Checker {
56    /// A checker of `members`, and the handle that stops its tests.
57    #[allow(clippy::too_many_arguments)]
58    pub fn new(
59        tag: &str,
60        members: Vec<AnyOutboundHandler>,
61        probe: HttpProbe,
62        dns_client: SyncDnsClient,
63        interval: Duration,
64        timeout: Duration,
65        idle: Option<Duration>,
66        on_tested: OnTested,
67    ) -> (Arc<Self>, AbortHandle) {
68        let n = members.len();
69        let checker = Arc::new(Self {
70            tag: tag.to_string(),
71            members,
72            probe,
73            dns_client,
74            interval,
75            timeout,
76            idle,
77            latencies: Arc::new(RwLock::new(vec![None; n])),
78            tested: Default::default(),
79            last_used: Mutex::new(Instant::now()),
80            wake: Notify::new(),
81            on_tested,
82            task: Mutex::new(None),
83        });
84        // The loop holds the checker weakly: a checker that is dropped
85        // before its loop was ever spawned is not kept alive by it.
86        let (task, abort_handle) = abortable(test_loop(Arc::downgrade(&checker)));
87        if let Ok(mut slot) = checker.task.lock() {
88            *slot = Some(task.map(|_| ()).boxed());
89        }
90        checker.start();
91        (checker, abort_handle)
92    }
93
94    /// Spawns the test loop, once there is a runtime: a configuration can
95    /// be built without one, to be checked.
96    fn start(&self) {
97        let Ok(runtime) = tokio::runtime::Handle::try_current() else {
98            return;
99        };
100        let task = self.task.lock().ok().and_then(|mut slot| slot.take());
101        if let Some(task) = task {
102            runtime.spawn(task);
103        }
104    }
105
106    pub fn latencies(&self) -> MemberLatencies {
107        self.latencies.clone()
108    }
109
110    /// Whether member `i` passed its last test, or none was done yet.
111    pub fn is_up(&self, i: usize) -> bool {
112        if !self.tested.load(std::sync::atomic::Ordering::Relaxed) {
113            return true;
114        }
115        self.latencies
116            .read()
117            .map(|l| l.get(i).is_some_and(Option::is_some))
118            .unwrap_or(true)
119    }
120
121    /// Notes that the group is being used, which resumes paused tests.
122    pub fn used(&self) {
123        self.start();
124        let now = Instant::now();
125        let was_idle = match self.last_used.lock() {
126            Ok(mut last) => {
127                let idle = self
128                    .idle
129                    .is_some_and(|idle| now.duration_since(*last) > idle);
130                *last = now;
131                idle
132            }
133            Err(_) => false,
134        };
135        if was_idle {
136            self.wake.notify_one();
137        }
138    }
139
140    /// Asks for the members to be tested again soon, after a connection
141    /// through one failed.
142    pub fn retest(&self) {
143        self.wake.notify_one();
144    }
145
146    fn is_idle(&self) -> bool {
147        let Some(idle) = self.idle else {
148            return false;
149        };
150        self.last_used
151            .lock()
152            .map(|last| last.elapsed() > idle)
153            .unwrap_or(false)
154    }
155
156    async fn test_all(&self) {
157        let tests = self.members.iter().map(|member| async move {
158            match tokio::time::timeout(
159                self.timeout,
160                self.probe.run(self.dns_client.clone(), member),
161            )
162            .await
163            {
164                Ok(Ok(latency)) => Some(latency),
165                Ok(Err(e)) => {
166                    debug!("[{}] test of [{}] failed: {}", self.tag, member.tag(), e);
167                    None
168                }
169                Err(_) => {
170                    debug!("[{}] test of [{}] timed out", self.tag, member.tag());
171                    None
172                }
173            }
174        });
175        let latencies = futures::future::join_all(tests).await;
176        debug!(
177            "[{}] tested: {}",
178            self.tag,
179            self.members
180                .iter()
181                .zip(&latencies)
182                .map(|(m, l)| match l {
183                    Some(l) => format!("{}({}ms)", m.tag(), l.as_millis()),
184                    None => format!("{}(failed)", m.tag()),
185                })
186                .collect::<Vec<_>>()
187                .join(" ")
188        );
189        if let Ok(mut current) = self.latencies.write() {
190            current.clone_from(&latencies);
191        }
192        self.tested
193            .store(true, std::sync::atomic::Ordering::Relaxed);
194        (self.on_tested)(&latencies);
195    }
196}
197
198async fn test_loop(checker: Weak<Checker>) {
199    loop {
200        let Some(c) = checker.upgrade() else {
201            return;
202        };
203        if !c.is_idle() {
204            c.test_all().await;
205        } else {
206            debug!("[{}] not used lately, tests paused", c.tag);
207        }
208        let tested = Instant::now();
209        let interval = c.interval;
210        // Woken early by a failure, or by use after a pause.
211        let woken = tokio::time::timeout(interval, c.wake.notified())
212            .await
213            .is_ok();
214        drop(c);
215        if woken {
216            tokio::time::sleep_until(tested + MIN_RETEST.min(interval)).await;
217        }
218    }
