1use std::io;
2
3use async_trait::async_trait;
4use tracing::Instrument;
5
6use crate::{adapter::*, session::Session};
7
8use super::plan::{Input, Kind, Plan};
9
10pub struct Handler {
11    pub actors: Vec<AnyOutboundHandler>,
12}
13
14/// What the chain is carrying between two actors.
15///
16/// A chain that ends in datagrams may still start as a stream: a datagram
17/// tunnelled over a reliable protocol is a stream until the actor that
18/// tunnels it, and there is no way back.
19enum Carried {
20    Nothing,
21    Stream(AnyStream),
22    Datagram(AnyOutboundDatagram),
23}
24
25impl Carried {
26    fn from(transport: Option<AnyOutboundTransport>) -> Self {
27        match transport {
28            Some(OutboundTransport::Stream(stream)) => Carried::Stream(stream),
29            Some(OutboundTransport::Datagram(datagram)) => Carried::Datagram(datagram),
30            None => Carried::Nothing,
31        }
32    }
33
34    fn input(&self) -> Input {
35        match self {
36            Carried::Nothing => Input::Nothing,
37            Carried::Stream(_) => Input::Stream,
38            Carried::Datagram(_) => Input::Datagram,
39        }
40    }
41}
42
43#[async_trait]
44impl OutboundDatagramHandler for Handler {
45    fn connect_addr(&self) -> OutboundConnect {
46        // What is dialled does not depend on what the chain is handed: the
47        // first actor that names an endpoint names it either way.
48        Plan::for_datagram(&self.actors, Input::Nothing).dial
49    }
50
51    /// What the *first* actor accepts, which is what an enclosing chain is
52    /// asking about when it decides whether it may convert a stream into a
53    /// datagram before reaching this one. Not what this chain produces.
54    fn transport_type(&self) -> DatagramTransportType {
55        self.actors
56            .first()
57            .and_then(|actor| {
58                actor
59                    .datagram()
60                    .ok()
61                    .map(|handler| handler.transport_type())
62            })
63            .unwrap_or(DatagramTransportType::Unknown)
64    }
65
66    async fn handle<'a>(
67        &'a self,
68        sess: &'a Session,
69        transport: Option<AnyOutboundTransport>,
70    ) -> io::Result<AnyOutboundDatagram> {
71        tracing::trace!("handling outbound datagram");
72        let mut carried = Carried::from(transport);
73        let plan = Plan::for_datagram(&self.actors, carried.input());
74
75        for stage in &plan.stages {
76            let actor = &self.actors[stage.index];
77            let sess = stage.session(sess);
78            carried = match stage.kind {
79                Kind::Datagram => {
80                    let transport = match carried {
81                        Carried::Nothing => None,
82                        Carried::Stream(stream) => Some(OutboundTransport::Stream(stream)),
83                        Carried::Datagram(datagram) => Some(OutboundTransport::Datagram(datagram)),
84                    };
85                    Carried::Datagram(
86                        actor
87                            .datagram()
88                            .map_err(|err| stage.error(err))?
89                            .handle(&sess, transport)
90                            .instrument(tracing::Span::current())
91                            .await
92                            .map_err(|err| stage.error(err))?,
93                    )
94                }
95                Kind::Stream => {
96                    let stream = match carried {
97                        Carried::Nothing => None,
98                        Carried::Stream(stream) => Some(stream),
99                        // The actor before produced a datagram and this one
100                        // speaks only streams. Nothing can be done with it,
101                        // and silently dropping it is how this used to end as
102                        // "invalid input" from an actor that was handed
103                        // nothing.
104                        Carried::Datagram(_) => {
105                            return Err(stage
106                                .failed("cannot carry the datagram the actor before it produced"))
107                        }
108                    };
109                    Carried::Stream(
110                        actor
111                            .stream()
112                            .map_err(|err| stage.error(err))?
113                            .handle(&sess, None, stream)
114                            .instrument(tracing::Span::current())
115                            .await
116                            .map_err(|err| stage.error(err))?,
117                    )
118                }
119            };
120        }
121
122        match carried {
123            Carried::Datagram(datagram) => Ok(datagram),
124            _ => Err(io::Error::other(format!(
125                "chain [{}] carries no datagrams: no actor in it turns a stream into one",
126                plan.stages
127                    .iter()
128                    .map(|stage| stage.tag.as_str())
129                    .collect::<Vec<_>>()
130                    .join(", ")
131            ))),
132        }
133    }
134}
