1//! DNS over TLS (RFC 7858): each message is prefixed with its length, on a
2//! TLS connection that is kept for the next query (RFC 7858 §3.4).
3
4use std::net::SocketAddr;
5use std::sync::Mutex;
6use std::time::{Duration, Instant};
7
8use anyhow::{anyhow, Result};
9use tokio::io::{AsyncReadExt, AsyncWriteExt};
10use tokio::time::timeout;
11use tracing::debug;
12
13use super::Upstream;
14use crate::adapter::AnyStream;
15use crate::app::dns::DnsClient;

17/// Idle connections kept per upstream: as many as queries that ran at once,
18/// up to this.
19const MAX_IDLE: usize = 4;
20/// How long a connection is kept idle. Servers close theirs after some
21/// seconds (RFC 7766 §6.2.3 suggests 10), and a connection they closed
22/// costs a failed query before a new one.
23const IDLE_TIMEOUT: Duration = Duration::from_secs(10);
24
25/// The idle connections of one upstream.
26#[derive(Default)]
27pub(super) struct Pool {
28    idle: Mutex<Vec<(AnyStream, Instant)>>,
29}
30
31impl Pool {
32    /// The connection idle for the least time, if one is still fresh.
33    pub(super) fn take(&self) -> Option<AnyStream> {
34        let mut idle = self.idle.lock().unwrap_or_else(|e| e.into_inner());
35        let now = Instant::now();
36        idle.retain(|(_, since)| now.saturating_duration_since(*since) < IDLE_TIMEOUT);
37        idle.pop().map(|(stream, _)| stream)
38    }
39
40    pub(super) fn put(&self, stream: AnyStream) {
41        let mut idle = self.idle.lock().unwrap_or_else(|e| e.into_inner());
42        if idle.len() >= MAX_IDLE {
43            idle.remove(0);
44        }
45        idle.push((stream, Instant::now()));
46    }
47}
48
49impl DnsClient {
50    pub(super) async fn exchange_dot(
51        &self,
52        upstream: &Upstream,
53        pool: &Pool,
54        addr: SocketAddr,
55        is_direct: bool,
56        request: &[u8],
57    ) -> Result<Vec<u8>> {
58        if let Some(mut stream) = pool.take() {
59            match timeout(
60                self.reused_connection_timeout(),
61                exchange(&mut stream, request),
62            )
63            .await
64            {
65                Ok(Ok(response)) => {
66                    pool.put(stream);
67                    return Ok(response);
68                }
69                Ok(Err(e)) => debug!("{}: kept connection failed: {}", upstream, e),
70                Err(_) => debug!("{}: kept connection timed out", upstream),
71            }
72        }
73        let stream = self.dial_stream(is_direct, addr).await?;
74        let tls = self.upstream_tls_client()?;
75        let mut stream: AnyStream = Box::new(
76            tls.connect(&upstream.host, stream, None, None)
77                .await
78                .map_err(|e| anyhow!("tls handshake failed: {}", e))?,
79        );
80        let response = exchange(&mut stream, request).await?;
81        pool.put(stream);
82        Ok(response)
83    }
84}
85
86/// Writes `request` and reads the answer that follows, each prefixed with
87/// its length.
88async fn exchange(stream: &mut AnyStream, request: &[u8]) -> Result<Vec<u8>> {
89    let len = u16::try_from(request.len()).map_err(|_| anyhow!("dns query too long"))?;
90    let mut buf = Vec::with_capacity(2 + request.len());
91    buf.extend_from_slice(&len.to_be_bytes());
92    buf.extend_from_slice(request);
93    stream.write_all(&buf).await?;
94    stream.flush().await?;
95    let len = stream.read_u16().await? as usize;
96    let mut response = vec![0u8; len];
97    stream.read_exact(&mut response).await?;
98    Ok(response)
99}
