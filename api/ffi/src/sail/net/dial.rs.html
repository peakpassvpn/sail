1//! How an outbound's sockets are opened: the interface or address they are
2//! bound to, their routing mark, and how long a connect may take.
3
4use std::io;
5use std::net::{IpAddr, Ipv4Addr, Ipv6Addr, SocketAddr};
6use std::time::Duration;
7
8use tracing::debug;
9
10/// The default time a TCP connect may take.
11pub const DEFAULT_CONNECT_TIMEOUT: Duration = Duration::from_secs(8);
12
13/// How the host keeps outbound sockets out of its VPN (Android).
14#[derive(Debug, Clone, PartialEq, Eq)]
15pub enum SocketProtect {
16    /// An endpoint that takes each socket's file descriptor as an int32
17    /// and answers 0 once it is protected: a Unix domain socket, by path.
18    Unix(String),
19    /// The same, over TCP.
20    Tcp(SocketAddr),
21    /// The host's `Platform::protect_socket`.
22    Platform(crate::runtime::PlatformRef),
23}
24
25/// Options for the sockets one outbound opens, already combined with the
26/// instance's defaults (`route.default_interface`, `route.default_mark`).
27#[derive(Debug, Clone, PartialEq, Eq)]
28pub struct DialOptions {
29    /// The interface to send through, by name.
30    pub bind_interface: Option<String>,
31    /// The local address for IPv4 destinations.
32    pub inet4_bind_address: Option<Ipv4Addr>,
33    /// The local address for IPv6 destinations.
34    pub inet6_bind_address: Option<Ipv6Addr>,
35    /// `SO_MARK`, Linux only.
36    pub routing_mark: Option<u32>,
37    pub connect_timeout: Duration,
38    /// From the host, never from a configuration.
39    pub protect: Option<SocketProtect>,
40    /// Whether IPv6 is used at all (`dns.strategy`), which makes the UDP
41    /// sockets that are not bound to anything in particular dual-stack.
42    pub ipv6: bool,
43}
44
45impl Default for DialOptions {
46    fn default() -> Self {
47        Self {
48            bind_interface: None,
49            inet4_bind_address: None,
50            inet6_bind_address: None,
51            routing_mark: None,
52            connect_timeout: DEFAULT_CONNECT_TIMEOUT,
53            protect: None,
54            ipv6: false,
55        }
56    }
57}
58
59impl DialOptions {
60    /// These options, with what they leave unset taken from `defaults`.
61    pub fn or(&self, defaults: &DialOptions) -> DialOptions {
62        let binds_itself = self.bind_interface.is_some()
63            || self.inet4_bind_address.is_some()
64            || self.inet6_bind_address.is_some();
65        DialOptions {
66            // An outbound bound to an address is not also bound to the
67            // default interface, which could contradict it.
68            bind_interface: self.bind_interface.clone().or_else(|| {
69                if binds_itself {
70                    None
71                } else {
72                    defaults.bind_interface.clone()
73                }
74            }),
75            inet4_bind_address: self.inet4_bind_address.or(if binds_itself {
76                None
77            } else {
78                defaults.inet4_bind_address
79            }),
80            inet6_bind_address: self.inet6_bind_address.or(if binds_itself {
81                None
82            } else {
83                defaults.inet6_bind_address
84            }),
85            routing_mark: self.routing_mark.or(defaults.routing_mark),
86            connect_timeout: self.connect_timeout,
87            protect: self.protect.clone().or_else(|| defaults.protect.clone()),
88            ipv6: defaults.ipv6,
89        }
90    }
91
92    /// The local address for a UDP socket that is not bound to anything in
93    /// particular.
94    pub fn unspecified(&self) -> SocketAddr {
95        if self.ipv6 {
96            (Ipv6Addr::UNSPECIFIED, 0).into()
97        } else {
98            (Ipv4Addr::UNSPECIFIED, 0).into()
99        }
100    }
101
102    /// The instance's defaults, as `route` sets them. What
103    /// `auto_detect_interface` finds is added at start, where the system is
104    /// asked.
105    pub fn defaults(route: &crate::config::Route) -> anyhow::Result<DialOptions> {
106        let defaults = DialOptions {
107            bind_interface: route.default_interface.clone(),
108            routing_mark: route.default_mark,
109            ..Default::default()
110        };
111        if defaults.routing_mark.is_some() && !supports_routing_mark() {
112            anyhow::bail!("route.default_mark: only supported on Linux");
113        }
114        if let Some(name) = &defaults.bind_interface {
115            if !supports_bind_interface() {
116                anyhow::bail!("route.default_interface: not supported on this platform");
117            }
118            if interface_exists(name) == Some(false) {
119                anyhow::bail!(
120                    "route.default_interface: there is no interface \"{}\"",
121                    name
122                );
123            }
124        }
125        Ok(defaults)
126    }
127
128    /// Whether any option would be applied to a socket.
129    fn binds(&self) -> bool {
130        self.bind_interface.is_some()
131            || self.inet4_bind_address.is_some()
132            || self.inet6_bind_address.is_some()
133            || self.routing_mark.is_some()
134    }
135}

137/// Applies `dial` to `socket`, which is about to talk to `target`, and
138/// returns whether that bound it to a local address.
139///
140/// A socket to a loopback address is bound to loopback and nothing else:
141/// binding it to an interface would make the destination unreachable.
142pub(crate) fn bind(
143    socket: &socket2::Socket,
144    target: &SocketAddr,
145    dial: &DialOptions,
146) -> io::Result<bool> {
147    if target.ip().is_loopback() {
148        let loopback: SocketAddr = match target {
149            SocketAddr::V4(_) => (Ipv4Addr::LOCALHOST, 0).into(),
150            SocketAddr::V6(_) => (Ipv6Addr::LOCALHOST, 0).into(),
151        };
152        socket.bind(&loopback.into())?;
153        debug!("socket bind loopback {}", loopback);
154        return Ok(true);
155    }
156    if !dial.binds() {
157        return Ok(false);
158    }
159    if let Some(iface) = &dial.bind_interface {
160        bind_interface(socket, target, iface)
161            .map_err(|e| io::Error::new(e.kind(), format!("bind to interface {}: {}", iface, e)))?;
162        debug!("socket bind {}", iface);
163    }
164    let address = match target.ip() {
165        IpAddr::V4(_) => dial.inet4_bind_address.map(IpAddr::V4),
166        IpAddr::V6(_) => dial.inet6_bind_address.map(IpAddr::V6),
167    };
168    if let Some(address) = address {
169        socket.bind(&SocketAddr::new(address, 0).into())?;
170        debug!("socket bind {}", address);
171    }
172    if let Some(mark) = dial.routing_mark {
173        set_mark(socket, mark)?;
174    }
175    Ok(address.is_some())
176}

178#[cfg(target_os = "macos")]
179fn bind_interface(socket: &socket2::Socket, target: &SocketAddr, iface: &str) -> io::Result<()> {
180    use std::os::unix::io::AsRawFd;
181    let name = std::ffi::CString::new(iface.as_bytes())
182        .map_err(|_| io::Error::new(io::ErrorKind::InvalidInput, "invalid interface name"))?;
183    let index = unsafe { libc::if_nametoindex(name.as_ptr()) };
184    if index == 0 {
185        return Err(io::Error::last_os_error());
186    }
187    let (level, option) = match target {
188        SocketAddr::V4(_) => (libc::IPPROTO_IP, libc::IP_BOUND_IF),
189        SocketAddr::V6(_) => (libc::IPPROTO_IPV6, libc::IPV6_BOUND_IF),
190    };
191    let ret = unsafe {
192        libc::setsockopt(
193            socket.as_raw_fd(),
194            level,
195            option,
196            &index as *const _ as *const libc::c_void,
197            std::mem::size_of::<libc::c_uint>() as libc::socklen_t,
198        )
199    };
200    if ret == -1 {
201        return Err(io::Error::last_os_error());
202    }
203    Ok(())
204}

232/// Whether `routing_mark` can be used on this platform, for checking a
233/// configuration before anything is dialled.
234pub fn supports_routing_mark() -> bool {
235    cfg!(any(target_os = "linux", target_os = "android"))
236}

238/// Whether the host has an interface named `name`, where that can be
239/// asked. It is not asked on Android, where interfaces come and go with the
240/// network.
241pub fn interface_exists(name: &str) -> Option<bool> {
242    #[cfg(any(target_os = "macos", target_os = "linux"))]
243    {
244        let name = std::ffi::CString::new(name).ok()?;
245        Some(unsafe { libc::if_nametoindex(name.as_ptr()) } != 0)
246    }
247    #[cfg(not(any(target_os = "macos", target_os = "linux")))]
248    {
249        let _ = name;
250        None
251    }
252}
253
254/// Whether `bind_interface` can be used on this platform.
255pub fn supports_bind_interface() -> bool {
256    cfg!(any(
257        target_os = "macos",
258        target_os = "linux",
259        target_os = "android"
260    ))
261}

267    #[test]
268    fn an_outbound_bound_to_an_address_does_not_take_the_default_interface() {
269        let defaults = DialOptions {
270            bind_interface: Some("en0".into()),
271            routing_mark: Some(1),
272            ..Default::default()
273        };
274        let own = DialOptions {
275            inet4_bind_address: Some(Ipv4Addr::new(10, 0, 0, 2)),
276            ..Default::default()
277        };
278        let combined = own.or(&defaults);
279        assert_eq!(combined.bind_interface, None);
280        assert_eq!(
281            combined.inet4_bind_address,
282            Some(Ipv4Addr::new(10, 0, 0, 2))
283        );
284        // A mark is not an address and still applies.
285        assert_eq!(combined.routing_mark, Some(1));
286    }

294        let combined = DialOptions::default().or(&defaults);
295        assert_eq!(combined.bind_interface.as_deref(), Some("en0"));
296    }
